\section{Parameterized Witness}

\begin{definition}[Parameter set]
The parameter set is
\[
  \mathcal{P}_{\ge 3}=\{n\in\mathbb{N}\mid n\ge 3\}.
\]
For each \(n\in\mathcal{P}_{\ge 3}\), \(\PFLFW_n(E)\) is a finite source
package over the directed cyclic source-index graph
\[
  C_n=(V_n,A_n),\qquad
  V_n=\mathbb{Z}/n\mathbb{Z},\quad
  A_n=\{i\to i+1\mid i\in\mathbb{Z}/n\mathbb{Z}\}.
\]
The graph is a source index only. It is not target topology, not a target
atlas, not a spacetime manifold, and not a metric scaffold.
\end{definition}

\begin{definition}[Sectors and objects]
For every \(i\in V_n\), define a sector object
\[
  S_i^{(n)}=(i,\mathsf{loc}_i,\mathsf{edge}^{-}_i,\mathsf{edge}^{+}_i).
\]
The symbols \(\mathsf{loc}_i,\mathsf{edge}^{-}_i,\mathsf{edge}^{+}_i\) are
source tags. They are not locations in target space and carry no detector
semantics.
\end{definition}

\begin{definition}[Source tokens]
Fix a nonempty finite source alphabet \(A_E\) and a distinguished bottom token
\(\bot_A\in A_E\). The token set at \(i\) is
\[
  \SrcTok_i^{(n)} =
  \{a_i,b_i,f_i,q_i\mid a_i,b_i,f_i,q_i\in A_E\}.
\]
Here \(a_i\) is a sector source tag, \(b_i\) is a source-balance tag, \(f_i\)
is an edge-flux tag, and \(q_i\) is a quality tag. These names are source
syntax only; they are not physical matter fields, not a stress-energy tensor,
and not a matter action.
\end{definition}

\begin{definition}[Source states]
The source-state set at \(i\) is
\[
  \SrcSt_i^{(n)} =
  \{(\alpha_i,\beta_i,\phi_i,\chi_i)\in A_E^4
  \mid \chi_i\in\{\mathsf{ok},\mathsf{bottom}\}\}.
\]
The full source state is \(\sigma^{(n)}=(\sigma_i)_{i\in V_n}\). A state is
successful only when every \(\chi_i=\mathsf{ok}\); otherwise the package maps
to \(\BottomSrc\) with a declared failure label.
\end{definition}

\begin{definition}[Edge-local source-interface records]
For each arrow \(i\to i+1\), define
\[
  R_i^{(n)} =
  (S_i^{(n)},S_{i+1}^{(n)}, f_i,\beta_i,\beta_{i+1},
  \mathsf{local\_check}_i).
\]
The predicate \(\mathsf{local\_check}_i\) reads only the two incident sectors,
their source tokens, and the arrow tag \(f_i\). It does not read target
geometry, target metric data, empirical matter behavior, or downstream GR
objects.
\end{definition}

\section{Morphisms, Restrictions, And Variations}

\begin{definition}[Nontrivial relabelings]
For each \(k\in\mathbb{Z}/n\mathbb{Z}\), define a relabeling
\[
  \tau_k(i)=i+k\pmod n.
\]
The transport of sectors, tokens, states, and edge records is componentwise:
\[
  \tau_k(S_i^{(n)})=S_{i+k}^{(n)},\quad
  \tau_k(a_i,b_i,f_i,q_i)=(a_{i+k},b_{i+k},f_{i+k},q_{i+k}),
\]
and similarly for source states and \(R_i^{(n)}\). The map \(\tau_1\) is a
nonidentity declared relabeling for all \(n\ge 3\).
\end{definition}

\begin{definition}[Restriction maps]
For a finite interval subgraph
\[
  H=[r,r+\ell]\subset C_n,\qquad 1\le \ell<n,
\]
the restriction map \(\Restrict_{n,H}\) keeps exactly the sectors, tokens,
states, and arrows whose support lies in \(H\), and records the two boundary
arrows as source-boundary stubs. If a relabeling sends \(H\) to
\(\tau_k(H)\), then
\[
  \Restrict_{n,\tau_k(H)}\circ \tau_k
  = \tau_k\circ \Restrict_{n,H}
\]
on the declared common source domain.
\end{definition}

\begin{definition}[Allowed finite/local variations]
An allowed variation \(\delta\in\VarSrc_n(E)\) is generated by finitely many
single-site or single-edge edits:
\[
  \delta_i^{a},\ \delta_i^{b},\ \delta_i^{f},\ \delta_i^{q},
\]
each replacing one source tag by another element of \(A_E\) or setting
\(\chi_i=\mathsf{bottom}\). The support of \(\delta\) is a finite subset of
\(V_n\cup A_n\). No variation may introduce target topology, target metric
data, detector semantics, stress-energy semantics, matter-action semantics,
benchmark fit, or field-equation structure.
\end{definition}

\section{Witness Theorem}

\begin{theorem}[Parameterized finite/local witness eligibility]
For each \(n\ge 3\), if the declared finite source alphabet \(A_E\) is
nonempty and Conditions PFLW-CAND-001 through PFLW-CAND-008 hold, then
\(\mathcal{F}^{\mathrm{par}}_{E,n}\) is a witness for \(\PFLFT(E)\) at
parameter \(n\). The family \(\PFLFW(E)\) is strictly more general than
\(F_E^\sharp=\{S_0,S_1\}\), supports a nontrivial relabeling, supports
finite/local variation tests, and is eligible only for Smuggling Auditor
review before stress testing or protected review.
\end{theorem}

\begin{proof}
For \(n\ge 3\), \(C_n\) has at least three sector objects and the cyclic
shift \(\tau_1\) is nonidentity. Definitions above supply source sectors,
source tokens, source states, edge-local records, relabelings, finite/local
variations, restriction maps, sector-uniformity, finite/locality, bottom
branches, no-target-import, and invariance ledgers. Thus the component list
matches \(\PFLFT(E)\).

The relabeling equations follow from componentwise transport around \(C_n\).
Restriction stability follows because restrictions retain exactly the source
data supported on finite interval subgraphs and boundary stubs, and cyclic
shift maps intervals to intervals. Variation stability follows because each
variation is generated by finitely many source-token or source-state edits and
either yields another declared source state or bottom.

The no-target-import and evidence-as-adoption conditions exclude the forbidden
target and promotion classes. Therefore the construction is a draft/control
source-family witness only. It supplies no source-law adoption, no coupling-law
adoption, no matter-coupling derivation, no stress-energy semantics, no
stress-energy tensor, no matter action, no Einstein equations, no benchmark
promotion, and no completed derivation.
\end{proof}
