\section{Target Choice}

P4-T01 allowed four target forms: finite-cardinality families \(F_n\),
graph/DAG-indexed finite families, category-like finite packages with
morphisms, or a hybrid graph-indexed family with finite variations. This
packet chooses the hybrid form.

\begin{definition}[Hybrid graph-indexed finite/local target]
For a source context \(E\), \(\PFLFT(E)\) is the obligation to define a finite
source package
\[
  \mathcal{F}^{\mathrm{par}}_E =
  (\Graph_E,S_E,\SrcTok_E,\SrcSt_E,R_E,\mathcal{M}_E,
   \VarSrc_E,\Restrict_E,\mathsf{SectUnif}_E,\mathsf{Loc}_E,
   \mathsf{Bot}_E,\NoTargetImport_E,\Inv_E).
\]
Here \(\Graph_E=(V_E,A_E)\) is a finite directed source-index graph or finite
DAG, \(S_E=\{S_v\}_{v\in V_E}\) is the source-sector family,
\(\SrcTok_E=\{\SrcTok_v\}_{v\in V_E}\) is the source-token family,
\(\SrcSt_E=\{\SrcSt_v\}_{v\in V_E}\) is the source-state family, \(R_E\)
records edge-local source-interface relations, \(\mathcal{M}_E\) is the
declared relabeling or morphism family, \(\VarSrc_E\) is the allowed
finite/local variation family, \(\Restrict_E\) is the family restriction
operation, \(\mathsf{SectUnif}_E\) is the sector-uniformity rule,
\(\mathsf{Loc}_E\) is the finite/locality rule, \(\mathsf{Bot}_E\) records
bottom branches, \(\NoTargetImport_E\) is the no-target-import certificate,
and \(\Inv_E\) is the invariance-test ledger.
\end{definition}

\begin{remark}[Strictly stronger than the two-sector family]
The earlier two-sector style \(F_E^\sharp=\{S_0,S_1\}\) is a degenerate
subcase obtained by taking \(|V_E|=2\), no nontrivial edge-local morphism
structure, and only the declared two-sector variation support. The target
\(\PFLFT(E)\) is strictly more general because it admits arbitrary finite
\(|V_E|\), edge-indexed source interfaces, nonidentity relabelings or
morphisms, restriction maps to finite subfamilies, and finite/local variation
tests over vertices and edges.
\end{remark}

\section{Required Definitions}

\begin{definition}[Objects and sectors]
The objects of \(\PFLFT(E)\) are the source sectors \(S_v\) indexed by
vertices \(v\in V_E\), together with edge-local source-interface records
\(R_a\) for arrows \(a:u\to v\). The graph \(\Graph_E\) is an index of source
relations only. It is not a target manifold, not target topology, not a
spacetime atlas, and not a target metric scaffold.
\end{definition}

\begin{definition}[Source tokens and source states]
Each sector \(S_v\) carries a finite or declared source-local token set
\(\SrcTok_v\) and a source-state domain \(\SrcSt_v\). A source state
\(\sigma_v\in\SrcSt_v\) may reference only declared source tokens, declared
source relations, and previously scoped source-side context. It may not use
target fields, detector semantics, empirical matter behavior, a matter action,
or Einstein-equation premises.
\end{definition}

\begin{definition}[Morphisms and relabelings]
A morphism or relabeling
\[
  m:\mathcal{F}^{\mathrm{par}}_E\to
  \mathcal{F}^{\mathrm{par}}_{E'}
\]
consists of a finite graph map \(m_V:V_E\to V_{E'}\), arrow transport
where defined, token maps \(m_{\Tok,v}:\SrcTok_v\to\SrcTok_{m_V(v)}\), and
state maps \(m_{\State,v}:\SrcSt_v\to\SrcSt_{m_V(v)}\), all preserving the
declared source-interface relations. If any part is undefined, the result is
a fail-closed relabeling failure, not an invitation to import target
covariance.
\end{definition}

\begin{definition}[Allowed finite/local variations]
An allowed variation \(\delta\in\VarSrc_E\) is a finite-support modification
of declared source tokens, source states, or edge-local source-interface
records. Its support \(\mathrm{supp}(\delta)\subseteq V_E\cup A_E\) must be
finite and declared. A variation may not add target geometry, target metric
data, stress-energy semantics, matter-action semantics, detector semantics,
benchmark fit, or field-equation structure.
\end{definition}

\begin{definition}[Family restriction maps]
For each declared finite subgraph \(H\subseteq \Graph_E\), the restriction map
\[
  \Restrict_{E,H}:\mathcal{F}^{\mathrm{par}}_E
  \longrightarrow \mathcal{F}^{\mathrm{par}}_{E|H}
\]
keeps exactly the sectors, tokens, states, edge-local records, morphism
fragments, and variations whose support lies in \(H\). Restriction must
commute with relabeling on its common domain or fail closed with a missing
morphism behavior label.
\end{definition}

\begin{condition}[Sector-uniformity rule]
There must be one source-side rule schema \(\mathcal{R}^{\mathrm{par}}\) for
all sectors and edge-local records. The rule may depend on declared source
parameters carried by vertices or arrows, but it may not be hand-tailored by
sector, by target interpretation, or by desired downstream matter behavior.
\end{condition}

\begin{condition}[Finite/locality rule]
All successful target checks must depend on a finite source neighborhood in
\(\Graph_E\). A check at vertex \(v\) may inspect only declared source data in
a bounded graph-neighborhood of \(v\), incident edge-local records, declared
restriction data, and declared finite variations. Nonlocal dependence fails
closed.
\end{condition}

\begin{definition}[Bottom and fail-closed branches]
\(\mathsf{Bot}_E\) maps every failed target condition to
\(\BottomSrc\) together with a primary obstruction label from the failure
family below. Bottom is a source-control result, not a physical field, not a
stress-energy value, and not a benchmark datum.
\end{definition}

\section{No-Target-Import Certificate}

\begin{condition}[No-target-import certificate]
The certificate \(\NoTargetImport_E\) must state that the package uses none of
the following as mathematical premises: target topology, target atlas, target
metric, Lorentzian signature, proper time, detector calibration, empirical
matter-field semantics, stress-energy tensor, matter action, Bianchi identity,
Einstein equations, benchmark fit, generated derivative content, registry
metadata authority, validator status, role authority, handoff status,
approval status, local cache state, file order, or commit status.
\end{condition}

\section{Invariance Tests}

\begin{condition}[Relabeling invariance]
For every declared relabeling or morphism \(m\), successful checks in
\(\Inv_E\) must transport along \(m\). If an output is not preserved under
declared relabeling, the target returns relabeling failure rather than
repairing the output with target covariance.
\end{condition}

\begin{condition}[Restriction stability]
For every finite subgraph \(H\subseteq\Graph_E\), target checks must be
compatible with \(\Restrict_{E,H}\) on declared common domains. Failure to
define compatible restriction behavior is missing morphism behavior.
\end{condition}

\begin{condition}[Finite-variation stability]
For each \(\delta\in\VarSrc_E\), target checks must either remain well-defined
under the declared finite/local support of \(\delta\), or return
variation-instability bottom. This is a finite/local test only and not a
global finite-variation robustness theorem.
\end{condition}

\begin{condition}[Evidence-as-adoption guard]
Accepted bridge-candidate evidence/precondition data may be cited only as
route context. If any check treats \(\MCBridgeCand(E)\), \(\SMI(E)\), bridge
certificate data, a validator PASS, a registry row, or this target
formalization as source-law adoption, coupling-law adoption, matter coupling,
stress-energy semantics, Einstein equations, benchmark promotion, or completed
derivation, the target fails closed.
\end{condition}

\section{Eligibility Theorem}

\begin{theorem}[Parameterized target eligibility without matter-coupling adoption]
Suppose a future packet supplies a package
\(\mathcal{F}^{\mathrm{par}}_E\) satisfying the definitions and conditions
above. Then the package is eligible only for a bounded P4-T02 Candidate
Constructor attempt to construct a parameterized finite/local witness
compatible with the current bridge path, or to return a precise obstruction.
It does not adopt a source law, derive or adopt matter coupling, supply
stress-energy semantics, construct a stress-energy tensor, import a matter
action, derive Einstein equations, or promote benchmark status.
\end{theorem}

\begin{proof}
The hypotheses name only finite source-index graphs, source sectors, source
tokens, source states, edge-local source-interface records, declared
relabelings or morphisms, finite/local variations, restriction maps,
sector-uniformity, finite/locality, bottom branches, a no-target-import
certificate, and invariance tests. The forbidden target and promotion classes
are excluded by the no-target-import certificate and the evidence-as-adoption
guard.

Therefore satisfying the target can at most establish that a later constructor
has a well-defined source-side test object. It cannot supply a coupling law,
matter-coupling derivation, stress-energy semantics, a stress-energy tensor,
a matter action, Einstein equations, benchmark success, or completed
derivation, because none of those objects appears among the hypotheses.
\end{proof}
