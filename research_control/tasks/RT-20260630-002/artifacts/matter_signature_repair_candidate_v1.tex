\section{Source-Signature Data}

\begin{definition}[Source-incidence repair package]
For an admissible finite/local source instance \(E\), a source-incidence repair
package is a tuple
\[
  \mathcal{S}_E=(T_E,A_E,R_E,\mathcal{R}_E,\mathcal{V}_E,\NoTarget_E)
\]
where:
\begin{enumerate}[label=(\alph*)]
\item \(T_E\) is a finite or locally finite source-token set;
\item \(A_E\) is a finite source-interface index set, read as source syntax
  only;
\item \(R_E\subseteq T_E\times A_E\) is a source-incidence relation;
\item \(\mathcal{R}_E\) is the family of permitted source relabelings
  \(\rho=(\rho_T,\rho_A)\) preserving source incidence;
\item \(\mathcal{V}_E\) is the declared finite-variation family;
\item \(\NoTarget_E\) certifies that \(T_E,A_E,R_E,\mathcal{R}_E,\mathcal{V}_E\)
  are not defined by target matter names, detector outcomes, stress-energy
  variables, matter-action variables, target metric or atlas data, benchmark
  behavior, or process authority.
\end{enumerate}
\end{definition}

\begin{definition}[Incidence signature]
For \(t\in T_E\), define the source incidence signature
\[
  \sigma_E(t)=\{a\in A_E \mid (t,a)\in R_E\}.
\]
This is a source-side signature only. It is not a physical matter property,
not a detector outcome, not stress energy, and not a matter action variable.
\end{definition}

\begin{definition}[Source-signature label set]
Let \(L_E\) be the finite set of signature labels
\[
  L_E=\{\ell_{\Sigma} \mid \Sigma\subseteq A_E
  \text{ occurs as } \sigma_E(t) \text{ for some } t\in T_E\}\cup\{\Bottom\}.
\]
The bottom label \(\Bottom\) is reserved for failed certificates, missing
incidence data, or finite-variation collapse.
\end{definition}

\section{Candidate Construction}

\begin{definition}[Matter-signature discriminator candidate]
The proposal-only repair candidate \(\MatSigDisc(E)\) is the total map
\[
  D_E:T_E\to L_E,\qquad D_E(t)=
  \begin{cases}
    \ell_{\sigma_E(t)}, & \text{if } \NoTarget_E \text{ passes and }
      \sigma_E(t) \text{ is source-defined},\\
    \Bottom, & \text{otherwise.}
  \end{cases}
\]
The associated proposal-only payload is
\[
  \SemMatterSig(E)=(T_E,A_E,R_E,L_E,D_E,\rho_L,\mathcal{V}_E,\NoTarget_E).
\]
\end{definition}

\begin{definition}[Source-defined label transport]
For a permitted source relabeling
\(\rho=(\rho_T,\rho_A):\mathcal{S}_E\to\mathcal{S}_{E'}\), define
\[
  \rho_L(\ell_{\Sigma})=\ell_{\rho_A[\Sigma]},
  \qquad
  \rho_L(\Bottom)=\Bottom.
\]
This transport is defined by the source relabeling of incidence indices, not
by preserving target matter interpretation.
\end{definition}

\begin{lemma}[Naturality of the repair candidate]
If \(\rho=(\rho_T,\rho_A)\) preserves source incidence, then
\[
  D_{E'}(\rho_T(t))=\rho_L(D_E(t))
\]
for every \(t\in T_E\) whose no-target certificate passes.
\end{lemma}

\begin{proof}
Incidence preservation means
\[
  (t,a)\in R_E \quad\Longleftrightarrow\quad
  (\rho_T(t),\rho_A(a))\in R_{E'} .
\]
Therefore
\[
  \sigma_{E'}(\rho_T(t))=\rho_A[\sigma_E(t)].
\]
By definition of \(D_E\), \(D_{E'}\), and \(\rho_L\), the required equality
follows. If the certificate fails, both sides fall closed through
\(\Bottom\) or record the corresponding obstruction label.
\end{proof}

\section{Finite/Local Non-Bottom Witness}

Let \(E_\star\) be the finite source instance
\[
  T_{E_\star}=\{u,v\},\qquad
  A_{E_\star}=\{a\},\qquad
  R_{E_\star}=\{(u,a)\},
\]
with \(\NoTarget_{E_\star}=\mathrm{pass}\) and no target interpretation
attached to \(u\), \(v\), or \(a\). Then
\[
  \sigma_{E_\star}(u)=\{a\},\qquad
  \sigma_{E_\star}(v)=\varnothing,
\]
and hence
\[
  D_{E_\star}(u)=\ell_{\{a\}},\qquad
  D_{E_\star}(v)=\ell_{\varnothing}.
\]
Both labels are non-bottom and source-defined. This supplies a finite/local
non-bottom instantiation of the proposal-only discriminator target.

\section{Finite-Variation And Fail-Closed Behavior}

\item if \(w\) preserves labels only by explicit names without source

\end{enumerate}
