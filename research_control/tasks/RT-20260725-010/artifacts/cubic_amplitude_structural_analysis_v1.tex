\section{Declared structural tests}

\begin{definition}[Patch-independence locality test]
For the present candidate family, call disjoint nonempty open sets
\(U,V\subset\mathbb T^4\) independently patchable if, for every
\(q_a,q_b\in\Qamp\), there is \(q_c\in\Qamp\) such that
\[
 q_c|_U=q_a|_U,\qquad q_c|_V=q_b|_V.
\]
This is a deliberately narrow source-side test of independent local degrees
of freedom.  Failure of this test is not a theorem that every possible source
extension is nonlocal.
\end{definition}

\begin{definition}[Three distinct composition questions]
We distinguish:
\begin{enumerate}
 \item parameter composition
 \(\Phi^\gamma_{\lambda+\mu}
 =\Phi^\gamma_\lambda\circ\Phi^\gamma_\mu\);
 \item closure of the state family under addition,
 \(q_a+q_b=q_{a+b}\);
 \item dynamical additivity,
 \(X_\gamma(q_a+q_b)=X_\gamma(q_a)+X_\gamma(q_b)\).
\end{enumerate}
Only the first two hold generally.  Parameter composition is not a substitute
for independent state composition.
\end{definition}

\section{Local patching and composition}

\begin{theorem}[Global-mode patching theorem]
\label{thm:patching}
Let \(U,V\subset\mathbb T^4\) be nonempty open sets.  If
\(q_c|_U=q_a|_U\), then \(c=a\).  Consequently, if also
\(q_c|_V=q_b|_V\), then \(a=b=c\).  Hence disjoint nonempty patches carrying
different amplitudes cannot be glued inside \(\Qamp\).
\end{theorem}

\begin{proof}
The zero set of \(e(\theta)=\sin\theta_1\) has empty interior.  Thus every
nonempty open \(U\) contains a point \(\theta\) with \(e(\theta)\neq0\).
At that point,
\[
 (c-a)e(\theta)=q_c(\theta)-q_a(\theta)=0,
\]
so \(c=a\).  Apply the same argument on \(V\) to obtain \(c=b\).
\end{proof}

\begin{obstruction}[Candidate-specific locality obstruction]
\label{obs:locality}
The P5-T03 state family fails the declared patch-independence locality test:
choose \(a\neq b\) on two disjoint nonempty open patches.  Theorem
\ref{thm:patching} rules out a glued state in \(\Qamp\).  The exact failure is
the single global amplitude shared by every patch, not a contradiction in
the forward-action law.
\end{obstruction}

\begin{proposition}[Nonlinear composition defect]
\label{prop:additivity}
Although \(\Qamp\) is closed under addition,
\[
 X_\gamma(q_{a+b})-X_\gamma(q_a)-X_\gamma(q_b)
 =-3\gamma ab(a+b)e.
 \label{eq:generator-defect}
\]
Thus dynamical additivity fails unless \(ab(a+b)=0\).  For
\(\lambda>0\), the integrated map is likewise nonadditive in general:
\[
 f^\gamma_\lambda(a+b)\neq
 f^\gamma_\lambda(a)+f^\gamma_\lambda(b).
\]
\end{proposition}

\begin{proof}
Expand \((a+b)^3-a^3-b^3=3ab(a+b)\).  The integrated claim follows, for
example, from \(a=b=1\), since
\[
 f^\gamma_\lambda(2)
 =\frac{2}{\sqrt{1+8\gamma\lambda}}
 \neq
 \frac{2}{\sqrt{1+2\gamma\lambda}}
 =2f^\gamma_\lambda(1).
\]
\end{proof}

The exact P5-T03 parameter-semigroup law remains valid.  Proposition
\ref{prop:additivity} does not invalidate that law; it denies only
superposition compatibility.  No tensor product, disjoint union, factorized
state space, or independent subsystem operation is supplied by the current
candidate, so no stronger composition property is inferred.

\section{Smooth first integrals and orbit labels}

\begin{theorem}[Triviality of global smooth first integrals]
\label{thm:first-integrals}
Let \(I:\Qamp\to\mathbb R\) be \(C^1\), and write \(i(a)=I(q_a)\).  If \(I\)
is conserved along every P5-T03 orbit, equivalently
\[
 dI_{q_a}\!\left(X_\gamma(q_a)\right)=0
 \quad\text{for all }a\in\mathbb R,
\]
then \(I\) is constant on \(\Qamp\).
\end{theorem}

\begin{proof}
The conservation condition is
\[
 -\gamma a^3 i'(a)=0.
\]
For \(a\neq0\), \(i'(a)=0\), so \(i\) is constant separately on
\((-\infty,0)\) and \((0,\infty)\).  Continuity at \(a=0\) makes the two
constants equal to \(i(0)\).
\end{proof}

The sign of \(a\) and the fixed-point stratum \(\{0\}\) label forward orbits,
but those labels do not define a nonconstant continuous global first
integral.  The P5-T03 functional \(L=a^2/2\) satisfies
\(dL/d\lambda=-\gamma a^4\), so it is monotone rather than conserved.  It is
not a physical energy, Hamiltonian, entropy, or Noether charge.

\begin{theorem}[Nonexpansion of the admitted tangent variation]
\label{thm:variation}
For an initial tangent variation \(\eta_0e\) at \(q_{a_0}\), the linearized

\[
 \eta(\lambda)
 =D f^\gamma_\lambda(a_0)\eta_0

\begin{proof}

\section{Conditional probabilistic normalization}

\begin{proposition}[Pushforward normalization and unique invariant measure]
\label{prop:probability}
If a Borel probability measure \(\mu\) on \(\mathbb R\cong\Qamp\) is added
as extra structure, every measurable map \(f^\gamma_\lambda\) preserves total
mass under pushforward.  For any fixed \(\lambda>0\), the only invariant Borel
probability measure is \(\delta_0\).
\end{proposition}

\begin{proof}
Pushforward gives
\((f^\gamma_\lambda{}_\ast\mu)(\mathbb R)
=\mu((f^\gamma_\lambda)^{-1}(\mathbb R))=\mu(\mathbb R)=1\).
For uniqueness, set
\[
 h(a)=\frac{a^2}{1+a^2}.
\]
For \(\lambda>0\),
\(h(f^\gamma_\lambda(a))<h(a)\) whenever \(a\neq0\), and equality holds at
zero.  If \(\mu\) is invariant, then
\[
 \int h(a)\,d\mu(a)
 =\int h(f^\gamma_\lambda(a))\,d\mu(a).
\]
The nonnegative difference is strictly positive off zero, so it vanishes
\(\mu\)-almost surely only when \(\mu(\{0\})=1\).  Thus
\(\mu=\delta_0\), which is invariant.
\end{proof}

The current ontology and P5-T03 candidate select no probability law,
measure, stochastic semantics, or ensemble.  Proposition
\ref{prop:probability} is conditional mathematics: normalization is automatic
after a probability measure is supplied, and the only stationary probability
for the deterministic candidate is the fixed-point mass.
