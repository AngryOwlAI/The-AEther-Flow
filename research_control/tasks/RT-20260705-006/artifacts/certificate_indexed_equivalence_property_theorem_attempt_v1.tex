\section{Theorem Target}

\begin{definition}[Selected P5-T03 target]
The selected target is the P5-T02 target
\[
  \text{transitivity under valid compatible certificate composition}.
\]
It concerns three source object records \(A,B,C\in\ObjRec\), source scopes
\(S_{AB}\) and \(S_{BC}\), valid certificates \(c_{AB}:A\to B\) and
\(c_{BC}:B\to C\), endpoint compatibility
\(\Ccod(c_{AB})=\Cdom(c_{BC})=B\), and a supplied source composition record
\[
  c_{AC}=\CComp(c_{BC},c_{AB})
\]
with \(\ValidTwo(c_{AC};A,C,S_{AB}\cup S_{BC})\).
\end{definition}

\section{Precise Assumptions}

Let \(S_{AC}=S_{AB}\cup S_{BC}\). The theorem attempt uses exactly these
premises:
\begin{enumerate}[leftmargin=*]
  \item \(A,B,C\in\ObjRec\), with declared source scopes \(S_{AB}\) and
  \(S_{BC}\).
  \item \(c_{AB},c_{BC},c_{AC}\in\CertRec\).
  \item \(\Cdom(c_{AB})=A\), \(\Ccod(c_{AB})=B\),
  \(\Cdom(c_{BC})=B\), \(\Ccod(c_{BC})=C\).
  \item \(\Cdom(c_{AC})=A\), \(\Ccod(c_{AC})=C\), and
  \(\Scope(c_{AC})=S_{AC}\).
  \item The endpoint compatibility field is source-side data:
  \(\Ccod(c_{AB})=\Cdom(c_{BC})=B\).
  \item The composition field is source-side data:
  \(c_{AC}=\CComp(c_{BC},c_{AB})\).
  \item \(\ValidTwo(c_{AB};A,B,S_{AB})\),
  \(\ValidTwo(c_{BC};B,C,S_{BC})\), and
  \(\ValidTwo(c_{AC};A,C,S_{AC})\).
  \item The no-target/no-process guard \(N_c\) in each certificate is pass.
\end{enumerate}
No premise is supplied by target topology, target atlas, target metric, proper
time, detector semantics, stress-energy semantics, matter-action semantics,

\section{Proof Attempt}

\begin{theorem}[Conditional transitivity under supplied valid composition]
Under the assumptions above,
\[
  \GenRel(A,B;S_{AB},c_{AB})\wedge
  \GenRel(B,C;S_{BC},c_{BC})
  \Longrightarrow
  \GenRel(A,C;S_{AC},c_{AC}).
\]
Consequently the transitive branch of \(\EqMSTwo\) is established only for
the explicit certificate-indexed triple
\((A,B,C;c_{AB},c_{BC},c_{AC};S_{AB},S_{BC},S_{AC})\).
\end{theorem}

\begin{proof}
By P5-T02,
\[
  \GenRel(X,Y;S,c)\quad\Longleftrightarrow\quad
  \ValidTwo(c;X,Y,S).
\]
The antecedent gives \(\GenRel(A,B;S_{AB},c_{AB})\) and
\(\GenRel(B,C;S_{BC},c_{BC})\), which by definition correspond to valid
source certificates for the two composable legs. The separate composition
premise supplies \(c_{AC}=\CComp(c_{BC},c_{AB})\) as source-side certificate
data, with compatible endpoints \(A\to B\to C\), declared scope
\(S_{AC}=S_{AB}\cup S_{BC}\), and
\(\ValidTwo(c_{AC};A,C,S_{AC})\). Applying the same P5-T02 definition to
\(c_{AC}\) yields
\[
  \GenRel(A,C;S_{AC},c_{AC}).
\]
This proves the conditional statement. The proof does not construct
\(c_{AC}\) from the two input certificates alone; it uses the supplied valid
composition record as an explicit premise.
\end{proof}

\section{Finite/Local Example From P4}

\begin{definition}[Three-step finite fixture]
Let
\[
  I=\{0,1,2\}.
\]
Define source object records \(A_0,B_0,C_0\) over carrier \(I\) with labels
\(L_A(i)=a_i\), \(L_B(i)=b_i\), \(L_C(i)=c_i\), source guards
\(\mathrm{inside}(I,i)\), and response tokens \(R_A(i)=R_B(i)=R_C(i)=r_i\).
Let
\[
  \tau_{AB}(A_0,i)=(B_0,i),\qquad
  \tau_{BC}(B_0,i)=(C_0,i),
\]
and define the composition table
\[
  \tau_{AC}(A_0,i)=\tau_{BC}(\tau_{AB}(A_0,i))=(C_0,i).
\]
\end{definition}

\begin{lemma}[Finite composition check]
The certificate records \(c_{AB}\), \(c_{BC}\), and \(c_{AC}\) with witness
tables
\[
\begin{array}{c|c|c|c}
i & \tau_{AB}(A_0,i) & \tau_{BC}(B_0,i) & \tau_{AC}(A_0,i)\\
\hline
0 & (B_0,0) & (C_0,0) & (C_0,0)\\
1 & (B_0,1) & (C_0,1) & (C_0,1)\\
2 & (B_0,2) & (C_0,2) & (C_0,2)
\end{array}
\]
satisfy the conditional theorem assumptions inside the declared finite scope
\[
  S_{AC}=I_A\cup I_B\cup I_C.
\]
\end{lemma}

\begin{proof}
The construction is the P4 finite-transport pattern applied twice, with the
P4 factorization check \(h=q\circ p\) reused as the row-wise composition
discipline. Every row is explicit and source-side. Endpoints are compatible:
\(\Ccod(c_{AB})=B_0=\Cdom(c_{BC})\). The supplied composition record
\(c_{AC}\) has witness table \(\tau_{AC}=\tau_{BC}\circ\tau_{AB}\), source
guards only, and no target import. Therefore the three validity predicates
hold in the declared finite scope and the theorem yields
\(\GenRel(A_0,C_0;S_{AC},c_{AC})\).
\end{proof}

\section{Obstruction If The Stronger Proof Fails}

\begin{obstruction}[Missing compatible composition certificate]
The unconditional claim
\[
  \GenRel(A,B;S_{AB},c_{AB})\wedge
  \GenRel(B,C;S_{BC},c_{BC})
  \Longrightarrow
  \exists c\,\GenRel(A,C;S_{AB}\cup S_{BC},c)
\]
is not proved by P5-T02. The obstruction is:
\[
  \mathrm{OB\mbox{-}P5T03\mbox{-}MISSING\mbox{-}COMPATIBLE\mbox{-}COMPOSITION\mbox{-}CERTIFICATE}.
\]
Two valid input certificates do not themselves fill the separate
composition-certificate slot. The slot may be filled only by a source-side
record \(c_{AC}\) that passes \(\ValidTwo(c_{AC};A,C,S_{AB}\cup S_{BC})\).
\end{obstruction}

\section{Minimal Countermodel Without Extra Premises}

\begin{countermodel}[Two valid legs and no supplied composite]
Let \(A,B,C\) be three finite source object records over the one-point carrier
\(\{0\}\). Let the certificate inventory contain exactly two valid certificate
records:
\[
  c_{AB}:A\to B,\qquad c_{BC}:B\to C,
\]
each with explicit source witness table \(0\mapsto 0\), matching source guard,
matching response token, and no-target guard pass. Let the inventory contain
no certificate record from \(A\) to \(C\), and in particular no record
\(c_{AC}=\CComp(c_{BC},c_{AB})\).

Then \(\GenRel(A,B;\{0\},c_{AB})\) and
\(\GenRel(B,C;\{0\},c_{BC})\) hold by \(\ValidTwo\), while
\[
  \neg\exists c\,\GenRel(A,C;\{0\},c)
\]
holds because every generated relation requires an explicit valid certificate
record and the \(A\to C\) slot is absent. Hence the unconditional
transitivity statement is false without the supplied valid composition
premise.
\end{countermodel}
