\section{Admissibility and Transition Repair}

\begin{definition}[Nontrivial admissibility]
For \(j,j'\in\coprod_{S\in\Fsrc}\Tok_E^{\sharp}(S)\), define
\[
  j\leadsto_E^{\sharp} j'
  \quad\Longleftrightarrow\quad
  \Adm_E^{\sharp}(j,j')=\mathsf{true}.
\]
The predicate \(\Adm_E^{\sharp}\) is true exactly when:
\begin{enumerate}[leftmargin=*]
  \item \(\Sem_E^{\sharp}(j)\) and \(\Sem_E^{\sharp}(j')\) are defined in
  \(\mathcal{I}_E^{\sharp}\);
  \item the update kind \(u_E(j,j')\) is a declared source update;
  \item \(\Compat_E^{\sharp}(j,j')\) records only scoped source-extension
  compatibility;
  \item every declared source transition preserves the update or returns
  \(\BottomSrc\);
  \item no target, detector, benchmark, adoption, or process-authority repair
  is required.
\end{enumerate}
It admits \(j_S^{0}\leadsto_E^{\sharp}j_S^{+}\) for each \(S\in\Fsrc\) when
the five conditions hold, and rejects
\(j_S^{0}\not\leadsto_E^{\sharp}j_S^{\mathrm{rej}}\) because
\(j_S^{\mathrm{rej}}\) is the declared forbidden-repair token.
\end{definition}

\begin{definition}[Transition transport]
For every declared \(\tau\in T_{\mathrm{src}}(E)\), define
\[
  \Trans_E^{\sharp}(\tau,j)=\tau_{\ast}^{\sharp}j
\]
only when \(\tau\) acts on source sector labels, support syntax, update kind,
and scoped compatibility class. If \(\tau\) requires target-atlas covariance,
target inverse repair, target metric compatibility, detector repair, or
benchmark repair, \(\Trans_E^{\sharp}\) returns \(\BottomSrc\) or
\(\OBMC\)-CRIT-TRANSITION.
\end{definition}

\begin{lemma}[Criteria-aligned transition coherence]
If \(j\leadsto_E^{\sharp} j'\) and \(\tau\) is a declared source transition
whose repaired transport is defined, then
\[
  \tau_{\ast}^{\sharp}j
  \leadsto_{\tau E}^{\sharp}
  \tau_{\ast}^{\sharp}j'.
\]
If the transported relation cannot be evaluated using source records alone,
the branch fails closed.
\end{lemma}

\begin{proof}
The definition of \(\Adm_E^{\sharp}\) requires source transition preservation
as a condition of admissibility. The transport \(\Trans_E^{\sharp}\) acts only
on source sector labels, support syntax, update kind, and scoped compatibility
class. Therefore a defined transport preserves the admissibility conditions.
If any required component is absent or target-repaired, \(B_E^{\sharp}\) and
\(O_E^{\sharp}\) apply.
\end{proof}
