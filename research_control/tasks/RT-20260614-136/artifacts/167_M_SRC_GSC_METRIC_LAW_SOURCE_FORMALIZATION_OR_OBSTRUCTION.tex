\section{Required Conditions}

\begin{condition}[ML1: source-only definability]
Every non-bottom premise and every non-bottom output of \(\MetricLaw\) must
factor through source-side data in \(D_{\MetricLaw}(E)\). No target topology,
target atlas, target transition map, target metric, Lorentzian signature,
proper-time semantics, detector semantics, benchmark success, generated

local cache, file order, or commit status may enter as a mathematical premise.
\end{condition}

\begin{condition}[ML2: orientation or signature selection]
\(\Sigma_{\mathrm{src}}\) must be determined by source-side tokens, response
data, or an explicit source-side gauge. If source data do not determine this
selector, the branch returns
\(\mathrm{OB\mbox{-}GEFF\mbox{-}SIGNATURE\mbox{-}UNDERDETERMINED}\).
\end{condition}

\begin{condition}[ML3: scale or measure selection]
\(\Lambda_{\mathrm{src}}\) must determine interval scale, measure, or a stated
source-side scale gauge. If only target proper time, benchmark fit, or
empirical detector calibration fixes scale, the branch returns
\(\mathrm{OB\mbox{-}GEFF\mbox{-}SCALE\mbox{-}UNDERDETERMINED}\) or
\(\mathrm{OB\mbox{-}GEFF\mbox{-}TARGET\mbox{-}METRIC\mbox{-}IMPORT}\).
\end{condition}

\begin{condition}[ML4: causal preorder or cone analogue]
\(C_{\mathrm{src}}\) must be a source preorder, relation, cone analogue, or
explicitly weaker causal-response placeholder. If no source-side relation can
be defined, the branch returns
\(\mathrm{OB\mbox{-}GEFF\mbox{-}CAUSAL\mbox{-}CONE\mbox{-}UNDERDETERMINED}\).
\end{condition}

\begin{condition}[ML5: source transition covariance]
\(\mathrm{Cov}_{\tau}^{\mathrm{met}}\) must state how
\(\MetricData(E)\) is invariant or covariant under source transition tokens on
overlaps. If the law depends on chart naming or target transition maps, the
branch returns
\(\mathrm{OB\mbox{-}GEFF\mbox{-}TRANSITION\mbox{-}COVARIANCE\mbox{-}FAILURE}\).
\end{condition}

\begin{condition}[ML6: nondegeneracy or controlled degeneracy]
\(N_{\mathrm{src}}\) must define a source-side nondegeneracy condition or a
controlled degeneracy branch. Uncontrolled collapse returns
\(\mathrm{OB\mbox{-}GEFF\mbox{-}NONDEGENERACY\mbox{-}FAILURE}\).
\end{condition}

\begin{condition}[ML7: finite-variation compatibility]
\(\mathrm{VarML}_{\mathrm{src}}\) must state whether the metric-law data are
stable, repaired, or fail-closed under the finite-variation discipline
available inside the scoped \(\Msrc\) boundary. Fragility returns
\(\mathrm{OB\mbox{-}GEFF\mbox{-}FINITE\mbox{-}VARIATION\mbox{-}FRAGILITY}\).
\end{condition}

\begin{condition}[ML8: uniqueness or selector discipline]
If multiple incompatible metric-law data packages satisfy ML1--ML7 with no
source-only selector, the branch returns
\(\mathrm{OB\mbox{-}GEFF\mbox{-}METRIC\mbox{-}LAW\mbox{-}NONUNIQUE}\).
\end{condition}

law interface whose non-bottom output names the missing selector and
covariance data. Each branch that cannot be source-defined returns
\(\BottomSrc\) or an \(\mathrm{OB\mbox{-}GEFF}\) obstruction. This proves only
that the interface is formulable as proposal-only law data; it does not prove
that the law is adopted or that all selectors exist.
\end{proof}
