\section{Positive character and common associated line}

\begin{definition}[Positive source character]
For each source state $x$, define
\[
\chi_x(\gamma)=\exp\!\bigl(\mu(x)\ell(\gamma)\bigr),
\qquad \gamma\in\Gsrc.
\]
\end{definition}

\begin{theorem}[Positive-character associated-line descent]
\label{thm:character}
For every finite $x\in\Csrc$, $\chi_x:\Gsrc\to\R_{>0}$ is a groupoid
character.  It therefore defines an oriented one-dimensional representation
line $\Lchi$ over the objects of $\Gsrc$.
\end{theorem}

\begin{proof}
The identity law follows from $\ell(\mathrm{id}_i)=0$.  Inversion follows from
$\ell(\gamma^{-1})=-\ell(\gamma)$.  For composable arrows,
\begin{align*}
\chi_x(\eta\circ\gamma)
&=\exp\!\bigl(\mu(x)[\ell(\eta)+\ell(\gamma)]\bigr)\\
&=\chi_x(\eta)\chi_x(\gamma).
\end{align*}
Every value is strictly positive, so multiplication by $\chi_x(\gamma)$ is an
orientation-preserving line isomorphism.  These maps define $\Lchi$.
\end{proof}

The oriented fundamental cycle has total winding $3$.  Since $\mu(x_*)=1$,
its $\chi$-holonomy at the base root is $e^3$, not the identity.  This gives a
nontrivial flat scale representation on the source cycle.  It is still only a
formal source line: it has no licensed physical-unit, causal, temporal, or
metric interpretation.

\section{Declared-sector lines and exact intertwiners}

Let $S_{\mathrm{decl}}=\{R,S,D\}$.  These are formal labels only.  Declare the
vertex $0$ to be the source base object.  Give the common line and each sector

(k_{S,0},k_{S,1},k_{S,2})=(0,-2,-1),\quad
(k_{D,0},k_{D,1},k_{D,2})=(0,3,-1).
\]
For $s\in S_{\mathrm{decl}}$, define
\[
a_s(i;x)=\exp\!\bigl(k_{s,i}\mu(x)\bigr)>0
\]
and the sector representation
\begin{equation}
\rho_{s,x}(\gamma)
=a_s(i;x)^{-1}\chi_x(\gamma)a_s(j;x),
\qquad \gamma:i\to j.
\label{eq:sectorrep}
\end{equation}

\begin{lemma}[Sector representation law]
Each $\rho_{s,x}$ is a positive one-dimensional representation of $\Gsrc$.
\end{lemma}

\begin{proof}
For $\gamma:i\to j$ and $\eta:j\to k$, substitution of
\eqref{eq:sectorrep} cancels the middle factors:
\[
\rho_{s,x}(\gamma)\rho_{s,x}(\eta)
=a_s(i;x)^{-1}\chi_x(\gamma)\chi_x(\eta)a_s(k;x)
=\rho_{s,x}(\eta\circ\gamma).
\]
Identity and inverse laws follow in the same way.  Every factor is positive.
\end{proof}

\begin{theorem}[Cochain intertwiners]
\label{thm:intertwiner}
Let $L_s$ be the oriented line representation defined by $\rho_{s,x}$.  The
fiber maps
\[
j_{s,i}:(L_s)_i\longrightarrow (\Lchi)_i,
\qquad j_{s,i}(e_{s,i})=a_s(i;x)e_i,
\]
are invertible natural intertwiners.  Equivalently, for every
$\gamma:i\to j$,
\begin{equation}
a_s(i;x)\rho_{s,x}(\gamma)=\chi_x(\gamma)a_s(j;x).
\label{eq:intertwine}
\end{equation}
Pairwise sector ratios are positive coboundaries.
\end{theorem}

\begin{proof}
Equation \eqref{eq:intertwine} follows immediately from
\eqref{eq:sectorrep}; positivity makes every $j_{s,i}$ invertible.  Dividing
the formulas for sectors $s$ and $t$ gives
\[
\frac{\rho_{s,x}(\gamma)}{\rho_{t,x}(\gamma)}
=\frac{(a_s/a_t)(j;x)}{(a_s/a_t)(i;x)},
\]
the coboundary of the positive object cochain $a_s/a_t$.
\end{proof}

\begin{proof}

diagonal constant.  The displayed exponents all obey $k_{s,0}=0$, hence both
anchored solutions obey $a_s(0;x)=\widetilde a_s(0;x)=1$ and $c_s=1$.
\end{proof}

\section{Nonvacuous admissible variations}

Let
\[
v=(1,-1,-1,1)^{\mathsf T}.
\]
The two rows of $B$ are linearly independent, while
\[
Bd=0,\qquad Bv=0,
\]
and $d,v$ are independent.  Hence
\[
\ker B=\operatorname{span}\{d,v\},\qquad
\Asrc(x):=\ker(DE_{\mathrm{src}}|_x)/\operatorname{span}\{d\}
\cong\R[v].
\]

\begin{theorem}[Nonvacuous integrable source variation]
\label{thm:variation}
$\Asrc$ is one-dimensional and nonzero.  The flow
\[
\operatorname{Flow}_{A}(x,t)=x+t v
\]
integrates its nonzero class globally on $\Csrc$.  Starting at $x_*$, it stays
inside $(\R_{>0})^4$ precisely for $-2<t<1$.
\end{theorem}

\begin{proof}
The kernel and quotient statements follow from the rank and independence
calculation above.  Since $Bv=0$,
$E_{\mathrm{src}}(x+tv)=E_{\mathrm{src}}(x)$ for every $t$.  Addition gives
the flow composition and inverse laws.  At the base root,
\[
x_*+tv=(2+t,1-t,1-t,2+t),
\]
whose four entries are positive exactly when $t>-2$ and $t<1$.
\end{proof}

The variation is not diagonal: $v$ is not proportional to $d$.  It is also
relevant to the descent data rather than living in an unused auxiliary factor.
Indeed,
\[
D\mu(v)=2,
\]
so the logarithm of the cycle holonomy varies at rate $3D\mu(v)=6$.  For a
generator $\gamma:i\to j$,
\[
\frac{d}{dt}\log\rho_{s,x+tv}(\gamma)
=2\bigl(\ell(\gamma)+k_{s,j}-k_{s,i}\bigr).
\]
On $e_{01}$ the three rates are
\[
(4,-2,8)_{R,S,D},
\]
which are distinct and nonzero.

\begin{theorem}[Non-diagonal source flow through common-character descents]
\label{thm:flowdescent}
Along $x(t)=x+tv$, the common character and all three sector transition
families vary nontrivially, yet the intertwining identity
\eqref{eq:intertwine} holds for every $t$.
\end{theorem}

\begin{proof}
Nontriviality follows from the rates just computed.  The formulas defining
$\chi_{x(t)}$, $a_s(i;x(t))$, and $\rho_{s,x(t)}$ are fixed before evaluation.
Substituting $x(t)$ into the algebraic proof of Theorem
\ref{thm:intertwiner} proves the identity for every $t$.  The admissible class
was obtained from $B$ and $d$, not by solving for the stabilizer of that
identity.
\end{proof}
