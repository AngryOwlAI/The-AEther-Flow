\section{Finite Source-Variation Domain}

\begin{definition}[Finite variation records]
The proposal-only variation domain \(\VarDom\) consists of records
\[
  v =
  (
    \Delta Q,
    \Delta\!\sim_{\mathrm{src}},
    \Delta\mathsf{Supp},
    \Delta\Tau,
    \Delta r,
    \sigma
  ),
\]
where \(\Delta Q\) is a finite source-support change,
\(\Delta\!\sim_{\mathrm{src}}\) is a finite quotient-relation preservation or
refinement record, \(\Delta\mathsf{Supp}\) is a finite source-chart support
transport record, \(\Delta\Tau\) is a finite source transition-token transport
record, \(\Delta r\) is a finite \(\RespLC\) readout-token transport record,
and \(\sigma\) is a source-scope tag.
\end{definition}

\begin{definition}[Admissible source variations]
For \(v\in\VarDom\) and a draft/control \(\RegSold\) certificate \(C\), define
\[
  \AdmVar(v,C) \in \mathbb{B}_{\Bot}.
\]
\(\AdmVar(v,C)=\mathsf{true}\) only when all of the following source-side
conditions are supplied:
\begin{enumerate}
  \item finite support of the variation is explicitly named;
  \item quotient classes are preserved or refined by a source relation;
  \item chart-support transport is given without target-coordinate
        interpretation;
  \item transition-token transport includes inverse-token and cocycle-token
        checks where those checks were part of \(C\);
  \item soldering transport either preserves attachment uniqueness or gives a
        source equivalence relation for nonunique attachment; and
  \item no forbidden premise is used.
\end{enumerate}
\(\AdmVar(v,C)=\Bot\) if any branch requires target geometry or process
authority. \(\AdmVar(v,C)=\mathsf{false}\) if source data are present but
explicitly fail a stated admissibility condition.
\end{definition}

\begin{definition}[Preservation outcomes]
For an admissible variation \(v\) and a \(\RegSold\) certificate \(C\), define
\[
  \PresVar(v,C)
  \in
  \{
    \mathsf{preserved},
    \mathsf{transformed\_equivalent},
    \mathsf{failed},
    \Bot
  \}.
\]
The outcome is \(\mathsf{preserved}\) when the same source certificate
families remain valid. It is \(\mathsf{transformed\_equivalent}\) when
transported source certificates satisfy the same obligations under an
explicit source equivalence. It is \(\mathsf{failed}\) when source data show a
defined preservation failure. It is \(\Bot\) when the packet lacks source data
or a forbidden premise is needed.
\end{definition}

\section{No-Target-Import Lemma}

\begin{lemma}[Packet-interface no-target-import]
Every non-bottom certificate produced by \(\FVR\), as defined in this packet,
depends only on \(\mathcal{S}_{\mathrm{src}}\), a draft/control \(\RegSold\)
certificate, and explicit source-extension assumptions.
\end{lemma}

\begin{proof}
Each component of \(\FVR\) has a source-domain signature. The variation domain
\(\VarDom\) contains only finite source variation records. The predicate
\(\AdmVar\) reads a source variation record and a draft/control \(\RegSold\)
certificate. The transport maps read only source separation, source chart
support, source transition-token, and source soldering certificate data.
\(\PresVar\) returns preservation, transformed equivalence, defined failure,
or \(\Bot\) from those source records. The forbidden premises are not elements
of \(\mathcal{S}_{\mathrm{src}}\). If any forbidden premise is required, the
\(\BottomSrc\) rule returns \(\Bot\). Therefore a non-bottom certificate
cannot depend on target topology, target atlas, target coordinate transition,
target metric, empirical semantics, benchmark success, or process authority
under this packet interface.
\end{proof}

This lemma proves only the packet-interface guard. It does not prove that
non-bottom robustness certificates exist for arbitrary source variations.

\section{Failure Branches}

\begin{branch}[Unsupported merge]
If a finite source variation merges quotient classes without source-side
separation transport, return \(\Bot\) or route to a scoped obstruction.
\end{branch}

\begin{branch}[Incompatible soldering]
If readout-token attachment changes without a source equivalence relation,
return \(\Bot\) or route to soldering-law refinement.
\end{branch}

\begin{branch}[Transition-token degeneration]
If inverse-token or cocycle-token checks fail under variation, record
\(\mathsf{failed}\) and route to Refuter stress or scoped obstruction.
\end{branch}

\begin{branch}[Forbidden import]
If a branch requires target topology, target atlas, target coordinate
transition, target metric, proper time, empirical detector semantics,
benchmark success, validation authority, registry authority, role authority,
handoff authority, generated derivatives, or file order, return \(\Bot\) and
route to Smuggling Auditor failure.
\end{branch}

\begin{branch}[Finite or local scope only]
If preservation holds only for listed finite/local cases, preserve that result
as source-extension data. Do not infer arbitrary finite-variation robustness,
source-law adoption, or \(\Msrc\) adoption.
\end{branch}

\begin{branch}[Source definition failure]
If the law cannot be source-defined, route to scoped no-go refinement or route
freeze. Do not claim future source-extension impossibility without a separate
no-go theorem.
\end{branch}
