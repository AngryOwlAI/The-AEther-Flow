\begin{abstract}
This draft/control artifact supplies the narrowest source-family evidence
allowed by \texttt{handoff-0046}. It declares a source-only family generated
by admissible relabelings of the registered manifest object and proves
relation properties only inside that relabeling orbit. The resulting
relation is \(\EqOrb\), not general \(\EqSrc\). Negative controls remain
retained for annotation weakening, unmapped proxy use, and non-orbit object

inside the orbit and cannot erase adverse behavior. General \(\EqSrc\),

remain blocked.
\end{abstract}

\section{Declared Source-Object Family}

\begin{definitionblock}[Admissible relabeling]
Let \(X_{0}\) denote the manifest-bound source object retained by
\texttt{RT-20260613-026}. An admissible relabeling \(v\in\Adm\) is a
bijection on manifest identifiers that preserves all of the following
source-only structure:
\begin{enumerate}
    \item token class incidence for \(\SrcTok\);
    \item forbidden-channel orientation for \(\Forbid\);
    \item proxy-class coverage for \(\Proxy\);
    \item annotation content for \(\Annot\);
    \item influence-edge incidence for \(\Influence\);
    \item no-closed-world discipline for absent influence edges; and
    \item the local adjudicator shape used by the retained
    \(\EqIso\) datum.
\end{enumerate}
No component of \(v\) may be chosen from target success, target metric
behavior, target observer protocols, or proxy selection favorable to a
benchmark.
\end{definitionblock}

\begin{definitionblock}[Relabeling orbit family]
The declared source-object family is
\[
    \ObjOrb=\{v(X_{0})\mid v\in\Adm\}.
\]
This is a source-family in the limited sense that every member is a
manifest-preserving presentation of the same source object under an
admissible identifier bijection. It is not a nontrivial family of distinct
source hypotheses, source dynamics, or physical configurations.
\end{definitionblock}

\begin{definitionblock}[Orbit relation]
For \(X_{a},X_{b}\in\ObjOrb\), define
\[
    X_{a}\,\EqOrb\,X_{b}
    \quad\Longleftrightarrow\quad
    \exists v_{a},v_{b}\in\Adm:
    X_{a}=v_{a}(X_{0}),\ X_{b}=v_{b}(X_{0}),
    \text{ and } v_{b}\circ v_{a}^{-1}\in\Adm .
\]
The comparison receipt is
\[
    \CmpOrb(X_{a},X_{b};\InvOrb)=\EqOrb(X_{a},X_{b}),
\]
where \(\InvOrb\) is the invariant ledger consisting exactly of the seven
preserved source-only structures listed above.
\end{definitionblock}

\section{Witness Ledger}

\begin{ledger}[Positive orbit witnesses]
For any \(X_{a},X_{b}\in\ObjOrb\), the pair
\((X_{a},X_{b})\) is an \(\EqOrb\) witness when the linking map
\(v_{b}\circ v_{a}^{-1}\) is admissible. The registered local pair
\((X_{0},X_{1})\) from \texttt{RT-20260613-026} is one instance of this
schema.
\end{ledger}

\begin{ledger}[Negative-control retention]
The following cases are retained as \(\Block\):
\begin{enumerate}
    \item \((X_{0},X_{\mathrm{ann}})\), where annotation content is weakened
    or emptied rather than preserved;
    \item \((X_{0},X_{\mathrm{proxy}})\), where an unmapped proxy is added or
    proxy-class coverage is altered; and
    \item \((X_{0},Y)\), where \(Y\) requires a change of source content,
    proxy inventory, annotation evidence, influence-edge coverage, or
    manifest scope not induced by an admissible relabeling.
\end{enumerate}
Such cases are outside \(\ObjOrb\). They cannot be converted into positive
orbit witnesses by notation choice.
\end{ledger}

\section{Source-Only Relation-Property Receipts}

\begin{receipt}[Reflexivity inside the orbit]
For every \(X_{a}=v_{a}(X_{0})\in\ObjOrb\), the identity map on the
presentation \(X_{a}\) preserves the seven source-only structures. Hence
\(X_{a}\,\EqOrb\,X_{a}\).
\end{receipt}

\begin{receipt}[Symmetry inside the orbit]
If \(X_{a}\,\EqOrb\,X_{b}\), then the linking map
\(w=v_{b}\circ v_{a}^{-1}\) is admissible. Because admissible relabelings
are bijections that preserve the same source-only structures in both
directions, \(w^{-1}=v_{a}\circ v_{b}^{-1}\) is admissible. Hence
\(X_{b}\,\EqOrb\,X_{a}\).
\end{receipt}

\begin{receipt}[Transitivity inside the orbit]
If \(X_{a}\,\EqOrb\,X_{b}\) and \(X_{b}\,\EqOrb\,X_{c}\), then
\(v_{b}\circ v_{a}^{-1}\in\Adm\) and
\(v_{c}\circ v_{b}^{-1}\in\Adm\). The composition
\[
    (v_{c}\circ v_{b}^{-1})\circ(v_{b}\circ v_{a}^{-1})
    = v_{c}\circ v_{a}^{-1}
\]
also preserves the seven source-only structures. Hence
\(X_{a}\,\EqOrb\,X_{c}\).
\end{receipt}

\begin{receipt}[No target receipt used]
The receipts above use only identity, inverse, and composition of
manifest-preserving identifier bijections. No target metric, target
proper-time, target locality, target observer protocol, benchmark-success
rank, or proxy choice favorable to a benchmark is used.
\end{receipt}

\section{Quotient and Downstream Limits}

\begin{limit}[Orbit quotient only]
The quotient \(\ObjOrb/{\EqOrb}\) may identify source-neutral
presentations of the same registered manifest object. It cannot identify
annotation weakening, unmapped proxy use, influence-edge perturbation,
manifest-scope expansion, or non-orbit source-object variation. Therefore
the quotient is presentation-local only.
\end{limit}

\begin{limit}[No general EqSrc discharge]
This packet does not prove \(\EqSrc\) over an independently declared
source-object family. It proves only that admissible relabelings form an
equivalence relation on the relabeling orbit of \(X_{0}\). General
\(\EqSrc\) remains in controlled pause.
\end{limit}

\begin{limit}[No downstream use]
Because the family is only a relabeling orbit, it supplies no retention
theorem, no generator theorem, no candidate construction rule, no
benchmark-success evidence, and no Gate Chair-ready result. \(\RetainH\),
\(\GenH\), candidate reconstruction, canonical ontology edit, benchmark
promotion, and completed-derivation language remain blocked.
\end{limit}
