\section{Finite Certificate-Fiber Domain}

\begin{definition}[Certificate fiber]
Let \(C_0=C_+\).  The finite certificate fiber
\[
  \mathcal F_{\Gamma}(C_0;E,\Bcur)\subset \MSAdm(E;\Bcur)
\]
contains exactly those admissible records \(C=(S_E,P_E,R_E,Q_E,\Gamma_E)\)
whose source support \(S_E\), probe-token family \(P_E\), and no-target
certificate bundle \(\Gamma_E\) agree with \(C_0\), or are transported from
\(C_0\) by an explicit source-side identity transport of the certificate
bundle.  The fiber ignores no certificates: \(\NoDetector\), \(\NoSE\),
\(\NoAction\), and \(\NoTarget\) must remain present and unchanged.
\end{definition}

\begin{definition}[Certificate-fiber relation]
For \(C_1,C_2\in\MSAdm(E;\Bcur)\), define
\[
  C_1 \mathrel{\MSCertEq} C_2
\]
if and only if both records are in \(\mathcal F_{\Gamma}(C_0;E,\Bcur)\) and
their source support, probe-token family, and no-target certificate bundle
agree or are related by the explicit identity transport above.  The relation
does not compare target detector readouts, target stress-energy semantics,
matter actions, target metric structure, proper time, benchmark behavior, or
Einstein-equation premises.  The readout values \(R_E\) and response markers
\(Q_E\) are treated only as uninterpreted source bookkeeping labels inside
this finite certificate fiber.
\end{definition}

\begin{definition}[Quotient projection]
Define
\[
  \pi^{\mathrm{cert}}_E:\MSAdm(E;\Bcur)\to
  \mathcal F_{\Gamma}(C_0;E,\Bcur)/{\MSCertEq}\cup\{\bot\}
\]
by sending records in the certificate fiber to their \(\MSCertEq\)-class and
sending all records outside the stated fiber to \(\bot\).  Define
\(D^q_E(C)=\pi^{\mathrm{cert}}_E(C)\) on the same domain.  This is a quotient
candidate, not a physical discriminator and not a coupling law.
\end{definition}

\section{Proof Checks}

\begin{proposition}[Equivalence on the finite fiber]
\(\MSCertEq\) is reflexive, symmetric, and transitive on
\(\mathcal F_{\Gamma}(C_0;E,\Bcur)\).
\end{proposition}

\paragraph{Proof.}
Reflexivity follows because each record has the same support, probe-token
family, and certificate bundle as itself.  Symmetry follows because equality
and identity certificate transport are symmetric in this bounded fiber.
Transitivity follows because equality and identity transport of \(S_E\),
\(P_E\), and \(\Gamma_E\) compose inside the fiber.  No target metric,
detector semantics, stress-energy semantics, matter action, benchmark
behavior, or Einstein-equation premise is used.

\begin{proposition}[Pressure-pair quotient]
For the RT-035 pressure pair,
\[
  C_+ \mathrel{\MSCertEq} C_-,
  \qquad
  \pi^{\mathrm{cert}}_E(C_+)=\pi^{\mathrm{cert}}_E(C_-).
\]
\end{proposition}

\paragraph{Proof.}
RT-035 records \(C_+\) and \(C_-\) with the same source support \(S_E\), the
same probe-token family \(P_E\), and the same no-target certificate bundle
\(\Gamma_0\).  They differ only in source readout values and response-marker
syntax.  By definition of \(\MSCertEq\), such records lie in the same
certificate fiber and are identified by the quotient projection.

\begin{proposition}[Certificate preservation and fail-closed boundary]
If \(C_1\mathrel{\MSCertEq}C_2\), the no-target certificate bundle is
preserved.  If either record falls outside the declared certificate fiber,
\(\pi^{\mathrm{cert}}_E\) returns \(\bot\).
\end{proposition}

\paragraph{Proof.}
Certificate-bundle equality or explicit identity transport is a defining
condition of \(\MSCertEq\).  Records outside the fiber do not satisfy the
domain predicate and are mapped to \(\bot\) by definition.
