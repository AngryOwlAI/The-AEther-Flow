\begin{definition}[Source balance predicate]
For each \(S\in F_E^\sharp\), define \(\BalSrc_E\) by
\[
  \BalSrc_E(j,x)=
  \begin{cases}
  \mathsf{accept},&
  j=j_S^0,\ x=(S,\sigma_S^0,u_S^+,\tau_{\mathrm{id}},\ell_S^+),\\
  \mathsf{reject},&
  j=j_S^{\mathrm{rej}}\ \text{or}\ x\notin \mathsf{SrcState}_E,\\
  \mathsf{undetermined},&\text{otherwise.}
  \end{cases}
\]
The accepting and rejecting branches are source-intrinsic branches. They do
not use target stress-energy, matter action, detector semantics, or empirical
matter behavior.
\end{definition}

\begin{definition}[Source flux ledger]
Define \(\FluxSrc_E\) as the finite ledger
\[
  \FluxSrc_E(S,j,\tau)=
  (S,\tau,j,\delta_S(j,\tau),\lambda_S(j,\tau)),
\]
where \(\delta_S\) is the declared source-token update and
\(\lambda_S\in\{\mathsf{preserve},\mathsf{modify},\mathsf{bottom}\}\) is the
source transition verdict. For the declared local transition
\(\tau_{\mathrm{id}}\),
\[
  \lambda_S(j_S^0,\tau_{\mathrm{id}})=\mathsf{preserve},
  \qquad
  \lambda_S(j_S^{\mathrm{rej}},\tau_{\mathrm{id}})=\mathsf{bottom}.
\]
If any flux entry requires target geometry or empirical matter semantics, the
ledger returns \(\BottomSrc\) with \(\OBMCSEI\)-FLUX-FAIL or
\(\OBMCSEI\)-TARGET-IMPORT.

\section{Finite/Local Witness Checks}

\begin{lemma}[Nonempty source-side interface output]
For each \(S\in F_E^\sharp\), let
\[
  x_S^+=(S,\sigma_S^0,u_S^+,\tau_{\mathrm{id}},\ell_S^+)
\]
with \(\NoTargetImport_E(j_S^0,x_S^+)=\mathsf{pass}\). Then
\[
  \SEICand(E,F_E^\sharp)(j_S^0,x_S^+)=i_E(j_S^0,x_S^+)
  \neq \BottomSrc .
\]
\end{lemma}

\begin{proof}
The declared state uses only source-sector, support, update, transition, and
local label data. By definition \(\BalSrc_E(j_S^0,x_S^+)=\mathsf{accept}\)
and \(\FluxSrc_E(S,j_S^0,\tau_{\mathrm{id}})\neq\BottomSrc\). The accepted
recovery-bridge evidence is used only as a source-side compatibility
precondition, the scoped \(\geff\) flag is boundary-preserving only, and the
no-target-import scan passes by hypothesis. Therefore the non-bottom branch
of \(\SEICand\) applies.
\end{proof}

\begin{lemma}[Rejected and forbidden branches fail closed]
If \(j=j_S^{\mathrm{rej}}\), if \(x\notin\mathsf{SrcState}_E\), if
\(\BalSrc_E(j,x)\neq\mathsf{accept}\), if
\(\FluxSrc_E(S,j,\tau)=\BottomSrc\), if bridge evidence is read as matter
coupling, if scoped \(\geff\) is read as \(\MetricData(E)\), or if
\(\NoTargetImport_E(j,x)\neq\mathsf{pass}\), then
\[
  \SEICand(E,F_E^\sharp)(j,x)=\BottomSrc
\]
with a label from \(\mathcal{O}_E^{\mathrm{sei}}\).
\end{lemma}

\begin{proof}
Each listed case negates one of the conjuncts in the non-bottom branch of
\(\SEICand\). The definition therefore takes the bottom branch. The
obstruction map is total on these failure classes by the table below.
\end{proof}
