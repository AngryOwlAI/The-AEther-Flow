\section{Exact Closed Typed Source Signature}

\begin{definition}[Rank-0 source carriers and comparisons]
Let \(\mathcal F_{\mathrm{src}}\) be a declared set of source objects. For
every \(A,B\in\mathcal F_{\mathrm{src}}\), declare:
\begin{enumerate}[leftmargin=*]
  \item a ledger fiber \(\Lambda(A,B)\) with equivalence relation
  \(\equiv_{\Lambda}^{A,B}\);
  \item a proxy fiber \(\mathcal P(A,B)\) with equivalence relation
  \(\equiv_{\mathcal P}^{A,B}\);
  \item a set \(\Pi^0(A,B)\) of primitive provenance leaves with equivalence
  relation \(\equiv_{\Pi^0}^{A,B}\);
  \item a finite source index set \(\mathcal N_{\mathrm{src}}\), and for every
  \(n\in\mathcal N_{\mathrm{src}}\), an outcome set \(\mathcal O_n\), an
  equivalence relation \(\equiv_n\), and an
  \(\equiv_n\)-saturated pass set \(\mathcal O_n^{\Pass}\subseteq\mathcal O_n\);
  and
  \item the Boolean carrier \(\mathbb B=\{\Pass,\Block\}\), with missing,
  ill-typed, or undefined evaluation interpreted as \(\Block\).
\end{enumerate}
All endpoint indices are part of the types. Thus elements of different fibers
are not comparable unless a displayed map moves them to one common fiber.
\end{definition}

\begin{definition}[Rank-1 total ledger and proxy constructors]
The signature declares the following total source maps:
\[
\begin{aligned}
 e_A&\in\Lambda(A,A),&
 (-)^\dagger_{A,B}&:\Lambda(A,B)\longrightarrow\Lambda(B,A),\\
 \odot_{A,B,C}&:\Lambda(B,C)\times\Lambda(A,B)
                 \longrightarrow\Lambda(A,C),\\[2mm]
 \iota_A&\in\mathcal P(A,A),&
 \jmath_{A,B}&:\mathcal P(A,B)\longrightarrow\mathcal P(B,A),\\
 \mu_{A,B,C}&:\mathcal P(B,C)\times\mathcal P(A,B)
                 \longrightarrow\mathcal P(A,C).
\end{aligned}
\]
Each map respects its displayed component congruence. In particular, if its
inputs are congruent, its outputs are congruent in the codomain fiber. These
declarations replace the undeclared v2 expressions
\(p_{\mathsf{id},A}\), \(p_{\mathsf{inv}}(w)\), and
\(p_{\mathsf{comp}}(v,w)\) by the typed symbols
\(\iota_A\), \(\jmath_{A,B}(p_w)\), and
\(\mu_{A,B,C}(p_v,p_w)\).
\end{definition}

\begin{definition}[Rank-1 provenance algebra and congruence]
For every ordered pair \(A,B\), \(\PiSrc(A,B)\) is the least typed term family
generated by
\[
\begin{aligned}
 \pi\in\Pi^0(A,B)&\Longrightarrow \pi\in\PiSrc(A,B),\\
 \mathsf{id}[A]&\in\PiSrc(A,A),\\
 \pi\in\PiSrc(A,B)&\Longrightarrow
     \mathsf{inv}[\pi]\in\PiSrc(B,A),\\
 \pi\in\PiSrc(A,B),\ \rho\in\PiSrc(B,C)&\Longrightarrow
     \mathsf{comp}[\rho,\pi]\in\PiSrc(A,C).
\end{aligned}
\]
The relation \(\equiv_\Pi^{A,B}\) is the least typed constructor congruence
containing \(\equiv_{\Pi^0}^{A,B}\) and the unit, double-inverse, inverse-unit,
and associativity equations for these constructors. There are no other
provenance leaves, constructors, comparison rules, or opaque escape values.
\end{definition}

\begin{definition}[Rank-2 raw certificate fibers and constructors]
The raw certificate fiber is the displayed product
\[
 \Cert(A,B)=
 \Lambda(A,B)\times\mathcal P(A,B)\times\PiSrc(A,B).
\]
Write \(w=(\lambda_w,p_w,\pi_w):A\to B\). Before acceptance is defined,
declare the total raw-certificate constructors
\[
\begin{aligned}
 \mathbf 1_A
   &=(e_A,\iota_A,\mathsf{id}[A])\in\Cert(A,A),\\
 \mathsf{Inv}_{A,B}(w)
   &=(\lambda_w^\dagger,\jmath_{A,B}(p_w),
      \mathsf{inv}[\pi_w])\in\Cert(B,A),\\
 \mathsf{Comp}_{A,B,C}(v,w)
   &=(\lambda_v\odot\lambda_w,
      \mu_{A,B,C}(p_v,p_w),
      \mathsf{comp}[\pi_v,\pi_w])\in\Cert(A,C).
\end{aligned}
\]
Their domains and codomains are fixed entirely by the endpoint fibers.
\end{definition}

\begin{definition}[Rank-2 certificate congruence]
For \(w,w'\in\Cert(A,B)\), first define
\[
 w\equiv_{\Cert}^{A,B}w'
 \quad\Longleftrightarrow\quad
 \lambda_w\equiv_\Lambda^{A,B}\lambda_{w'}
 \ \wedge\
 p_w\equiv_{\mathcal P}^{A,B}p_{w'}
 \ \wedge\
 \pi_w\equiv_\Pi^{A,B}\pi_{w'}.
\]
This relation is available before any decision map is declared and is an
equivalence because its three component comparisons are equivalences.
\end{definition}

\begin{definition}[Rank-2 source-only decision maps]
For every endpoint pair, declare total maps
\[
\begin{aligned}
 L_{A,B}&:\Lambda(A,B)\longrightarrow\mathbb B,\\
 P_{A,B}&:\mathcal P(A,B)\longrightarrow\mathbb B,\\
 \delta_{n,A,B}&:\Cert(A,B)\longrightarrow\mathcal O_n
   \qquad(n\in\mathcal N_{\mathrm{src}}).
\end{aligned}
\]
Define
\(\mathsf{LedgerOK}_{A,B}(\lambda)\) by \(L_{A,B}(\lambda)=\Pass\),
\(\mathsf{ProxyOK}_{A,B}(p)\) by \(P_{A,B}(p)=\Pass\), and
\[
 \mathsf{NegPass}_{n,A,B}(w)
 \quad\Longleftrightarrow\quad
 \delta_{n,A,B}(w)\in\mathcal O_n^{\Pass}.
\]
The negative-control domain is therefore the already displayed raw certificate
fiber. Each decision map is primitive source data or has a displayed
rank-\(\leq2\) defining expression. None may mention \(\Accept\),
\(\Law\), \(\sim_{\mathrm{cand},v3}\), target data, or process authority.
The maps \(L\), \(P\), and every \(\delta_n\) respect
\(\equiv_\Lambda\), \(\equiv_{\mathcal P}\), and
\(\equiv_{\Cert}\), respectively. Therefore
\[
\ w\equiv_{\Cert}^{A,B}w'
\Longrightarrow
\delta_{n,A,B}(w)\equiv_n\delta_{n,A,B}(w')
\]
without defining certificate congruence from negative outcomes. Each pass set
is saturated, so certificate-congruent inputs have the same pass/block status.
\end{definition}

\section{Acyclic Acceptance and Source-Purity Guard}

\begin{definition}[Complete definition-dependency order]
Every nonlogical symbol in the proposal is assigned exactly one rank:
\begin{longtable}{p{0.10\linewidth}p{0.80\linewidth}}
\toprule
Rank & Objects permitted at that rank \\
\midrule
0 & Source object, carrier, primitive-leaf, outcome, pass-set, and component-equivalence data. \\
1 & Total ledger and proxy constructors, the least provenance algebra, and its constructor congruence. \\
2 & Raw certificate fibers and constructors, certificate congruence, component decision maps, and negative-control predicates. \\
3 & The acceptance predicate, defined only as the displayed conjunction below. \\
4 & The candidate law, accepted-totality assertions, induced relation, coherence obligations, and conditional theorems. \\
\bottomrule
\end{longtable}
A defining edge may point only from a lower rank to a strictly higher rank.
Symbols at one rank may use previously declared symbols at that rank only where
the defining expression is visibly structural and contains no back-edge.
In particular, rank-2 decisions cannot depend on rank-3 acceptance or rank-4
law satisfaction, and no decision or constructor may define
\(\equiv_{\Cert}\) backward. A proposed interpretation with a missing rank, an
undeclared symbol, a same-rank semantic recursion, or a backward edge is not an
interpretation of \(\Law\) and returns \(\Block\).
\end{definition}

\begin{definition}[Complete no-target and no-process guard]
For every nonlogical rank-0 through rank-2 symbol \(s\), its declaration must
include a finite definition tree whose leaves are declared source primitives
and whose internal nodes are lower-rank source constructors. The forbidden
leaf and constructor classes are:
\begin{itemize}[leftmargin=*]
  \item target atlas, topology, differentiability, transition, target metric,
  connection, curvature, cone, geodesic, or proper-time data;
  \item reconstructed geometry, detector outcome, calibration, benchmark
  success, downstream field equation, or exact-GR comparison;
  \item generated derivative, registry metadata, role or agent authority,
  validator result, handoff, checkpoint, approval, adoption, or Gate Chair
  status; and
  \item \(\Accept\), \(\Law\), the induced relation, or the truth of any
  closure, reconstruction, benchmark, or downstream claim.
\end{itemize}
An undeclared, opaque, target-derived, process-derived, or cyclic leaf returns
\(\Block\). Naming a symbol ``source'' is not sufficient. Workflow records may
prove that this proposal was handled consistently; they cannot populate a
carrier, constructor, decision, witness, or scientific premise.

The guard is also semantic: holding all declared source leaves fixed while
varying any target or workflow datum must leave every rank-0 through rank-2
interpretation, every \(\Accept\) value, and candidate-law satisfaction
unchanged. Aliases, lookup tables, hashes, learned parameters, opaque oracles,
proxy labels, provenance leaves, congruence classes, and negative controls are
classified by their transitive provenance rather than their names. This
factorization condition distinguishes source-sorted syntax from source-only
semantics and remains an explicit target for fresh audit.
\end{definition}

\begin{definition}[Rank-3 acceptance]
Define the Boolean decision
\(\Accept_{A,B}:\Cert(A,B)\to\mathbb B\). For a well-formed
\(w=(\lambda_w,p_w,\pi_w)\in\Cert(A,B)\),
\[
\begin{split}
 \Accept_{A,B}(w)=\Pass\quad\Longleftrightarrow\quad&
 L_{A,B}(\lambda_w)=\Pass\\
 &\wedge\ P_{A,B}(p_w)=\Pass\\
 &\wedge\
 \bigwedge_{n\in\mathcal N_{\mathrm{src}}}
 \delta_{n,A,B}(w)\in\mathcal O_n^{\Pass}.
\end{split}
\]
In every other case, \(\Accept_{A,B}(w)=\Block\).
Endpoint and provenance typing are already enforced by membership in
\(\Cert(A,B)\). Missing, ill-typed, undefined, conflicting, extra-grammatical,
or guard-violating data yields \(\Block\).
Here ``acceptance'' means internal certificate admissibility only; it is not an
audit pass, human approval, ontology adoption, or benchmark acceptance.
\end{definition}

\begin{proposition}[Acyclic acceptance is total and non-circular]
For any fixed rank-0 through rank-2 source signature satisfying the dependency
order, \(\Accept_{A,B}:\Cert(A,B)\to\mathbb B\) is a uniquely determined total
function for every endpoint pair.
\end{proposition}

\begin{proof}
Each raw certificate has three typed components by the rank-2 product
definition. The total maps \(L_{A,B}\), \(P_{A,B}\), and
\(\delta_{n,A,B}\) evaluate those components using only rank-0 through rank-2
data. The finite set \(\mathcal N_{\mathrm{src}}\) makes the displayed
conjunction finite. Its Boolean value is therefore unique. No defining path
returns from ranks 0--2 to rank 3 or 4, so neither \(\Accept(w)\leftrightarrow
\Accept(w)\) nor \(\Accept(w)\leftrightarrow\neg\Accept(w)\) can arise as a
component definition. Undefined evaluation is explicitly \(\Block\), proving
totality and fail-closure.
\end{proof}

\begin{proposition}[Declaration closure at statement level]
Every nonlogical term used below is a term of the typed language generated by
the displayed v3 signature.
\end{proposition}

\begin{proof}
The component terms are rank-0 elements. The ledger, proxy, and provenance
parts of identity, inverse, and composition are generated by the rank-1 maps.
Their tuples therefore belong to the rank-2 certificate codomains. All
component comparisons and decision applications use displayed typed maps.
Rank-3 acceptance is an explicit finite conjunction, and rank-4 clauses apply
only previously declared terms and predicates. Structural induction on term
rank covers every occurrence. This is a statement-level closure proof; a
fresh audit must still test whether the intended interpretations respect the
source-purity guard.
\end{proof}

\section{Candidate Accepted-Totality and Coherence Law}

\begin{definition}[Proposal-only candidate closure law v3]
A declared source signature satisfies \(\Law\) only if the guard and dependency
order hold and all following rank-4 clauses are true.
\begin{enumerate}[leftmargin=*]
  \item \textbf{Accepted identity totality.}
  For every \(A\in\mathcal F_{\mathrm{src}}\),
  \(\Accept_{A,A}(\mathbf 1_A)=\Pass\).

  \item \textbf{Accepted inverse totality.}
  For every \(w\in\Cert(A,B)\), if \(\Accept_{A,B}(w)=\Pass\), then
  \[
    \Accept_{B,A}(\mathsf{Inv}_{A,B}(w))=\Pass.
  \]

  \item \textbf{Accepted composition totality.}
  For all \(w\in\Cert(A,B)\) and \(v\in\Cert(B,C)\), if both are accepted,
  then
  \[
    \Accept_{A,C}(\mathsf{Comp}_{A,B,C}(v,w))=\Pass.
  \]

  \item \textbf{Congruence and acceptance invariance.}
  Each \(\equiv_{\Cert}^{A,B}\) is an equivalence, acceptance is constant on
  its classes, and \(\mathsf{Inv}\) and \(\mathsf{Comp}\) preserve it.

  \item \textbf{Unit compatibility.}
  For every accepted \(w:A\to B\),
  \[
   \mathsf{Comp}(w,\mathbf 1_A)\equiv_{\Cert}w,\qquad
   \mathsf{Comp}(\mathbf 1_B,w)\equiv_{\Cert}w.
  \]

  \item \textbf{Inverse compatibility.}
  For every accepted \(w:A\to B\),
  \[
  \mathsf{Inv}(\mathsf{Inv}(w))\equiv_{\Cert}w,\quad
  \mathsf{Comp}(\mathsf{Inv}(w),w)\equiv_{\Cert}\mathbf 1_A,\quad
  \mathsf{Comp}(w,\mathsf{Inv}(w))\equiv_{\Cert}\mathbf 1_B.
  \]

  \item \textbf{Associativity compatibility.}
  For every composable accepted triple \(w,v,u\),
  \[
  \mathsf{Comp}(u,\mathsf{Comp}(v,w))
  \equiv_{\Cert}
  \mathsf{Comp}(\mathsf{Comp}(u,v),w).
  \]

  \item \textbf{Component and negative-outcome coherence.}
  The ledger, proxy, provenance, and negative-outcome components of the
  preceding equations agree under
  \(\equiv_\Lambda,\equiv_{\mathcal P},\equiv_\Pi,\equiv_n\), respectively.
  This is a substantive candidate-law obligation, not a consequence of naming
  the constructors.

  \item \textbf{Fail-closed model condition.}
  If any carrier, constructor, congruence, decision map, definition tree,
  accepted witness, typing judgment, compatibility equation, or guard
  obligation is absent or fails, the structure does not satisfy \(\Law\).
  No target or process datum supplies a default witness.
\end{enumerate}
\end{definition}

\section{Conditional Equivalence Theorem}

Define
\[
 A\sim_{\mathrm{cand},v3}B
 \quad\Longleftrightarrow\quad
 \exists w\in\Cert(A,B)\ \Accept_{A,B}(w)=\Pass.
\]

\begin{proposition}[Equivalence under the repaired candidate law]
If a declared source family satisfies every clause of \(\Law\), then
\(\sim_{\mathrm{cand},v3}\) is an equivalence relation on
\(\mathcal F_{\mathrm{src}}\).
\end{proposition}

\begin{proof}
Accepted identity totality supplies \(\mathbf 1_A:A\to A\) for every \(A\),
so the relation is reflexive. If an accepted \(w:A\to B\) witnesses
\(A\sim_{\mathrm{cand},v3}B\), accepted inverse totality supplies the accepted
\(\mathsf{Inv}(w):B\to A\), proving symmetry. If accepted
\(w:A\to B\) and \(v:B\to C\) witness the two premises of transitivity,
accepted composition totality supplies the accepted
\(\mathsf{Comp}(v,w):A\to C\). All witnesses are rank-2 terms and acceptance
is the rank-3 source-only conjunction, so the proof contains no hidden witness
choice or back-reference to the relation being proved.
\end{proof}

\begin{remark}
The theorem is conditional on a structure satisfying the proposal. The
coherence clauses make its certificate algebra groupoid-like modulo declared
congruence; the coarse equivalence proof itself uses only the three explicit
accepted-totality clauses. Neither fact shows that current ontology supplies
or selects such a structure.
\end{remark}

\section{Obstruction Repair and Fail-Closed Branches}

\paragraph{Exact statement-level repair.}
Every v2 escape named by \texttt{OBST-EQSRC-CLOSURE-SIGNATURE-001} now has a
displayed declaration: proxy identity, inverse, and composition are typed
rank-1 data; ledger, proxy, provenance, outcome, and certificate congruences
have explicit fibers; negative-control maps have the raw-certificate domain;
and the finite rank order excludes acceptance recursion. Thus the exact
signature and non-circularity repair is supplied at proposal statement level,
pending fresh independent audit.

\paragraph{Missing or ill-typed declaration branch.}
If a symbol lacks a declared domain or codomain, or a constructor returns an
element in the wrong endpoint fiber, the structure returns \(\Block\) and is
not a model of \(\Law\).

\paragraph{Cyclic decision branch.}
If \(L\), \(P\), or any \(\delta_n\) depends on \(\Accept\), \(\Law\), the
induced relation, or its own later consequence, the rank condition fails.
There is no least- or greatest-fixed-point default.

\paragraph{Target or process leaf branch.}
If any carrier, constructor, congruence, decision, or witness is populated by
a forbidden target or process class, the complete source-purity guard fails.
Artifact registration, validation, or checkpointing cannot cure the failure.

\paragraph{Coherence branch.}
If accepted constructors fail unit, inverse, associativity, component
congruence, or negative-outcome compatibility, the structure fails \(\Law\)
even if its coarse accepted-certificate relation happens independently to be
an equivalence.
