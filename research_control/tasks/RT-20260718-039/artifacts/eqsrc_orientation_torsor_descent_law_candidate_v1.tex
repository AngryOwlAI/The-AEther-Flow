\section{Orientation Torsor, Fibre Data, and Dependency Order}

\begin{definition}[Orientation torsor]
Set
\[
 O_U=\{(P,Q),(Q,P)\}.
\]
The group \(C_2=\{1,\sigma\}\) acts freely and transitively by
\(\sigma(P,Q)=(Q,P)\). No element of \(O_U\) is distinguished.
\end{definition}

Put \(p=\mathbf1_P\), \(q=\mathbf1_Q\), and
\[
 A_U=\operatorname{span}_{\Ftwo}\{p,q\}.
\]
For \(o=(P_o,Q_o)\in O_U\), define
\[
 B_{U,o}=\Ftwo\mathbf1_{Q_o},\qquad
 a\,R_{U,o}\,a'\Longleftrightarrow a-a'\in B_{U,o}.
\]
Both fibres are exact candidate-v1 boundary relations because both displayed
first blocks are odd. Define the order-independent complement involution
\[
 c_U(\alpha p+\beta q)=\beta p+\alpha q.
\]
Then \(c_U^2=1\) and
\[
 c_U(B_{U,o})=B_{U,\sigma o}.
\]

The dependency order is
\[
(X,\Pi,\Xi_{\mathrm{src}})
\longrightarrow(O_U,A_U,c_U,\text{pre-relation controls})

\section{Associated State and Relation Objects}

\begin{definition}[Equivariant bundles]
Set
\[
\mathcal B_U=\{(o,b):o\in O_U,\ b\in B_{U,o}\},
\]
\[
\mathcal R_U=\{(o,a,a'):o\in O_U,\ a-a'\in B_{U,o}\}.
\]
The diagonal \(C_2\)-actions are
\[
\sigma(o,a)=(\sigma o,c_Ua),\qquad
\sigma(o,a,a')=(\sigma o,c_Ua,c_Ua').
\]
\end{definition}

\begin{definition}[Associated objects]
Define
\[
\bar A_U=(O_U\times A_U)/C_2,\qquad
\bar B_U=\mathcal B_U/C_2\subset\bar A_U.
\]
The associated vector operations are computed after taking the unique
representatives of all operands over one common \(o\); equivariance makes the
result independent of that temporary representative. Define
\[
[o,a]\,\bar R_U\,[o,a']
\Longleftrightarrow a-a'\in B_{U,o}.
\]
Equivalently, \(\bar R_U\) is the image of
\(\mathcal R_U/C_2\) in \(\bar A_U\times\bar A_U\).
\end{definition}

\begin{theorem}[Well-definedness and ordering independence]
\(\bar A_U\) is a two-dimensional associated \(\Ftwo\)-vector space,
\(\bar B_U\) is a one-dimensional subspace, and \(\bar R_U\) is its
kernel-pair congruence. None depends on initially displaying \(\Pi\) as
\((P,Q)\) rather than \((Q,P)\).
\end{theorem}

\begin{proof}
The indicators \(p,q\) are independent because their nonempty supports are
disjoint. Exchanging their displayed names applies \(c_U\). Since
\(c_U^2=1\), diagonal representatives related by the exchange give identical
associated classes and vector operations. The identity
\(c_U(B_{U,o})=B_{U,\sigma o}\) makes the line and relation quotients
well defined.
\end{proof}

\section{Inverse, Cocycle, Equivariance, and Functoriality}

For \(o,o'\in O_U\), let \(t_{o,o'}=1\) when \(o=o'\) and
\(t_{o,o'}=c_U\) otherwise. Orientation-change parity adds modulo two, so
\[
t_{o',o''}t_{o,o'}=t_{o,o''},\qquad
t_{o',o}=t_{o,o'}^{-1}.
\]
Moreover,
\[
a\,R_{U,o}\,a'
\Longleftrightarrow
c_Ua\,R_{U,\sigma o}\,c_Ua'.
\]

\begin{theorem}[Admitted-source functoriality]
For an admitted morphism \(f:U\to U'\), define
\[
O_f(P,Q)=(fP,fQ),\qquad
(F_fa)_{f(x)}=a_x.
\]
Then
\[
O_f\sigma=\sigma O_f,\quad
F_fc_U=c_{U'}F_f,\quad
F_f(B_{U,o})=B_{U',O_f(o)}.
\]
Consequently
\[
\bar F_f[o,a]=[O_f(o),F_fa]
\]
is well defined, preserves and reflects \(\bar R\), and respects identities,
composition, and inverses.
\end{theorem}

\begin{proof}
The bijection carries the two block indicators to the corresponding block
indicators and therefore intertwines their exchange. It transports the
opposite-block line in each orientation to the opposite-block line in the
image orientation. The associated formulas then commute, and functoriality
is inherited from pushforward of functions.
\end{proof}

This is exact functoriality inside the admitted proposal category. It is not
physical covariance. In particular, when \(|P|\neq|Q|\), \(c_U\) need not be
induced by any carrier bijection.

\section{Fibre Pullback and Strict Scalar Non-Descent}

\begin{theorem}[Exact fibre pullback]
For every \(o\in O_U\), the map
\[
\tau_o:A_U\longrightarrow\bar A_U,\qquad \tau_o(a)=[o,a]
\]
is a linear bijection satisfying
\[
\tau_{\sigma o}=\tau_oc_U,\qquad
\tau_o^{-1}(\bar B_U)=B_{U,o},\qquad
\tau_o^*(\bar R_U)=R_{U,o}.
\]
\end{theorem}

\begin{theorem}[Strict scalar non-descent]
There is no relation \(S_U\) on the original common \(A_U\) whose identity
pullback equals \(R_{U,o}\) for both orientations.
\end{theorem}

\begin{proof}
For \(o=(P,Q)\) and \(\sigma o=(Q,P)\),
\[
B_{U,o}=\langle q\rangle\neq\langle p\rangle=B_{U,\sigma o}.
\]
The kernel-pair relations of distinct subspaces are distinct. A single
identity-compared relation equal to both would force their equality. The
complement map transports one relation to the other, which proves twisted
bundle descent rather than scalar descent on the original \(A_U\).
\end{proof}

The associated object therefore changes the codomain and state
identification. Abstract isomorphism after choosing \(\tau_o\) does not turn
that choice into a source-derived orientation.

\section{Natural-Section Obstruction and Pointed-Torsor Repair}

\begin{theorem}[Scoped no-natural-section result]
On any admitted source subcategory containing an object \(U_*\) with an
orientation-exchange automorphism \(s\), there is no natural section choosing
one element of every \(O_U\).
\end{theorem}

\begin{proof}
Naturality at \(U_*\) would require the chosen orientation to be fixed by
\(O_s=\sigma\). The free two-element torsor has no fixed point.
\end{proof}

The qualifier is essential. A set-theoretic section always exists, and a
restricted category with unequal block sizes, parity asymmetry, or additional
source decoration may admit a natural pointing. Such data change the route.

\begin{proposition}[Pointed-torsor equivalence]
A choice \(s(U)\in O_U\) identifies
\((\bar A_U,\bar R_U)\) with
\((A_U,R_{U,s(U)})\). Conversely, because the two fibre relations are
distinct, selecting one fibre relation selects its orientation. Thus a
preferred section is exactly additional oriented-mark data recovering one
frozen candidate-v1 fibre.
\end{proposition}
