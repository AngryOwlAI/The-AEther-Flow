\section{Candidate Relation}

\(\MCBridgeCand(E)\) is the source-local partial relation

\begin{condition}[MCB-CAND-001 source-domain discipline]
Every input field must be a declared source-side field in
\(\mathcal{D}^{\mathrm{MCB}}_E\). Accepted evidence is used only as
precondition data and may not be converted into adoption.
\end{condition}

\begin{condition}[MCB-CAND-002 pre-tensorial codomain]
The output lands in \(\SMI(E)\subseteq\PTMI_E\). If tensorial semantics are
needed, the candidate fails closed with
\texttt{OB-MC-BRIDGE-TENSORIAL-OVERREAD}; it does not import a tensor.
\end{condition}

\begin{condition}[MCB-CAND-003 source-locality]
For each output component of \(X_E\), \(\Lambda_E\) records a finite support
inside \(F_E^\sharp\). No component may depend on target topology, target
atlas, target metric, empirical detector semantics, matter fields, benchmark
fit, or Einstein equations.
\end{condition}

\begin{condition}[MCB-CAND-004 morphism and relabeling behavior]
For each declared source-family morphism or relabeling
\[
  \rho:\mathcal{D}^{\mathrm{MCB}}_E \to
  \mathcal{D}^{\mathrm{MCB}}_{E'},
\]
the candidate supplies a source-side transport
\[
  \rho_*:\SMI(E)\to\SMI(E')
\]
by relabeling \(I_E,Q_E,\Lambda_E,N_E,U_E,C_E,\NoTargetImport_E\), and
\(\mathcal{O}_E\) componentwise. Successful outputs satisfy
\[
  \rho_*\bigl(\mathcal{B}^{\mathrm{MCB\text{-}cand}}_E(D_E)\bigr)
  =
  \mathcal{B}^{\mathrm{MCB\text{-}cand}}_{E'}(\rho D_E)
\]
whenever both sides are defined. If this compatibility cannot be supplied,
the candidate fails closed with \texttt{OB-MC-BRIDGE-COVARIANCE-FAIL}.
\end{condition}

\begin{condition}[MCB-CAND-005 graph-indexed sector-uniformity]
Let \(\Sigma_E\) be represented by a finite graph-indexed source-sector
family. The construction rule for \(I_E,Q_E,\Lambda_E,N_E,U_E,C_E\) is the
same rule at every sector vertex and along every declared sector edge, with
only declared source parameters varying. A two-sector shared-form shortcut is
not enough; hand-tailored sector cases fail with
\texttt{OB-MC-BRIDGE-SECTOR-UNIFORMITY-FAIL}.
\end{condition}

\begin{condition}[MCB-CAND-006 source variation behavior]
For each allowed source variation \(v\in\mathcal{V}_E\), the candidate

\texttt{OB-MC-BRIDGE-TARGET-IMPORT}.
\end{condition}

\begin{condition}[MCB-CAND-007 balance/flux compatibility]

\begin{condition}[MCB-CAND-008 scoped-\(\geff\) and interface evidence limits]
