  \item \(\chi_i^{\mathrm{src}}\) are chart-candidate names with source

  \item \(\rho\) anchors the finite replay data inside the extension record;
  \item \(\Sigma\) records source separation, support, inverse, cocycle,
        and soldering certificates;
  \item \(\mathcal{V}\) records admissible source-cover variations; and
  \item \(\mathcal{B}\) records bottom branches.
\end{itemize}
\end{definition}

\begin{definition}[Admissible extension predicate]
For \(E\) and \(R_{\mathrm{replay}}\), define
\[
  \AdmExt(E,R_{\mathrm{replay}})\in\mathbb{B}_{\Bot}.
\]
The value is \(\mathsf{true}\) only when all of the following source-side
conditions are supplied:
\begin{enumerate}
  \item \(\rho\) embeds the finite replay anchor into \(E\) by source-token
        identities or explicit source equivalence;
  \item source-local separation certificates extend the finite replay
        separation data to every declared source-overlap class;
  \item source chart-support records are supplied without target open sets or
        target coordinate neighborhoods;
  \item selected transition tokens have source inverse and cocycle
        certificates wherever overlap composition is declared;
  \item \(\RespLC\) readout-token attachment is mediated by
        \(\SolderSrc\) certificates or explicit source equivalence classes;
  \item \(\mathcal{V}\) states admissible source variations and what they
        preserve, transform equivalently, fail, or bottom; and
  \item no forbidden premise is used.
\end{enumerate}
The value is \(\Bot\) if a branch requires target geometry or process
authority. It is \(\mathsf{false}\) if source data are present but explicitly
fail a stated source condition.
\end{definition}

\section{Theorem With Hypotheses}

\begin{lemma}[Source-only non-bottom dependency]
Assume \(\AdmExt(E,R_{\mathrm{replay}})=\mathsf{true}\) and
\(\NonBottom(E)\neq\Bot\). Then every component of
\(\CoverExt(R_{\mathrm{replay}},E)\) depends only on
\(\mathcal{S}_{\mathrm{src}}\).
\end{lemma}

\begin{proof}
The admissible-extension predicate requires source anchoring, source-local
separation, source chart support, source transition tokens, source inverse
and cocycle certificates, source soldering certificates, admissible
source-variation records, and explicit bottom branches. The non-bottom
certificate repeats those components as source-only certificate families.
Every forbidden premise is excluded from \(\mathcal{S}_{\mathrm{src}}\), and
any branch requiring target topology, target atlas, target coordinate
transition, target metric, proper time, empirical detector semantics,
benchmark success, validation status, registry metadata, role identity,
handoff status, generated derivatives, or file order returns \(\Bot\).
Therefore a non-bottom output cannot depend on forbidden target or process
authority.
\end{proof}

\section{Failure Branches}

\begin{branch}[Unsupported extension]
If \(E\) lacks source-local separation, chart support, transition, inverse,
cocycle, soldering, or variation records required by \(\AdmExt\), then
\(\CoverExt(R_{\mathrm{replay}},E)=\Bot\).
\end{branch}

\begin{branch}[Selector nonuniqueness]
If transition candidates are tied under source data and no explicit source
equivalence resolves the tie, file order, registry order, validation status,
role identity, or target resemblance may not break the tie. The output is
\(\Bot\).
\end{branch}

\begin{branch}[Soldering nonuniqueness]
If \(\RespLC\) readout attachment is nonunique without an explicit source
equivalence relation, \(\SolderSrc\) bottoms the branch.
\end{branch}

\begin{branch}[Target or process import]
If a branch uses target topology, target atlas, target metric, proper time,
empirical detector semantics, benchmark success, validation status, registry
metadata, role identity, handoff status, generated derivatives, or file
order as a source premise, \(\BottomSrc\) returns \(\Bot\).
\end{branch}

\begin{branch}[Finite replay overread]
If the finite checker replay is treated as proof of arbitrary source-cover
existence, arbitrary finite-variation robustness, source-law adoption, or
\(\Msrc\) adoption, the branch violates this packet boundary and bottoms.
\end{branch}
