\begin{abstract}
This note constructs one bounded, target-free, proposal-only source-extension
candidate.  A codimension-one source foliation atlas consists of local
submersions $q_i$ related on overlaps by one-dimensional local
diffeomorphisms.  Their differential lines descend to an unoriented
projective conormal line $L_q\subset T^*M$.  The line defines a proper
reduction of the frame bundle to the stabilizer of a covector line.  The
resulting bridge is restriction-compatible, glues across overlaps, and is
natural in the declared proposal category.  Two exact structures on the same
underlying $\mathbb R^4$ have distinct bridge images but identical unchanged
$\Kstar$, so this bridge cannot factor through the frozen standalone
$\Kstar$ classifier.  The construction is a new proposal-only source
primitive, not a derivation from the current ontology, a physical time
orientation, a causal cone, a conformal geometry, or an effective metric.
\end{abstract}

\section{The proposal category}

\begin{definition}[Source foliation presentation]
An object of $\mathsf{SrcFol}_1$ is a tuple
\[
 \mathcal F=(M,\{U_i\},\{q_i\},\{\gamma_{ij}\}),
\]
where $M$ is a nonempty smooth four-manifold, $\{U_i\}$ is a locally finite
open cover, each $q_i:U_i\to\mathbb R$ is a smooth submersion, and on every
nonempty overlap
\[
 q_i=\gamma_{ij}\circ q_j
\]
for a one-dimensional local diffeomorphism $\gamma_{ij}$.  The transition
maps obey identity, inverse, and triple-overlap cocycle laws wherever their
compositions are defined.  The real line is an abstract local source-label
chart; it has no physical-time semantics.
\end{definition}

An arrow $f:\mathcal F\to\mathcal F'$ is a source diffeomorphism together
with a declared cover refinement that transports the descended line of
$\mathcal F'$ to that of $\mathcal F$.  This is a task-local proposal
category.  It is not adopted $\mathrm{EqSrc}$ and it is not a physical gauge
group.

\begin{definition}[Projective conormal line]
On $U_i$, define
\[
 L_i=\Span(dq_i)\subset T^*U_i.
\]
The intended global object is the rank-one line subbundle $L_q\subset T^*M$
obtained by descent.  The sign and normalization of $dq_i$ are quotiented, so
$L_q$ is unoriented.
\end{definition}

\begin{theorem}[Exact overlap descent]
The local lines $L_i$ agree on overlaps and glue uniquely to a smooth line
subbundle $L_q\subset T^*M$.  The result commutes with open restriction and
cover refinement.
\end{theorem}

\begin{proof}
The chain rule gives
\[
 dq_i=\gamma'_{ij}(q_j)\,dq_j.
\]
Because $\gamma_{ij}$ is a local diffeomorphism, its derivative is nowhere
zero.  Hence $\Span(dq_i)=\Span(dq_j)$ on every overlap.  The transition
cocycle makes the equality coherent on triple intersections, and ordinary
line-bundle descent gives a unique $L_q$.  Restriction preserves the same
equalities.  A refinement merely repeats them on smaller open sets, so the
descended line is unchanged.
\end{proof}

\section{Proper structure-group reduction}

Let $V=\mathbb R^4$ and $\ell_0=\Span(e^0)\subset V^*$.  Define
\[
 H=\{A\in GL(V):A^*\ell_0=\ell_0\}.
\]
For a frame $u:V\to T_xM$, let $u^*L_{q,x}$ denote the pulled-back covector
line.  Set
\[
 P_H(q)=\{u\in F(M):u^*L_{q,x}=\ell_0\}.
\]

\begin{proposition}[Proper reduction]
$H$ is a closed proper subgroup of $GL(4,\mathbb R)$ of dimension $13$, and
$P_H(q)$ is a principal $H$-subbundle of the frame bundle.
\end{proposition}

\begin{proof}
In a basis adapted to $e^0$, line preservation requires the first row to be
$(a,0,0,0)$ with $a\ne0$.  The lower-left column has three free entries and
the lower-right $3\times3$ block is arbitrary invertible.  Thus
$\dim H=1+3+9=13<16=\dim GL(4)$.  The matrix interchanging the first two basis
vectors lies outside $H$, so $H$ is proper.  The set of frames adapted to a
smooth line subbundle is locally nonempty and is acted on freely and
transitively on the right by $H$, which is the standard reduction
construction.
\end{proof}

\section{The bridge and its exact force}

\begin{definition}[Projective-conormal reduction bridge]
Define
\[
 \Bridge(\mathcal F)=(L_q,P_H(q)).
\]
For an admitted arrow $f:\mathcal F\to\mathcal F'$, define bridge transport by
the declared identity $f^*L_{q'}=L_q$ and the induced frame-bundle map.
\end{definition}

\begin{theorem}[Naturality and gluing]
$\Bridge$ is a natural, restriction-compatible construction on
$\mathsf{SrcFol}_1$.  It depends only on the descended line and not on the
chosen label normalization, sign, or cover presentation.
\end{theorem}

\begin{proof}
Overlap descent proves object formation.  Pullback commutes with exterior
differentiation and with line span.  Therefore a declared proposal arrow
transports $L_{q'}$ to $L_q$, and the adapted-frame construction transports
$P_H(q')$ to $P_H(q)$.  Identity and composition follow from pullback
functoriality.  Nonzero reparametrization derivatives change only the local
representative, never its line.  The restriction and refinement statements
were proved in the descent theorem.
\end{proof}

\subsection{Exact nonfactorization fixture}

Take $M=\mathbb R^4$ with a three-open-set cover.  Candidate $A$ uses local
covectors
\[
 (1,0,0,0),\quad(-2,0,0,0),\quad(6,0,0,0),
\]
with overlap multipliers $-2,-3,6$.  Candidate $B$ uses
\[
 (1,1,0,0),\quad(3,3,0,0),\quad(-6,-6,0,0),
\]
with multipliers $3,-2,-6$.  Both multiplier families satisfy their cocycle,
and each family descends.  Their bridge lines are respectively
\[
 \Span(dx^0)\quad\hbox{and}\quad\Span(dx^0+dx^1),
\]
which are distinct.  The linear source diffeomorphism
\[
 F(x^0,x^1,x^2,x^3)=(x^0+x^1,x^1,x^2,x^3)
\]
pulls the first line back to the second, giving an exact naturality control.

\begin{theorem}[Nonconstancy and frozen-$\Kstar$ nonfactorization]
The bridge is nonconstant.  It cannot factor through the unchanged
$\Kstar$ assignment on the underlying smooth source arena.
\end{theorem}

\begin{proof}
Candidates $A$ and $B$ have the same underlying smooth manifold, so the
unchanged objectwise classifier assigns the same $\Kstar(M)$ to both.  Their
projective conormal lines are distinct.  If
$\Bridge=G\circ\Kstar$ for any map $G$, equal $\Kstar$ inputs would have equal
bridge outputs, a contradiction.
\end{proof}

This theorem is the exact bridge force obtained here.  It does not say the
line is a response covector, causal normal, time orientation, conformal cone,
or metric datum.
