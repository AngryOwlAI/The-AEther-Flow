\section{Target Definitions}

\begin{definition}[\(\MatterSector(E;\Bcur)\)]
\(\MatterSector(E;\Bcur)\) is a candidate source-side record consisting of:
\begin{enumerate}[leftmargin=*]
\item a declared source-token domain \(T_E \subseteq \Tokens(E)\);
\item a finite or locally indexed set of source-sector labels
  \(L_E = \{\mathtt{matter\_sector}, \mathtt{nonmatter\_sector},
  \mathtt{guarded\_undetermined}\}\), or a documented source-equivalent
  refinement;
\item a candidate discriminator target \(D_E:T_E\to L_E\);
\item source-only evidence obligations for why each token is in the assigned
  sector;
\item fail-closed behavior for tokens that cannot be separated by source-only
  data.
\end{enumerate}

This definition is target grammar only. It does not assert that such a sector
record exists for any \(E\).
\end{definition}

\begin{definition}[\(\Discriminator(E;\Bcur,D_E)\)]
\(\Discriminator(E;\Bcur,D_E)\) holds only if a later candidate supplies a
source-only map \(D_E:T_E\to L_E\) satisfying all of the following:
\begin{enumerate}[leftmargin=*]
\item \textbf{totality on declared source tokens}: every token in \(T_E\) is
  assigned exactly one source-sector label;
\item \textbf{source-purity}: assignments use only source-side data in
  \(\Bcur\) and declared draft/control source-side assumptions;
\item \textbf{non-import}: assignments satisfy \(\NoSE\), \(\NoAction\), and
  \(\NoDetector\);
\item \textbf{relabeling stability}: if a source relabeling preserves the
  source structure used by the candidate, it preserves the induced sector
  classification up to label-preserving isomorphism;
\item \textbf{finite-variation guard}: if a finite variation deletes or
  perturbs irrelevant source labels, then either the sector classification
  remains invariant or the candidate records the affected token as
  \(\mathtt{guarded\_undetermined}\);
\item \textbf{no notation collapse}: renaming a token, deleting a label, or
  using a table row as a name is not a source-side discriminator unless the
  required source behavior is shown;
\item \textbf{fail-closed obstruction}: if two tokens cannot be distinguished
  by source-side data, then a later packet must record \(\FailureObs\) rather
  than import a target matter property.
\end{enumerate}
\end{definition}

\begin{definition}[\(\NoSE\)]
\(\NoSE\) holds when the candidate does not use stress-energy semantics,
stress-energy tensor components, energy density, pressure, flux interpreted
as physical stress energy, variational stress tensor, Bianchi identity,
Einstein tensor matching, or target field-equation behavior as premises or
conclusions.

Source-side bookkeeping labels are allowed only as uninterpreted source
syntax, and only after a later packet proves their source-side role. They are
not stress-energy semantics.
\end{definition}

\begin{definition}[\(\NoAction\)]
\(\NoAction\) holds when the candidate does not import a matter action,
Lagrangian density, action variation, Euler-Lagrange equation, path integral,
least-action principle, or target dynamics as a premise or conclusion.

A future source action analogue would require a separate P5 packet and remains
blocked here.
\end{definition}

\begin{definition}[\(\NoDetector\)]
\(\NoDetector\) holds when the candidate does not use empirical detector
calibration, apparatus behavior, proper-time readout, laboratory protocol,
observer instrumentation, or physical measurement semantics as a premise or
conclusion.

Source probe tokens may be introduced only in a later packet after the
matter-sector discriminator target is audited or constructed. This P5-T02
artifact does not define detector semantics.
\end{definition}

\begin{definition}[\(\FailureObs(E;\Bcur,O)\)]
\(\FailureObs(E;\Bcur,O)\) is the scoped failure record available to P5-T03
if the current source basis cannot supply a \(\Discriminator\) satisfying the
obligations above. The record must include:
\begin{enumerate}[leftmargin=*]
\item the declared basis \(\Bcur\);
\item the attempted token domain \(T_E\);
\item the exact failed obligation;
\item the minimal indistinguishability or missing-law reason, if available;
\item evidence path;
\item exact conclusion blocked;
\item same-milestone continuation status:
  \(\mathtt{blocked\_adoption\_open\_continuation}\).
\end{enumerate}

Allowed wording: ``current ontology does not derive a source-side
matter-sector discriminator under the declared basis and obligations.''

Forbidden wording: ``therefore matter semantics are impossible,'' ``therefore
matter coupling is impossible,'' or ``therefore the global theory is
rejected.''
\end{definition}
