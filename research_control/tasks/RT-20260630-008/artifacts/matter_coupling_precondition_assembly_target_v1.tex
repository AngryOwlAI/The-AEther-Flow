\section{Source Domain}

\begin{definition}[Source-domain package for precondition assembly]
For a source context \(E\), the domain of \(\MCPA(E)\) is a declared
finite/local source package
\[
  \mathcal{D}^{\mathrm{MCPA}}_E =
  (E,F_E^\sharp,\mathcal{W}_E,\mathcal{S}_E,\mathcal{M}_E,
  \mathcal{R}_E,\mathcal{V}_E,\mathcal{P}_E,\mathcal{N}_E).
\]
Here \(F_E^\sharp\) is the finite/local source family,
\(\mathcal{W}_E\) is the scoped parameterized-witness ledger containing
entries of the form \((\ParamWitness(E),\BridgeSlot_n(E),
\NoTargetImport_n)\), \(\mathcal{S}_E\) is the scoped
matter-signature ledger containing \(\MatSigDisc(E)\) and
\(\SemMatterSig(E)\) only as evidence/precondition data,
\(\mathcal{M}_E\) is the source-side matching ledger between witness slots
and signature records, \(\mathcal{R}_E\) is the source relabeling family,
\(\mathcal{V}_E\) is the finite variation family, \(\mathcal{P}_E\) is the
proof-obligation ledger, and \(\mathcal{N}_E\) is the no-target-import
certificate family.

The ledgers \(\mathcal{W}_E\) and \(\mathcal{S}_E\) may be cited only as
scoped source-extension evidence/precondition data. They are not source-law
adoption, not \(\MetricData(E)\), not \(\geff\), not a coupling law, not
matter coupling, not stress-energy semantics, not detector semantics, not
matter action, and not Einstein equations.
\end{definition}

\section{Target Codomain}

\begin{definition}[Pre-coupling assembly record]
The codomain of \(\MCPA(E)\) is not a physical matter sector, not an adopted
coupling law, not a stress-energy tensor, not a detector model, and not a
matter action. It is a source-side pre-coupling assembly record family
\[
  \PreCoupAsm_E =
  (A_E,L_E,Q_E,\mathsf{MatchOK}_E,\mathsf{LocalOK}_E,
  \mathsf{NatOK}_E,\mathsf{NoImportOK}_E,\mathcal{O}_E).
\]
The set \(A_E\) records assembled source-side precondition records, \(L_E\)
records their finite/local dependence data, \(Q_E\) is a quality ledger,
\(\mathsf{MatchOK}_E\) records compatibility between witness slots and
matter-signature records, \(\mathsf{LocalOK}_E\) records source-locality,
\(\mathsf{NatOK}_E\) records relabeling naturality,
\(\mathsf{NoImportOK}_E\) records no-target-import checks, and
\(\mathcal{O}_E\) maps failures to fail-closed labels.
\end{definition}

\begin{definition}[Matter-Coupling Precondition Assembly Target v1]
\(\MCPA(E)\) is the obligation that a future candidate provide either a
source-local partial map or source-local relation
\[
  \mathcal{A}^{\mathrm{MCPA}}_E:
  \mathcal{D}^{\mathrm{MCPA}}_E \dashrightarrow \PreCoupAsm_E
\]
or a precise fail-closed obstruction label from \(\mathcal{O}_E\), while
satisfying Conditions MCPA-TGT-001 through MCPA-TGT-008 below.
\end{definition}

\section{Target Conditions}

\begin{condition}[MCPA-TGT-001 source-evidence discipline]
All successful inputs must be source-side records in
\(\mathcal{D}^{\mathrm{MCPA}}_E\). Accepted scoped evidence may appear only as
precondition data. It may not be converted into source-law adoption,
\(\MetricData(E)\), \(\geff\), coupling-law adoption, matter coupling,
stress-energy semantics, detector semantics, matter action, Einstein
equations, or benchmark behavior.
\end{condition}

\begin{condition}[MCPA-TGT-002 scoped evidence is not adoption]
The candidate must keep separate ledgers for \(\mathcal{W}_E\),
\(\mathcal{S}_E\), and \(\mathcal{M}_E\). A match between witness slots and
matter-signature records is an eligibility precondition only. It is not
\(\MatSrcDiscLaw\), not \(\SemMatterSrc\), not \(\MetricData(E)\), not
\(\geff\), not a coupling law, and not matter coupling.
\end{condition}

\begin{condition}[MCPA-TGT-003 witness-signature compatibility]
For each assembled record in \(A_E\), the candidate must name the
source-local witness slot, the source-local matter-signature record, and the
declared compatibility relation in \(\mathcal{M}_E\). If no compatibility
relation is supplied, the output must fail closed rather than infer physical
matter content.
\end{condition}

\begin{condition}[MCPA-TGT-004 locality]
\(\mathcal{A}^{\mathrm{MCPA}}_E\) must depend only on declared finite/local
source records in \(F_E^\sharp\), \(\mathcal{W}_E\), \(\mathcal{S}_E\), and
\(\mathcal{M}_E\). It may not depend on target topology, target atlas, target
metric, Lorentzian signature, proper time, stress-energy tensor, matter
action, detector protocol, empirical calibration, benchmark fit, or Einstein
equations.
\end{condition}

\begin{condition}[MCPA-TGT-005 covariance and naturality]
For each source relabeling or source-family morphism
\[
  \rho: \mathcal{D}^{\mathrm{MCPA}}_E \to
  \mathcal{D}^{\mathrm{MCPA}}_{E'},
\]
the candidate must supply a source-side transport
\(\rho_*:\PreCoupAsm_E\to \PreCoupAsm_{E'}\) such that successful outputs
commute with \(\rho\). Failure to define this compatibility is a source-side
noncovariance obstruction, not a license to import target covariance.
\end{condition}

\begin{condition}[MCPA-TGT-006 finite-variation stability]
For every declared finite variation \(v\in\mathcal{V}_E\), the candidate must
state whether the assembled precondition record is stable, repaired by
declared source-local data, or fail-closed. Stability may not be repaired by
target-metric normalization, empirical calibration, stress-energy
conservation, detector semantics, matter action, or benchmark behavior.
\end{condition}

\begin{condition}[MCPA-TGT-007 no-target-import certificate]
The candidate must include \(\mathcal{N}_E\). The certificate must rule out
target topology, target atlas, target metric, Lorentzian signature, proper
time, stress-energy tensor, matter action, detector semantics, empirical
calibration, benchmark success, Einstein equations, generated derivatives,
registry metadata, validator status, role authority, handoff status,
approval status, local cache state, file order, or commit status as
mathematical premises.
\end{condition}

\begin{condition}[MCPA-TGT-008 fail-closed obstruction labeling]
If a target condition cannot be met, the candidate must return exactly one
primary obstruction label from \(\mathcal{O}_E\) and must not repair the
failure by importing a forbidden target object or by treating support-only
project-control evidence as proof authority.
\end{condition}

\section{Eligibility Theorem}

\begin{theorem}[Precondition-assembly eligibility without matter-coupling adoption]
Suppose a future packet supplies \(\mathcal{D}^{\mathrm{MCPA}}_E\),
\(\PreCoupAsm_E\), a source-local candidate map or relation
\(\mathcal{A}^{\mathrm{MCPA}}_E\), and checks MCPA-TGT-001 through
MCPA-TGT-008. Then that packet becomes eligible only for a bounded Candidate
Constructor attempt result, Smuggling Auditor review, Refuter stress test,
Theoretical Continuation Selector routing, or separately authorized protected
evidence-status review. It does not derive or adopt matter coupling.
\end{theorem}

\begin{proof}
The hypotheses require only source-domain declarations, a pre-coupling
source-side codomain, scoped evidence/precondition separation, compatibility
between witness slots and matter-signature records, source-local dependence,
source-family naturality, finite-variation behavior, a no-target-import
certificate, and fail-closed obstruction labels. None of these hypotheses
includes canonical ontology edit, source-law adoption, \(\MatSrcDiscLaw\)
adoption, \(\SemMatterSrc\) adoption, \(\MatSigDisc\) adoption,
\(\SemMatterSig\) canonical adoption, \(\MetricData(E)\) adoption,
\(\geff\) scope expansion, coupling-law adoption, stress-energy semantics,
a stress-energy tensor, detector semantics, matter action, Einstein
equations, benchmark success, or completed derivation.

Therefore satisfying the target can only make a future candidate suitable for
the next controlled research roles that construct, audit, stress, route, or
separately review evidence status. The conclusion is eligibility for bounded
continuation under blocked adoption, not matter-coupling derivation or
adoption.
\end{proof}
