\section{Finite Source Context}

\begin{definition}[Finite source context]
Let
\[
  \Esrc=\{0,1,2\}.
\]
Define the nontrivial cyclic source relabeling
\[
  \sigma(0)=1,\qquad \sigma(1)=2,\qquad \sigma(2)=0.
\]
The source label is constant:
\[
  L(i)=\ell_{\mathrm{orb}}\quad\text{for every }i\in\Esrc.
\]
The source guard is
\[
  G(i)=\mathrm{inside\_cycle}(\Esrc,i).
\]
The response-token family is
\[
  R(i)=r_i,
\]
with the declared source response relation
\[
  Q(i,j) \Longleftrightarrow j=\sigma(i).
\]
The variation acts on response tokens by
\[
  \rho_{\sigma}(r_i)=r_{\sigma(i)}.
\]
All labels, guards, response tokens, and relations are source-side records.
They are not target atlas labels, detector events, empirical invariances,
stress-energy data, matter-action data, benchmark behavior, registry state,
validator state, role state, approval state, commit state, or local cache
state.
\end{definition}

\begin{definition}[Source objects before and after variation]
Define the source object
\[
  \Asrc=(\Esrc,L,G,R,Q,\mathrm{cycle\_domain}).
\]
Define its relabeled source object
\[
  \Asig=(\Esrc,L^{\sigma},G^{\sigma},R^{\sigma},Q^{\sigma},
  \mathrm{cycle\_codomain})
\]
by
\[
  L^{\sigma}(\sigma(i))=L(i),\qquad
  G^{\sigma}(\sigma(i))=G(i),\qquad
  R^{\sigma}(\sigma(i))=\rho_{\sigma}(R(i)),
\]
and
\[
  Q^{\sigma}(\sigma(i),\sigma(j)) \Longleftrightarrow Q(i,j).
\]
\end{definition}

\section{Direct Finite Verification}

\begin{lemma}[Response-relation invariance]
For every \(i,j\in\Esrc\),
\[
  Q(i,j) \Longleftrightarrow Q^{\sigma}(\sigma(i),\sigma(j)).
\]
\end{lemma}

\begin{proof}
The relation \(Q\) has exactly the three true ordered pairs
\[
  (0,1),\quad (1,2),\quad (2,0).
\]
Applying \(\sigma\) to both coordinates gives
\[
  (1,2),\quad (2,0),\quad (0,1),
\]
which is the same set of true ordered pairs. Therefore the response-token
successor relation is invariant under the relabeling.
\end{proof}

\begin{lemma}[Invariance certificate validity inside declared scope]
The certificate \(c_{\mathrm{inv}}\) is a valid finite/local source invariance
certificate inside
\[
  \Scope(c_{\mathrm{inv}})=\Esrc.
\]
Consequently
\[
  \Asrc \DeclEq \Asig \quad\text{inside }\Scope(c_{\mathrm{inv}})
\]
as a declared source-object equivalence only.
\end{lemma}

\begin{proof}
The map \(\sigma\) is a bijection of \(\Esrc\). The finite table verifies that
labels, guards, and response tokens are preserved by the declared source-side
actions \(L^{\sigma}\), \(G^{\sigma}\), and \(\rho_{\sigma}\). The previous
lemma verifies the response relation \(Q\) by direct enumeration of the three
true pairs. Thus the certificate payload is explicit, finite, source-side,
and nontrivial: the index relabeling is not the identity, but the declared
source structure is invariant. By the P4-T01 schema, this is a
\texttt{valid\_invariance\_certificate} with
\[
  \mathrm{expected\_equivalence\_result}
  =\mathrm{declared\_equivalence\_allowed}
\]
only inside the declared source scope.
\end{proof}

\section{Scope Boundary}

\begin{boundary}[Outside declared scope]
The certificate is valid only for the finite source context
\(\Esrc=\{0,1,2\}\), the cyclic relabeling \(\sigma\), the declared constant
source label, the declared source guard, and the declared successor relation
on response tokens. If a later packet changes the carrier, relation, labels,
guards, token family, or admissible variation class, this certificate does
not automatically apply. Outside the declared scope the record fails closed.
\end{boundary}

\begin{nonconclusion}[No physics promotion]
The invariance relation is a source-side finite relabeling invariance. It is
not empirical invariance, detector-frame invariance, Lorentz invariance,
metric invariance, matter-coupling invariance, benchmark recovery, or a
completed derivation.
\end{nonconclusion}
