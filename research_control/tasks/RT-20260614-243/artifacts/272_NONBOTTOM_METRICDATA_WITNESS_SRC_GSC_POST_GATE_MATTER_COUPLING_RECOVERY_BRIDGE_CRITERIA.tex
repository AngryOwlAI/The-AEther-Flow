\section{Recovery-Bridge Certificate}

\begin{definition}[Source-side recovery-bridge certificate]
A matter-coupling recovery-bridge certificate over \((E,\mathcal{F}_E)\) is a
tuple
\[
  \BridgeCert(E,\mathcal{F}_E)
  =
  (\mathcal{F}_E,\EvCrit(E),B_E,\mathcal{L}_E,\mathcal{P}_E,\mathcal{T}_E,
  \NoTargetImport_E,\mathcal{O}_E,\mathcal{G}_E),
\]
where:
\begin{enumerate}[label=(\roman*),leftmargin=*]
  \item \(\mathcal{F}_E\) declares the finite/local source family and its
  source-side readout/token domains;
  \item \(\EvCrit(E)\) records evidence/precondition status only;
  \item \(B_E\) is a required slot for a future source-side bridge relation or
  map candidate, not a map constructed by this packet;
  \item \(\mathcal{L}_E\) records compatibility with NC-MC-001 through
  NC-MC-008;
  \item \(\mathcal{P}_E\) is a proof-obligation ledger;
  \item \(\mathcal{T}_E\) is a finite/local test suite;
  \item \(\NoTargetImport_E\) is a certificate against target-side imports;
  \item \(\mathcal{O}_E\) maps failed criteria to fail-closed obstruction
  labels;
  \item \(\mathcal{G}_E\) records route and gate separation.
\end{enumerate}
\end{definition}

\section{Required Criteria}

\begin{criterion}[RBC-MC-001 evidence-input separation]
The certificate must distinguish \(\EvCrit(E)\) from adoption. It may cite
accepted scoped evidence/precondition status as an input, but it must not use
that status as a source law, coupling-law adoption, matter-coupling derivation,
stress-energy semantics, Einstein-equation derivation, benchmark promotion, or
completed derivation.
\end{criterion}

\begin{criterion}[RBC-MC-002 finite/local bridge domain declaration]
The certificate must name \(\mathcal{F}_E\), the source-side token/readout
domains, admissibility relation, transition data, and local comparison
families over which any future bridge candidate will be tested.
\end{criterion}

\begin{criterion}[RBC-MC-003 explicit candidate bridge relation slot]
The certificate must specify the type signature required of a future
candidate bridge relation
\[
  B_E:\mathsf{SrcTok}_E \times \mathsf{SrcState}_E
  \longrightarrow \mathsf{BridgeTok}_E \cup \{\bot\},
\]
or a documented equivalent source-side relation. The slot remains empty in
this packet. A later Candidate Constructor packet must either fill it under
source-side rules or return a precise obstruction.
\end{criterion}

\begin{criterion}[RBC-MC-004 NC-MC compatibility ledger]
The certificate must show how the future \(B_E\) would be checked against
NC-MC-001 through NC-MC-008 from the positive matter-coupling criteria packet.
Compatibility with \(\geff\) is a boundary check only and does not expand
\(\geff\)'s scope or adopt \(\MetricData(E)\).
\end{criterion}

\begin{criterion}[RBC-MC-005 proof-obligation ledger]
The certificate must separate at least the following statuses:
\[
\begin{array}{ll}
\text{evidence/precondition accepted} & \neq \text{source-law adoption},\\
\text{bridge criteria satisfied} & \neq \text{bridge constructed},\\
\text{bridge constructed} & \neq \text{matter coupling derived},\\
\text{audit/stress survived} & \neq \text{protected adoption},\\
\text{source conservation analogue} & \neq \text{stress-energy semantics}.
\end{array}
\]
Any missing step must remain explicitly undischarged.
\end{criterion}

\begin{criterion}[RBC-MC-006 finite/local discriminating tests]
The certificate must require tests that could distinguish a genuine
source-side bridge candidate from a restatement of \(\EvCrit(E)\). At minimum,
it must define tests for nonempty bridge outputs, admissibility preservation,
transition coherence, sector-universality analogue, and local perturbation
stability on \(\mathcal{F}_E\).
\end{criterion}

\begin{criterion}[RBC-MC-007 no-target-import certificate]
The certificate must include \(\NoTargetImport_E\), verifying that the future
bridge attempt does not use target topology, target atlas, target metric,
Lorentzian signature, proper time, target covariance, stress-energy tensor,
matter action, detector calibration, benchmark fit, Einstein equations,
validator status, registry metadata, generated derivatives, role authority,
handoff status, approval status, file order, commit status, local caches, or
human-readable PDFs as mathematical premises.
\end{criterion}

\begin{criterion}[RBC-MC-008 fail-closed obstruction completeness]
The certificate must map every failed bridge criterion to one of the following
fail-closed labels:
\[
\begin{array}{ll}
\mathrm{OB\mbox{-}MC\mbox{-}BRIDGE\mbox{-}EVIDENCE\mbox{-}AS\mbox{-}LAW}, &
\mathrm{OB\mbox{-}MC\mbox{-}BRIDGE\mbox{-}DOMAIN\mbox{-}MISSING},\\
\mathrm{OB\mbox{-}MC\mbox{-}BRIDGE\mbox{-}MAP\mbox{-}MISSING}, &
\mathrm{OB\mbox{-}MC\mbox{-}BRIDGE\mbox{-}NC\mbox{-}MC\mbox{-}FAIL},\\
\mathrm{OB\mbox{-}MC\mbox{-}BRIDGE\mbox{-}PROOF\mbox{-}GAP}, &

\mathrm{OB\mbox{-}MC\mbox{-}BRIDGE\mbox{-}GATE\mbox{-}CONFUSION}.

These labels preserve continuation or obstruction status without becoming
global theory rejection.
\end{criterion}

\begin{criterion}[RBC-MC-009 route and gate separation]
The certificate must state the next authorized route. Criteria satisfaction

\section{Recovery-Bridge Criteria Eligibility Theorem}

\begin{theorem}[Recovery-bridge criteria eligibility theorem]
Suppose a later draft/control packet supplies
\(\BridgeCert(E,\mathcal{F}_E)\) satisfying RBC-MC-001 through RBC-MC-009 for
a declared finite/local source family \(\mathcal{F}_E\). Then the later packet
is eligible only for bounded Candidate Constructor bridge construction,
Smuggling Auditor review, Refuter stress testing, or protected Gate Chair
review if separately authorized:
\[
  \BridgeCert(E,\mathcal{F}_E) \in
  \EligibleBridge(E,\mathcal{F}_E).
\]
This eligibility does not adopt a source law, adopt
\(\CouplingLaw(E;\Gbeta,T_{\mathrm{src}}(E))\), derive or adopt matter
coupling, import stress-energy semantics, adopt \(\MetricData(E)\), change
scoped \(\geff\), derive Einstein equations, promote benchmark status, or
claim completed derivation.
\end{theorem}

\begin{proof}
RBC-MC-001 prevents evidence/precondition acceptance from becoming adoption.
RBC-MC-002 fixes a source-side finite/local domain. RBC-MC-003 requires an
explicit future bridge slot while stating that this packet does not fill it.
RBC-MC-004 ties the bridge attempt to the prior NC-MC criteria without
expanding metric scope. RBC-MC-005 keeps every proof obligation separated.
RBC-MC-006 requires discriminating source-side tests. RBC-MC-007 blocks target
imports and nonmathematical authority imports. RBC-MC-008 makes failures
fail-closed rather than promotional. RBC-MC-009 separates routing from gate
authority. Therefore satisfaction of the criteria can only establish
eligibility for later bounded construction, audit, stress, or separately
authorized protected review. None of the hypotheses supplies a source-law
adoption verdict, matter-sector semantics, stress-energy tensor, matter
action, variational field equations, benchmark verdict, or completed
derivation.
\end{proof}
