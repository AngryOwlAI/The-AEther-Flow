\section{Direct Finite Verification}

\begin{lemma}[Factorization equality inside declared scope]
For every \(i\in\Esrc\),
\[
  h(i)=q(p(i)).
\]
\end{lemma}

\begin{proof}
There are three source-domain rows. For \(i=0\), \(p(0)=\alpha\) and
\(q(\alpha)=b_0=h(0)\). For \(i=1\), \(p(1)=\beta\) and
\(q(\beta)=b_1=h(1)\). For \(i=2\), \(p(2)=\alpha\) and
\(q(\alpha)=b_0=h(2)\). Hence the endpoint comparison factors through
\(\Fsrc^{\ast}\) by direct finite enumeration.
\end{proof}

\begin{lemma}[Factorization certificate validity inside declared scope]
The certificate \(c_{\mathrm{fac}}\) is a valid finite/local source
factorization certificate inside
\[
  \Scope(c_{\mathrm{fac}})=\Esrc\cup\{\alpha,\beta\}\cup\{b_0,b_1\}.
\]
Consequently
\[
  \Asrc \DeclEq \Bsrc \quad\text{through }\Fsrc^{\ast}
\]
as a declared source-object factorization only.
\end{lemma}

\begin{proof}
The factor object \(\Fsrc^{\ast}\) is explicit. The maps \(p\), \(q\), and
\(h\) are explicit finite source maps. The previous lemma proves \(h=q\circ p\)
on every source-domain row. The table also checks that labels and response
tokens match the declared factor object. By the P4-T01 schema, this is a
\texttt{valid\_factorization\_certificate} with
\[
  \mathrm{expected\_equivalence\_result}
  =\mathrm{declared\_equivalence\_allowed}
\]
only inside the declared source scope.
\end{proof}

\section{Declared \(RR_E\) Boundary Behavior}
