\section{Source Domain}

\begin{definition}[Source-domain package for the bridge target]
For a source context \(E\), the domain of \(\MCBT(E)\) is a declared
finite/local package
\[
  \mathcal{D}^{\mathrm{MCB}}_E =
  (E,\Msrc_E,\geff_E^{\mathrm{sc}},F_E^\sharp,\Sigma_E,
  \SEICand_E,\SEICertCand_E,\BalSrc_E,\FluxSrc_E,\mathcal{V}_E,
  \mathcal{P}_E).
\]
Here \(\Msrc_E\) denotes only the previously scoped source-side
\(\Msrc\) context, \(\geff_E^{\mathrm{sc}}\) denotes only the previously
scoped \(\geff\) compatibility context, \(F_E^\sharp\) is the finite/local
source family, \(\Sigma_E\) is a source-sector family, \(\BalSrc_E\) and
\(\FluxSrc_E\) are source-side balance and flux ledgers,
\(\mathcal{V}_E\) is the allowed source-variation family, and
\(\mathcal{P}_E\) is the proof-obligation ledger.

The entries \(\SEICand_E\) and \(\SEICertCand_E\) may be cited only as scoped
source-extension evidence/precondition data. They are not source-law
adoption, not a stress-energy tensor, not stress-energy semantics, not matter
coupling, and not Einstein equations.
\end{definition}

\section{Target Codomain}

\begin{definition}[Pre-tensorial source matter-interface codomain]
The codomain of \(\MCBT(E)\) is not an adopted stress-energy tensor. It is a
pre-tensorial source matter-interface record family
\[
  \PTMI_E =
  (I_E,Q_E,\Lambda_E,\mathsf{BalOK}_E,\mathsf{FluxOK}_E,
  \mathsf{NatOK}_E,\mathsf{SectOK}_E,\NoTargetImport_E,\mathcal{O}_E).
\]
Here \(I_E\) is a source-interface record set, \(Q_E\) is a source-side
quality or eligibility ledger, \(\Lambda_E\) records source-local dependence
data, and the four ``OK'' fields record balance/flux, naturality, and
sector-uniformity checks. The obstruction map \(\mathcal{O}_E\) maps failures
to fail-closed labels.

Tensorial promotion is a separate missing burden. A future packet may not
reinterpret \(\PTMI_E\) as a stress-energy tensor unless a protected later
route supplies explicit tensorial semantics and the required gates authorize
that stronger claim.
\end{definition}

\begin{definition}[Matter-Coupling Bridge Target v1]
\(\MCBT(E)\) is the obligation that a future candidate provide either a
source-local partial map or source-local relation
\[
  \mathcal{B}^{\mathrm{MCB}}_E:
  \mathcal{D}^{\mathrm{MCB}}_E \dashrightarrow \PTMI_E
\]
or a precise fail-closed obstruction label from \(\mathcal{O}_E\), while
satisfying Conditions MCB-TGT-001 through MCB-TGT-008 below.
\end{definition}

\section{Target Conditions}

\begin{condition}[MCB-TGT-001 source-domain discipline]
All inputs must be source-side records in
\(\mathcal{D}^{\mathrm{MCB}}_E\). Accepted scoped evidence may appear only as
precondition data. It may not be converted into adoption, matter coupling,
stress-energy semantics, stress-energy tensor semantics, matter action, or
Einstein-equation premises.
\end{condition}

\begin{condition}[MCB-TGT-002 pre-tensorial codomain discipline]
The output must land in \(\PTMI_E\) or fail closed. If tensorial output is
needed, the candidate must record the missing tensorial-promotion burden
rather than importing a target stress-energy tensor.
\end{condition}

\begin{condition}[MCB-TGT-003 locality]
\(\mathcal{B}^{\mathrm{MCB}}_E\) must depend only on declared source-local
records in a finite/local neighborhood of \(E\) inside \(F_E^\sharp\). It may
not depend on target topology, target atlas, target metric, proper time,
detector calibration, empirical matter fields, benchmark fit, or Einstein
equations.
\end{condition}

\begin{condition}[MCB-TGT-004 covariance and naturality]
For each source-family morphism or relabeling
\[
  \rho: \mathcal{D}^{\mathrm{MCB}}_E \to
  \mathcal{D}^{\mathrm{MCB}}_{E'},
\]
the candidate must supply a corresponding source-side transport
\(\rho_*:\PTMI_E\to \PTMI_{E'}\) such that successful outputs commute with
\(\rho\). Failure to define this compatibility is noncovariance, not a reason
to import target covariance by hand.
\end{condition}

\begin{condition}[MCB-TGT-005 sector-uniformity]
The rule schema for \(\mathcal{B}^{\mathrm{MCB}}_E\) must be uniform across a
parameterized or graph-indexed source-sector family \(\Sigma_E\). It must be
stronger than a two-sector shared-form check: the same source-side schema must
operate across declared sectors with only declared source parameters varying.
\end{condition}

\begin{condition}[MCB-TGT-006 conservation-analogue compatibility]
The output must include a source-side balance/flux compatibility ledger using
\(\BalSrc_E\) and \(\FluxSrc_E\). This is not the Bianchi identity, not
stress-energy conservation, not a variational matter action, and not Einstein
equations. It is only a pre-tensorial compatibility obligation.
\end{condition}

\begin{condition}[MCB-TGT-007 no-target-import certificate]
The candidate must include \(\NoTargetImport_E\). The certificate must rule
out target topology, target atlas, target metric, Lorentzian signature,
proper time, stress-energy tensor, matter action, detector semantics,
empirical fit, benchmark success, Einstein equations, generated derivatives,
registry metadata, validator status, role authority, handoff status,
approval status, local cache state, file order, or commit status as
mathematical premises.
\end{condition}

\begin{condition}[MCB-TGT-008 fail-closed obstruction labeling]
If a target condition cannot be met, the candidate must return exactly one
primary obstruction label from the target obstruction family and must not
repair the failure by importing a forbidden target object.
\end{condition}

\section{Eligibility Theorem}

\begin{theorem}[Bridge-target eligibility without matter-coupling adoption]
Suppose a future packet supplies \(\mathcal{D}^{\mathrm{MCB}}_E\),
\(\PTMI_E\), a source-local candidate map or relation
\(\mathcal{B}^{\mathrm{MCB}}_E\), and checks MCB-TGT-001 through
MCB-TGT-008. Then that packet becomes eligible only for a bounded Candidate
Constructor attempt result, Smuggling Auditor review, Refuter stress test, or
separately authorized protected evidence-status review. It does not derive or
adopt matter coupling.
\end{theorem}

\begin{proof}
The hypotheses require only source-domain declarations, a pre-tensorial
codomain, source-local dependence, source-family naturality,
sector-uniformity, balance/flux compatibility, a no-target-import certificate,
and fail-closed obstruction labels. None of these hypotheses includes
canonical ontology edit, source-law adoption, \(\MetricData(E)\) adoption,
scoped-\(\geff\) expansion, coupling-law adoption, stress-energy semantics,
an adopted stress-energy tensor, matter action, detector semantics, Einstein
equations, benchmark success, or completed derivation.

Therefore satisfying the target can only make a future candidate suitable for
the next controlled research roles that test or review the candidate. The
conclusion is eligibility for construction, audit, stress, or separately
authorized protected evidence-status review, not matter-coupling derivation
or adoption.
\end{proof}
