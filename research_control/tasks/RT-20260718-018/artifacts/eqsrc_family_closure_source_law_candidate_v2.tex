\section{Closed Typed Source Grammar}

\begin{definition}[Declared source signature]
A candidate instance begins with a source signature
\[
 \Sigma_{\mathrm{src}}
 =
 \bigl(
 \mathcal F_{\mathrm{src}},
 \Lambda_{\mathrm{src}},
 \mathcal P_{\mathrm{src}},
 \mathcal N_{\mathrm{src}},
 \Pi^{0}_{\mathrm{src}},
 \mathsf{LedgerOK},
 \mathsf{ProxyOK},
 \mathsf{NegPass}
 \bigr)
\]
declared before any target reconstruction or comparison. Its components are:
\begin{enumerate}[leftmargin=*]
  \item a set \(\mathcal F_{\mathrm{src}}\) of declared source objects;
  \item a source ledger-label domain \(\Lambda_{\mathrm{src}}\), equipped with
  a declared congruence \(\equiv_{\mathcal L}\), typed neutral labels,
  source inverse \((-)^\dagger\), and composable product \(\odot\);
  \item a source proxy-incidence domain \(\mathcal P_{\mathrm{src}}\) with a
  declared typed predicate \(\mathsf{ProxyOK}_{A,B}\);
  \item a finite declared set \(\mathcal N_{\mathrm{src}}\) of negative
  controls and a source decision predicate
  \(\mathsf{NegPass}_{n}\) for every \(n\in\mathcal N_{\mathrm{src}}\);
  \item typed base provenance symbols
  \(\Pi^{0}_{\mathrm{src}}(A,B)\);
  \item source predicates \(\mathsf{LedgerOK}_{A,B}\),
  \(\mathsf{ProxyOK}_{A,B}\), and \(\mathsf{NegPass}_{n}\), whose arguments
  range only over the preceding source domains.
\end{enumerate}
No symbol in \(\Sigma_{\mathrm{src}}\) may be defined by a target atlas, target
metric, reconstructed geometry, detector outcome, benchmark result, registry
state, role declaration, validator result, generated derivative, handoff,
checkpoint, or approval status.
\end{definition}

\begin{definition}[Closed provenance term algebra]
For each ordered pair \(A,B\in\mathcal F_{\mathrm{src}}\), the provenance
terms \(\PiSrc(A,B)\) are the least typed family generated by:
\[
\begin{aligned}
 &\pi\in\Pi^{0}_{\mathrm{src}}(A,B)
   &&\Longrightarrow \pi\in\PiSrc(A,B),\\
 &\mathsf{id}[A] &&\in\PiSrc(A,A),\\
 &\pi\in\PiSrc(A,B)
   &&\Longrightarrow \mathsf{inv}[\pi]\in\PiSrc(B,A),\\
 &\pi\in\PiSrc(A,B),\ \rho\in\PiSrc(B,C)
   &&\Longrightarrow \mathsf{comp}[\rho,\pi]\in\PiSrc(A,C).
\end{aligned}
\]
``Least'' is a closure condition: there are no additional provenance
constructors and no opaque escape value. Thus source provenance is target-free
by construction rather than by an unconstrained predicate named ``source.''
\end{definition}

\begin{definition}[Typed certificates and acceptance]
A certificate in \(\Cert(A,B)\) is a tuple
\[
 w=(A,B,\lambda_w,p_w,\pi_w)
 \in
 \{A\}\times\{B\}\times\Lambda_{\mathrm{src}}
 \times\mathcal P_{\mathrm{src}}\times\PiSrc(A,B).
\]
Its acceptance predicate is the explicit conjunction
\[
\begin{split}
 \Accept(w)\quad\Longleftrightarrow\quad&
 w\in\Cert(A,B)
 \ \wedge\ \mathsf{LedgerOK}_{A,B}(\lambda_w)\\
 &\wedge\ \mathsf{ProxyOK}_{A,B}(p_w)
 \ \wedge\
 \bigwedge_{n\in\mathcal N_{\mathrm{src}}}
 \mathsf{NegPass}_{n}(w)
 \ \wedge\ \pi_w\in\PiSrc(A,B).
\end{split}
\]
Every conjunct has a declared source-only domain. Missing, ill-typed,
conflicting, or extra-grammatical data returns \(\Block\).
\end{definition}

\section{Candidate Accepted-Totality Law}

\begin{definition}[Proposal-only candidate closure law v2]
A declared signature and certificate family satisfy \(\Law\) only when all of
the following clauses hold.

\begin{enumerate}[leftmargin=*]
  \item \textbf{Accepted identity totality.}
  For every \(A\in\mathcal F_{\mathrm{src}}\), there is a distinguished
  certificate
  \[
    \mathbf 1_A
    =(A,A,e_A,p_{\mathsf{id},A},\mathsf{id}[A])
    \in\Cert(A,A)
  \]
  and \(\Accept(\mathbf 1_A)\) is true.

  \item \textbf{Accepted inverse totality.}
  For every \(w\in\Cert(A,B)\), if \(\Accept(w)\), then there exists a
  certificate
  \[
    w^{-1}
    =(B,A,\lambda_w^\dagger,p_{\mathsf{inv}}(w),
      \mathsf{inv}[\pi_w])
    \in\Cert(B,A)
  \]
  such that \(\Accept(w^{-1})\). The witness is source-side and its existence
  is asserted, not merely constrained if it happens to exist.

  \item \textbf{Accepted composition totality.}
  For all accepted \(w\in\Cert(A,B)\) and
  \(v\in\Cert(B,C)\), there exists a certificate
  \[
    v\circ w
    =(A,C,\lambda_v\odot\lambda_w,
      p_{\mathsf{comp}}(v,w),
      \mathsf{comp}[\pi_v,\pi_w])
    \in\Cert(A,C)
  \]
  such that \(\Accept(v\circ w)\).

  \item \textbf{Certificate congruence.}
  A typed source congruence \(\simeq_{\mathrm{src}}\) compares certificates
  only when endpoints agree and their ledger labels, proxy-incidence data,
  negative-control outcomes, and provenance terms agree under declared
  source congruences. It is itself an equivalence relation and preserves
  \(\Accept\).

  \item \textbf{Unit compatibility.}
  For each accepted \(w:A\to B\),
  \[
    w\circ\mathbf 1_A\simeq_{\mathrm{src}}w,
    \qquad
    \mathbf 1_B\circ w\simeq_{\mathrm{src}}w.
  \]

  \item \textbf{Inverse compatibility.}
  For each accepted \(w:A\to B\),
  \[
    (w^{-1})^{-1}\simeq_{\mathrm{src}}w,\qquad
    w^{-1}\circ w\simeq_{\mathrm{src}}\mathbf 1_A,\qquad
    w\circ w^{-1}\simeq_{\mathrm{src}}\mathbf 1_B.
  \]

  \item \textbf{Associativity compatibility.}
  For every composable accepted triple
  \(w:A\to B\), \(v:B\to C\), and \(u:C\to D\),
  \[
    (u\circ v)\circ w
    \simeq_{\mathrm{src}}
    u\circ(v\circ w).
  \]

  \item \textbf{Ledger congruence and constructor closure.}
  The source ledger operations satisfy, whenever the expressions are typed,
  \[
  e_B\odot\lambda_w\equiv_{\mathcal L}\lambda_w
  \equiv_{\mathcal L}\lambda_w\odot e_A,\quad
  (\lambda_w^\dagger)^\dagger\equiv_{\mathcal L}\lambda_w,
  \]
  \[
  \lambda_w^\dagger\odot\lambda_w
  \equiv_{\mathcal L}e_A,\quad
  \lambda_w\odot\lambda_w^\dagger
  \equiv_{\mathcal L}e_B,
  \]
  and
  \[
  (\lambda_u\odot\lambda_v)\odot\lambda_w
  \equiv_{\mathcal L}
  \lambda_u\odot(\lambda_v\odot\lambda_w).
  \]
  The proxy and negative-control predicates must also pass for the identity,
  inverse, and composite witnesses whose acceptance is required above.

  \item \textbf{Fail-closed source purity.}
  If any required witness, typing judgment, source constructor, ledger
  congruence, proxy-incidence judgment, negative-control result, or provenance
  term is missing or fails, the declared family does not satisfy \(\Law\).
  No target or process datum may repair that failure.
\end{enumerate}
\end{definition}

\section{Conditional Equivalence Theorem}

Define
\[
 A\sim_{\mathrm{cand},v2}B
 \quad\Longleftrightarrow\quad
 \exists w\in\Cert(A,B)\ \Accept(w).
\]

\begin{proposition}[Equivalence under the repaired candidate law]
If a declared source family satisfies every clause of \(\Law\), then
\(\sim_{\mathrm{cand},v2}\) is an equivalence relation on
\(\mathcal F_{\mathrm{src}}\).
\end{proposition}

\begin{proof}
For each \(A\), accepted identity totality supplies
\(\mathbf 1_A\in\Cert(A,A)\) with \(\Accept(\mathbf 1_A)\), proving
reflexivity. If \(A\sim_{\mathrm{cand},v2}B\), choose an accepted
\(w:A\to B\). Accepted inverse totality supplies an accepted
\(w^{-1}:B\to A\), proving symmetry. If
\(A\sim_{\mathrm{cand},v2}B\) and
\(B\sim_{\mathrm{cand},v2}C\), choose accepted witnesses
\(w:A\to B\) and \(v:B\to C\). Accepted composition totality supplies an
accepted \(v\circ w:A\to C\), proving transitivity. Every witness remains in
the closed typed source grammar by the constructor and acceptance clauses.
\end{proof}

\begin{remark}
Unit, inverse, associativity, and ledger-congruence compatibility make the
certificate operations coherent for later use. The three equivalence axioms
need only the accepted existence clauses identified in the proof. This
separation prevents coherence requirements from concealing the exact
totality burden.
\end{remark}

\section{Obstruction Repair and Fail-Closed Branches}

\paragraph{Statement-level repair.}
The accepted identity, inverse, and composition clauses above directly negate
the three v1 countermodel freedoms: an identity may no longer be typed but
unaccepted; an accepted \(A\to B\) witness may no longer lack an accepted
\(B\to A\) inverse; and composable accepted \(A\to B\) and \(B\to C\)
witnesses may no longer lack an accepted \(A\to C\) composite. Therefore
\texttt{OBST-EQSRC-CLOSURE-TOTALITY-001} is repaired at the proposal's
statement level.

\paragraph{Missing-witness branch.}
If any one of those accepted witnesses cannot be exhibited from the declared
source domains, the family fails to satisfy \(\Law\). The conditional theorem
cannot be invoked. The underlying accepted-certificate relation may be studied
separately, but no closure conclusion is supplied by this packet.

\paragraph{Open-grammar branch.}
If a ledger label, proxy-incidence datum, negative control, provenance token,
or decision rule is populated by an undeclared constructor or opaque external
value, acceptance returns \(\Block\). Naming a predicate ``source'' does not
close its grammar.

\paragraph{Ledger or coherence mismatch branch.}
If identity, inverse, or composition changes a protected source annotation or
fails unit, involution, inverse, or associativity compatibility, the candidate
returns \(\Block\). A target metric, detector output, benchmark result,
registry status, role text, validator pass, generated derivative, handoff, or
checkpoint cannot create the missing source witness.
