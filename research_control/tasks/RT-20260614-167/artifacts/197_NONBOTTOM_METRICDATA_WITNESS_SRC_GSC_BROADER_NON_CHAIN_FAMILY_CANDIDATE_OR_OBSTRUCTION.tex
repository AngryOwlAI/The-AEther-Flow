\label{source-transition-covariance-lemma}

\begin{lemma}[Source transition covariance]
Let \(\tau:\Bn\to \Bn\) be an admitted source transition token. Then
\[

\section{Source Distinguishability Lemma}
\label{source-distinguishability-lemma}

\begin{lemma}[Rooted tail-branch profile distinguishability]
For every \(n\ge 2\), the strict reachability profile \(p_n\) is injective on
\(S_n\).
\end{lemma}

\begin{proof}
The root \(r\) is unique because \(p_n(r)=(n+1,0)\), so it has zero strict
predecessors and maximal strict successors.

For tail sites, if \(i\ne j\), then
\[
p_n(x_i)=(n-i,i)\ne(n-j,j)=p_n(x_j),
\]
because the second coordinates differ.

For \(i<n\), \(p_n(x_i)\) has positive first coordinate \(n-i>0\), while
\(p_n(y)=(0,1)\). Hence \(y\ne x_i\) by profile. For \(i=n\),
\[
p_n(x_n)=(0,n)\ne(0,1)=p_n(y)
\]
because \(n\ge 2\). Therefore all source sites in \(S_n\) are profile
distinguished.
\end{proof}
