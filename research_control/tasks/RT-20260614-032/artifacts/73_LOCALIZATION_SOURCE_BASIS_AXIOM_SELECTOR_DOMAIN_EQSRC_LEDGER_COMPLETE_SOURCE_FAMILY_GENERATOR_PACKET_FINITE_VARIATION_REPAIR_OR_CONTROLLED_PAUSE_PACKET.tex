\section{Representative-Equivalence Interface}

\begin{definition}[Representative record]
For this repair packet, a source representative record is a tuple
\[
R(X)=
(q_X,b_X,i_X,c_X,\partial_X,h_X),
\]
where \(q_X\) is the quotient representative label, \(b_X\) is the
blocker ledger, \(i_X\) is the influence-provenance ledger, \(c_X\)
is the certificate-history ledger, \(\partial_X\) is the ordered
boundary-anchor record, and \(h_X\) is the contraction-history record.
All entries are source-ledger fields. No target metric, target proper
time, target locality, observer protocol, benchmark rank, candidate
behavior, validation status, registry status, generated visibility,
handoff status, or authority label is an entry of \(R(X)\).
\end{definition}

\begin{repair}[Exact representative baseline]
Define the conservative representative relation
\[
X \,\Theta_Q^{=}\, Y
\quad\Longleftrightarrow\quad
R(X)=R(Y).
\]
This supplies a source-only baseline for \(\RepLC\). It is intentionally
strict: it permits no loss of representative, branch, boundary,
certificate, or provenance information.
\end{repair}

\begin{lemma}[Exact representative preservation]
If \(X \,\Theta_Q^{=}\, Y\), then every coordinate of \(R(X)\) equals
the corresponding coordinate of \(R(Y)\). Therefore exact readout
comparison and exact inverse-provenance recovery do not lose
representative history under \(\Theta_Q^{=}\).
\end{lemma}

\begin{proof}
The conclusion is immediate from tuple equality. The proof is
source-side and does not use any downstream target primitive. It is
also conservative: it proves only the identity-policy case, not a
nontrivial quotient.
\end{proof}

\begin{bracket}[Nontrivial representative quotient]
Any relaxed relation \(\Theta_Q^{+}\) that identifies records with
different \(R\)-coordinates remains unsupplied. A future
\(\Theta_Q^{+}\) must provide:
\begin{enumerate}
    \item a source-only equivalence definition;
    \item a lifted readout or an explicit exact-readout retention rule;
    \item certificate-composition preservation;
    \item inverse-provenance preservation; and
    \item a negative witness showing what fails when any required
    coordinate is omitted.
\end{enumerate}
Until those data exist, \(\Theta_Q^{+}\) is in controlled \(\Pause\),
not \(\Pass\).
\end{bracket}

\section{Boundary-Equivalence Interface}

\begin{definition}[Boundary record]
Let
\[
B(X)=(\partial_X,\lambda_X,\rho_X)
\]
where \(\partial_X\) is the ordered boundary-anchor record,
\(\lambda_X\) is the source-side boundary-domain label, and \(\rho_X\)
is the source-side rule explaining how the boundary record is carried
through a certificate step.
\end{definition}

\begin{repair}[Exact boundary baseline]
Define
\[
X \,\Theta_{\partial}^{=}\, Y
\quad\Longleftrightarrow\quad
B(X)=B(Y).
\]
The exact boundary transport law is identity transport on \(B\) across
certificate steps whose source ledger already records the same
boundary record before and after the step.
\end{repair}

\begin{lemma}[Exact boundary preservation]
For certificate steps whose ledgers preserve \(B\) exactly, exact
boundary equivalence is preserved under finite composition whenever
the composed certificate is defined in the composition ledger of
Section~\ref{sec:composition}.
\end{lemma}

\begin{proof}
Each step preserves the same ordered boundary tuple. Concatenating
defined steps preserves the tuple by repeated equality. No target
geometry, observer protocol, causal structure, clock behavior, matter
coupling, benchmark rank, candidate behavior, or downstream success
enters the argument.
\end{proof}

\begin{bracket}[Boundary relaxation]
Any boundary relaxation \(\Theta_{\partial}^{+}\) that identifies
different boundary records remains unsupplied. It must supply a
source-only transport law, certificate preservation proof, inverse
recovery proof, and readout treatment before it can be counted as a
repair. Exact boundary equality is a conservative baseline, not
finite-variation boundary robustness.
\end{bracket}

\section{Composition Ledger and Inverse Provenance}
\label{sec:composition}

\begin{definition}[Partial source composition ledger]
Define a partial ledger category \(\mathcal{C}_{\mathrm{lc}}^{=}\) as
follows.
\begin{enumerate}
    \item Objects are finite \(\ObjLC\) source records carrying
    \(R(X)\), \(B(X)\), and \(\ReadLC(X)\).
    \item A certificate step \(s:X\to Y\) is recorded as
    \[
    s=(X,Y,R_{\mathrm{in}},R_{\mathrm{out}},B_{\mathrm{in}},
    B_{\mathrm{out}},P_s,I_s),
    \]
    where \(P_s\) is the forward source-provenance payload and \(I_s\)
    is the inverse-provenance token.
    \item Identity steps carry unchanged \(R\), \(B\), \(P\), and \(I\)
    records.
    \item \(s_2\circ s_1\) is defined only when the target object of
    \(s_1\) equals the source object of \(s_2\), the intermediate
    representative records match exactly, and the intermediate boundary
    records match exactly.
    \item The composed provenance payload is ordered concatenation
    \(P_{s_2\circ s_1}=(P_{s_1},P_{s_2})\), and the composed inverse
    token is ordered reverse recovery
    \(I_{s_2\circ s_1}=(I_{s_2},I_{s_1})\).
\end{enumerate}
\end{definition}

\begin{lemma}[Defined composition is associative]
For triples whose pairwise and triple intermediate records match
exactly, composition in \(\mathcal{C}_{\mathrm{lc}}^{=}\) is
associative.
\end{lemma}

\begin{proof}
The only composition operation on provenance data is ordered
concatenation of source records, and concatenation is associative when
all intermediate equality checks pass. The lemma is silent about
triples with nonmatching representatives or boundary records; those
remain undefined rather than repaired by assumption.
\end{proof}

\begin{repair}[Inverse-provenance token requirement]
For an accepted step \(s:X\to Y\), \(I_s\) must contain enough
source-ledger data to recover:
\[
(b_X,i_X,q_X,\partial_X,h_X)
\quad\text{from}\quad
(Y,P_s,I_s).
\]
If two endpoint-equal histories differ in branch provenance, influence
assignment, quotient representative, boundary anchor, or contraction
history, \(I_s\) must discriminate them or provide a source-side proof
that the difference is irrelevant to \(\CertLC\).
\end{repair}

\begin{bracket}[Composition beyond exact matching]
The ledger above repairs only the exact-matching case. It does not
repair noncomposing quotient representatives, relaxed boundary
records, omitted histories, or endpoint-equal provenance shifts.
Those cases require \(\Theta_Q^{+}\), \(\Theta_{\partial}^{+}\), and
additional inverse-provenance proofs before they can move out of
\(\Pause\) or \(\Repair\).
\end{bracket}

\section{Negative-Witness Coverage}

\begin{definition}[Schematic negative row]
A schematic negative row is a source-side test specification
\[
N=(V,\Delta,\mathrm{expected\;failed\;burden},\mathrm{forbidden\;imports})
\]
where \(V\in\VarLC\), \(\Delta\) is a source-ledger mutation, and the
expected failed burden is one of \(\RepLC,\BndLC,\CompLC,\InvLC,
\MinLC,\DiscLC,\NegLC\). A schematic row is not yet a concrete
\(\ObjLC\) witness.
\end{definition}

\begin{repair}[Negative-witness schematic suite]
The repair packet supplies the following missing-family schematic
suite:
\begin{enumerate}
    \item \(N_{\mathrm{infl}}\): move an influence record between
    refined branches while preserving endpoint influence counts;
    expected failure is \(\InvLC\) or \(\DiscLC\).
    \item \(N_{\mathrm{rep}}\): substitute a distinct representative
    tuple \(R'\neq R\) while keeping a coarse endpoint readout fixed;
    expected failure is \(\RepLC\).
    \item \(N_{\mathrm{comp}}\): compose two locally plausible steps
    whose intermediate representative records do not match exactly;
    expected failure is \(\CompLC\).
    \item \(N_{\partial}\): perturb the ordered boundary record while
    keeping endpoint counts fixed; expected failure is \(\BndLC\).
    \item \(N_{\mathrm{read\_del}}\): delete one audit coordinate and
    preserve the remaining endpoint readout; expected failure is
    \(\DiscLC\) unless irrelevance is proved.
    \item \(N_{\mathrm{read\_relax}}\): quotient a readout coordinate
    without a source-only lifted readout proof; expected failure is
    \(\DiscLC\) or \(\RepLC\).
    \item \(N_{\mathrm{endpoint}}\): preserve endpoint multisets while
    changing branch-level provenance; expected failure is \(\InvLC\).
\end{enumerate}
\end{repair}

\begin{bracket}[Concrete witness instantiation]
The schematic suite does not complete \(\NegLC\). Each row must still
be instantiated as a finite source-side \(\ObjLC\) object pair or
certificate-step family, checked against the actual \(\CertLC\)
relation, and audited for hidden target import. Until then,
\(\NegLC\) remains in \(\Repair\).
\end{bracket}
