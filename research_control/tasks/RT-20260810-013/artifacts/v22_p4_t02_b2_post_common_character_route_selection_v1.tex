\begin{definition}[Oriented sector configuration]
Let \(S\) be a finite nonempty declared sector set and \(V\) a finite-dimensional
real vector space of rank \(1\le r\le4\).  For every \(s\in S\), let
\([v_s]_+\) be a nonzero oriented ray.  Choose one positive representative
\(v_s\) and define
\[
  A:\R^S\longrightarrow V,\qquad A e_s=v_s .
\]
The set \(S\), each orientation, the exact representation of \(v_s\), and its
source provenance must be fixed before any feasibility calculation.
\end{definition}

\begin{definition}[Covectors, chamber, and positive circuits]
The complete realized sign-covector set, strict positive chamber, and family
of minimal positive circuits are
\begin{align}
  \Cstar(A)
    &=\left\{\bigl(\sgn k(v_s)\bigr)_{s\in S}:k\in V^*\right\},\\
  \GammaA(A)
    &=\left\{k\in V^*:k(v_s)>0\ \text{for every }s\in S\right\},\\
  \Cplus(A)
    &=\left\{C\subseteq S:
      \begin{array}{l}
      C\ne\varnothing,\ \exists\,\alpha_s>0\ (s\in C),\\
      \sum_{s\in C}\alpha_s v_s=0,\ \text{and }C
      \text{ is inclusion-minimal}
      \end{array}
      \right\}.
\end{align}
These are formal source-side compatibility data.  They are not physical
causal covectors, calibrated responses, or a characteristic cone.
\end{definition}

\begin{definition}[Total bridge-facing classifier]

\section{Selector-level mathematical payload}

\begin{theorem}[Strict finite alternative]
Exactly one of the following holds:
\begin{enumerate}
  \item there is \(k\in V^*\) such that \(k(v_s)>0\) for every \(s\in S\);
  \item there is \(\alpha\in\R_{\ge0}^{S}\setminus\{0\}\) such that
        \(A\alpha=0\).
\end{enumerate}
\end{theorem}
\begin{proof}
The branches are exclusive: if both held, then
\[
  0=k(A\alpha)=\sum_{s\in S}\alpha_s k(v_s)>0.
\]
For exhaustiveness, apply finite-dimensional strict separation to the linear
image \(A^*(V^*)\subseteq\R^S\) and the open positive orthant.  If the image
does not meet that orthant, the separating functional can be chosen as a
nonzero \(\alpha\ge0\) annihilating \(A^*(V^*)\), equivalently \(A\alpha=0\).
\end{proof}

\begin{lemma}[Minimal positive-circuit bound]
Choose an obstruction coefficient with inclusion-minimal support \(C\).
Then every coefficient on \(C\) is strictly positive,
\[
  \rank A_C=|C|-1,\qquad
  \dim\ker A_C=1,\qquad
  |C|\le r+1\le5.
\]
\end{lemma}
\begin{proof}
A zero coefficient could be removed.  If \(\ker A_C\) had dimension above one,
a nonproportional kernel vector could be added with a suitable scalar until
one positive coefficient reached zero while none became negative.  That would
give a nonzero nonnegative dependence on a proper subset, contradicting
minimality.  Thus nullity is one, rank-nullity gives
\(|C|=\rank A_C+1\), and \(\rank A_C\le r\le4\).
\end{proof}

\begin{theorem}[Positive-presentation descent]
Let \(T\in\operatorname{GL}(V)\), let \(D\) be a positive diagonal matrix on
\(\R^S\), and let \(P\) permute the sectors.  For
\[
  A'=TADP^{-1},
\]
the realized covectors, strict-feasibility branch, minimal positive-circuit
supports, circuit rank, and total certificate transport bijectively under the
induced dual and label maps.
\end{theorem}
\begin{proof}
For \(k'\in V^*\),
\[
  A'^*k'=PD\,A^*T^*k'.
\]
The positive diagonal matrix preserves every coordinate sign and \(P\) only
relabels coordinates.  Strict positivity therefore transports bijectively.
If \(A_C\alpha=0\) with \(\alpha>0\), then the transformed dependence has
positive coefficients \(D_C^{-1}\alpha\), up to the sector permutation.
Invertible maps preserve support minimality, rank, and kernel dimension.
\end{proof}

\begin{lemma}[Finite-sector refinement]
Let \(S\subseteq T\) and retain the old rays in one declared ambient source
space.  Then:
\begin{enumerate}
  \item coordinate restriction
        \(\rho_{T,S}:\Cstar(A_T)\to\Cstar(A_S)\) is surjective and composes
        under nested inclusions;
  \item \(\GammaA(A_T)\subseteq\GammaA(A_S)\);
  \item every positive circuit whose support lies in \(S\) persists after
        adjoining the sectors in \(T\setminus S\).
\end{enumerate}
\end{lemma}
\begin{proof}
Any \(k\in V^*\) realizing a sign vector on \(S\) realizes an extension on
\(T\), proving surjectivity.  Positivity on all of \(T\) implies positivity on
\(S\).  A dependence and the minimality of its fixed support depend only on
the columns in that support and hence are unchanged by adjoining columns.
\end{proof}

\begin{proposition}[Fine-magnitude and holonomy nonreconstruction]
The selected bridge forgets both fine positive-ray magnitude and
common-character holonomy data.
\end{proposition}
\begin{proof}
For fine magnitude, \(A=(e_1,e_2,e_3)\) and
\(A'=(2e_1,e_2,e_3)\)\space have identical realized signs, empty positive-circuit
families, and nonempty positive chambers, although a first-to-second magnitude
ratio changes by a factor of two.

For holonomy, enlarge the parameter space to
\(\operatorname{Conf}(S,V)\times H^1(K;\R)^S\) with
\(H^1(K;\R)\ne0\) and \(|S|\ge2\).  The bridge factors through projection onto
\(\operatorname{Conf}(S,V)\).  A fiber contains both
\((0,\ldots,0)\) and \((0,\eta,0,\ldots,0)\) for nonzero
\(\eta\in H^1(K;\R)\); the first is diagonal and the second is not.  Therefore
no bridge output can decide common-character holonomy protection.
\end{proof}

\begin{proposition}[Total-alternative preload neutrality]
For inputs fixed independently of the desired response, a classifier does not
preload success merely by returning a positive certificate when the same rule
is total and accepts the mutually exclusive obstruction certificate as an
equally terminal result.
\end{proposition}
\begin{proof}
Success is an output branch rather than an input predicate.  The strict
alternative proves both exhaustiveness and exclusivity.  The criterion fails
if sectors, orientations, refinements, or exact representatives are selected
after evaluation because they produce the desired branch.
\end{proof}
