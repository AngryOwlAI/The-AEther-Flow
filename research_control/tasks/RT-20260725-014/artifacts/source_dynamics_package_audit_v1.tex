\section{Fresh mathematical stress payload}

\begin{definition}[Odd-power stress family]
For an integer \(m\geq1\) and \(\gamma>0\), define
\[
  X_{m,\gamma}(a)=-\gamma a^{2m+1}
\]
and
\[
  \Flow^{(m,\gamma)}_\lambda(a)
  =
  \frac{a}{\left(1+2m\gamma a^{2m}\lambda\right)^{1/(2m)}},
  \qquad \lambda\geq0 .
  \tag{1}
\]
The P5-T03 candidate is the member \(m=1\).
\end{definition}

\begin{theorem}[Odd-power nonselection counterfamily]
\label{thm:counterfamily}
Every \(m\geq1\) and \(\gamma>0\) in (1) gives a total continuous unital
forward semiflow on \(\mathbb R\).  It is odd under \(a\mapsto-a\), has the
unique stationary amplitude \(a=0\), strictly decreases \(|a|\) along every
nonzero forward orbit, converges to zero polynomially, and has a finite
backward horizon for every nonzero state.  Consequently the qualitative
P5-T02 through P5-T05 requirements do not select the cubic exponent \(3\).
\end{theorem}

\begin{proof}
Direct differentiation gives
\[
 \frac{d}{d\lambda}\Flow^{(m,\gamma)}_\lambda(a)
 =-\gamma\left(\Flow^{(m,\gamma)}_\lambda(a)\right)^{2m+1}.
\]
Uniqueness for this smooth scalar vector field gives the semigroup law, which
also follows by substituting (1) twice.  The denominator is positive for all
\(\lambda\geq0\), so the forward action is total.  Oddness is immediate.
For \(a\neq0\),
\[
 \frac{d}{d\lambda}\frac{a(\lambda)^2}{2}
 =-\gamma a(\lambda)^{2m+2}<0,
\]
and (1) gives convergence to zero.  Backward continuation reaches a
denominator zero at
\(-1/(2m\gamma a^{2m})\).  The stationary equation has only \(a=0\).
All conclusions hold for every \(m\), so none selects \(m=1\).
\end{proof}

\begin{proposition}[Relabeling law and coefficient nonidentifiability]
\label{prop:relabel}
Let \(b=d\,a\) and \(\lambda'=c\,\lambda\), with \(c,d>0\).  Then the
odd-power generator becomes
\[
 \frac{db}{d\lambda'}
 =-\gamma' b^{2m+1},
 \qquad
 \gamma'=\frac{\gamma}{c\,d^{2m}}.
 \tag{2}
\]
Thus neither the numerical coefficient \(\gamma\) nor its separation from
amplitude and monoid-parameter normalization is intrinsic to the unchanged
package.
\end{proposition}

\begin{proof}
Use \(db/d\lambda'=(d/c)\,da/d\lambda\) and \(a=b/d\).
\end{proof}

\begin{proposition}[Resolution covariance requires transforming resolution]
\label{prop:resolution}
For \(d>0\), the P5-T06 odd nearest-lattice quantizer satisfies
\[
   \mathsf N_{d\varepsilon}(d a)=d\,\mathsf N_\varepsilon(a).
   \tag{3}
\]
Hence \(\varepsilon\) is covariant only as amplitude-coordinate data with
\(\varepsilon'=d\varepsilon\).  Holding its numerical value fixed under an
amplitude relabeling changes the coarse code and is not an intrinsic
source-scale prescription.
\end{proposition}

\begin{proof}
Insert \(da\) and \(d\varepsilon\) into
\[
 \mathsf N_\varepsilon(a)
 =\varepsilon\,\operatorname{sgn}(a)
 \left\lfloor |a|/\varepsilon+1/2\right\rfloor .
\]
The ratio and away-from-zero tie rule are unchanged.
\end{proof}
