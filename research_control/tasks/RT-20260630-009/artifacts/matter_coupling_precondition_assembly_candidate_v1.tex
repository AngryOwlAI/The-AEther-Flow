\section{Candidate Source Package}

\begin{definition}[Witness-slot record]
A witness-slot record over a source context \(E\) is a tuple
\[
  w=(b,\lambda_W,U_W,n_W)
\]
where \(b\) is a declared \(\BridgeSlot_n(E)\), \(\lambda_W\) is its
source-side witness label, \(U_W\subset F_E^\sharp\) is a finite/local source
support, and \(n_W\in\mathcal{N}_E\) is a no-target-import certificate
attached to the slot.
\end{definition}

\begin{definition}[Matter-signature record]
A matter-signature record over \(E\) is a tuple
\[
  s=(d,\lambda_S,U_S,n_S)
\]
where \(d\) is a scoped \(\MatSigDisc(E) / \SemMatterSig(E)\) record,
\(\lambda_S\) is its source-side signature label, \(U_S\subset F_E^\sharp\)
is a finite/local source support, and \(n_S\in\mathcal{N}_E\) is a
no-target-import certificate attached to the record.
\end{definition}

\begin{definition}[Compatibility certificate]
A compatibility certificate is a source-side tuple
\[
  c=(w,s,U_c,\chi_c,n_c)
\]
with \(w\in\WitSlot_E\), \(s\in\SigRec_E\), finite/local support
\(U_c\subset F_E^\sharp\), a source-only predicate value
\(\chi_c=\mathsf{match}\), and a no-target-import certificate
\(n_c\in\mathcal{N}_E\). The predicate \(\chi_c\) may inspect only declared
source labels, source supports, and scoped evidence/precondition ledgers. It
may not inspect a target atlas, target metric, Lorentzian signature, proper
time, stress-energy tensor, detector protocol, empirical calibration, matter
action, benchmark fit, or Einstein-equation premise.
\end{definition}

\begin{definition}[Candidate package]

\[

  (E,F_E^\sharp,\WitSlot_E,\SigRec_E,\CompCert_E,\mathcal{R}_E,

It is a refinement of the target domain \(\mathcal{D}^{\mathrm{MCPA}}_E\)

evidence/precondition ledgers only.
\end{definition}

\section{Naturality And Finite Variation}

\begin{definition}[Source relabeling transport]
For a source relabeling \(\rho:E\to E'\), suppose \(\rho\) transports supports
by \(U\mapsto \rho U\), witness slots by \(w\mapsto \rho w\), signature
records by \(s\mapsto \rho s\), and compatibility certificates by
\[
  (w,s,U_c,\mathsf{match},n_c)
  \mapsto
  (\rho w,\rho s,\rho U_c,\mathsf{match},\rho n_c).
\]
Define
\[
  \rho_*(w,s,c,U_a,q_a)
  =
  (\rho w,\rho s,\rho c,\rho U_a,q_a).
\]
\end{definition}

\begin{proposition}[Source-side naturality]
When \(\rho\) transports the candidate package as above and preserves
certificate pass/fail values, then
\[
  \rho_*\bigl(\mathcal{A}^{\mathrm{cand}}_E(\mathcal{D}^{\mathrm{cand}}_E)\bigr)
  =
  \mathcal{A}^{\mathrm{cand}}_{E'}(\rho\mathcal{D}^{\mathrm{cand}}_E).
\]
\end{proposition}

\begin{proof}
Both sides enumerate exactly the transported compatibility certificates with
predicate value \(\mathsf{match}\) and passed no-target-import certificates.
The support union operation commutes with \(\rho\) by definition:
\(\rho(U_W\cup U_S\cup U_c)=\rho U_W\cup\rho U_S\cup\rho U_c\). The quality
tag is unchanged and remains pre-coupling-only.
\end{proof}

\begin{definition}[Finite-variation response]
For each declared finite variation \(v\in\mathcal{V}_E\), the candidate has
three allowed responses only:
\[
  \mathsf{stable},\qquad
  \mathsf{source\text{-}local\text{-}repair},\qquad
  \BottomSrc.
\]
The repair response may alter witness labels, signature labels, supports, or
compatibility certificates only through declared source-local data. If a
variation needs a target metric, stress-energy conservation law, detector
calibration, matter action, benchmark fit, or Einstein-equation premise, the
candidate returns
\(\mathrm{OB\text{-}MCPA\text{-}FINITE\text{-}VARIATION\text{-}FAIL}\).
\end{definition}
