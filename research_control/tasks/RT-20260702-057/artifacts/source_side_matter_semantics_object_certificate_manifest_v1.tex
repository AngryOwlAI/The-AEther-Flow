\section{Object Definitions}

\begin{definition}[Source-side matter-semantics object]
A \(\MSObj\) over a source episode \(E\) is a task-local tuple
\[
  A_{\mathrm{src}} =
  (D_A,\Lambda_A,\rho_A,\Gamma_A,\mathcal C_A,\mathcal F_A)
\]
where \(D_A\) is a declared source-side domain, \(\Lambda_A\) is a
source-label family, \(\rho_A\) is a source readout or response-token family,
\(\Gamma_A\) is a declared source guard family, \(\mathcal C_A\) is an
explicit certificate set, and \(\mathcal F_A\) is a fail-closed branch map.
The tuple is source-side only. Its components are not detector semantics,
stress-energy semantics, a matter action, or a coupling law.
\end{definition}

\begin{definition}[Source-side matter-semantics record]
A \(\MSRec\) is an indexed record
\[
  r=(E,i,\ell_i,\rho_i,\gamma_i,\mathcal C_i)
\]
inside a \(\MSObj\), where \(i\) is a declared source index, \(\ell_i\) is a
source label, \(\rho_i\) is a source readout or response token, \(\gamma_i\)
is a guard, and \(\mathcal C_i\) is the explicit certificate bundle attached
to that record. If the bundle is absent, malformed, target-importing, or
object-mismatched, the record maps to \(\Bottom\) for equivalence purposes.
\end{definition}

\begin{definition}[Source-side matter-semantics bridge]
A \(\MSBridgeSrc(A_{\mathrm{src}},B_{\mathrm{src}})\) is a declared
source-side relation candidate between two \(\MSObj\) instances. It contains
only source-domain, source-label, source-readout, response-token, guard, and
certificate data. It may cite scoped evidence/preconditions such as
\(\MSStable\), \(\SMSAR\), and \(\PositiveMSProfile\) only within their
tracked scoped-evidence status.
\end{definition}

\begin{definition}[Declared \(RR_E\) object]
A \(\RREObj(F)\) is an \(RR_E\) record explicitly indexed by a declared source
object \(F\). Two such records may be identified only when an explicit
source-transport, source-invariance, or source-factorization certificate is
present for the same declared object and passes the no-target-import guard.
Otherwise \(RR_E\) separation is preserved.
\end{definition}

\section{Certificate Definitions}

\begin{definition}[Source transport certificate]
A source transport certificate \(\CTr(A,B)\) is an explicit source-side
witness that transports the declared source labels, guards, and record indices
of \(A_{\mathrm{src}}\) into those of \(B_{\mathrm{src}}\). It may not use
target topology, target atlas, target metric, detector semantics, benchmark
success, registry status, validator status, role identity, handoff status, or
commit state as a premise.
\end{definition}

\begin{definition}[Source invariance certificate]
A source invariance certificate \(\CInv(A,B)\) is an explicit source-side
witness that the relevant labels, guards, and response-token relations remain
invariant under a declared source relabeling or admissible source variation.
It is not an empirical invariance claim and not a target metric claim.
\end{definition}

\begin{definition}[Source factorization certificate]
A source factorization certificate \(\CFact(A,B;F)\) is an explicit
source-side witness that the proposed equivalence factors through a declared
source object \(F\) and preserves the object-indexed \(\RREObj(F)\) boundary.
If the factorization object changes, is absent, or imports target structure,
the certificate is malformed.
\end{definition}

\begin{definition}[No-target-import certificate]
A no-target-import certificate \(\CNTI\) is a negative guard showing that the
candidate construction does not use target topology, target smooth atlas,
target Lorentzian metric, proper time, empirical detector semantics,
benchmark behavior, generated derivatives, registry metadata, validators,
roles, handoffs, approvals, local caches, file order, or commit state as
mathematical premises. It supplies hygiene only; it does not supply positive
matter semantics.
\end{definition}

\begin{definition}[Certificate-indexed source equivalence]
For declared source objects \(A_{\mathrm{src}}\) and \(B_{\mathrm{src}}\),
\[
  A_{\mathrm{src}} \EqMS B_{\mathrm{src}}
\]
means that at least one of \(\CTr(A,B)\), \(\CInv(A,B)\), or
\(\CFact(A,B;F)\) exists, \(\CNTI\) passes, declared-object indexing is
consistent, and all fail-closed branches remain inactive. The relation is
source-side and certificate-indexed. It is not detector equivalence, physical
matter equivalence, stress-energy equivalence, matter-action equivalence,
coupling-law equivalence, empirical equivalence, or benchmark equivalence.
\end{definition}
