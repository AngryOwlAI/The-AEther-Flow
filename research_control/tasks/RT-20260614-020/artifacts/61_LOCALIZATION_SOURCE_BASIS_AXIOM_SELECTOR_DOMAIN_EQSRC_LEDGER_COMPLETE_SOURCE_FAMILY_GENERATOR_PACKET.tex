\section{Ledger-Complete Source Objects}

\begin{definition}[Ledger atom]
A ledger atom is a tuple
\[
    a=(b,\lambda,\alpha,p,f,k,\iota,q)
\]
where \(b\) is a finite source branch token, \(\lambda\) is a source
label, \(\alpha\in\Ann\) is an annotation token, \(p\in\Proxy\) is a
proxy guard, \(f\in\Forbid\) is a forbidden-channel guard,
\(k\in\Blocker\) is a blocker record, \(\iota\in\Infl\) is an influence
record, and \(q\) is quotient-fiber data. None of these coordinates may
be chosen by target metric success, target proper time, target locality,
observer protocol, benchmark rank, or candidate performance.
\end{definition}

\begin{definition}[Ledger-complete finite source object]
A ledger-complete source object \(X\) is a finite object
\[
    X=(B_X,i_X,o_X,A_X,E_X,C_X)
\]
where \(B_X\) is a finite set of ledger atoms, \(i_X,o_X\) are fixed
source boundary anchors, \(A_X\) is the branch-local annotation
aggregation rule, \(E_X\) is the source influence coverage relation, and
\(C_X\) is the set of admissible contraction certificates. The object is
ledger-complete only when every branch token has annotation, proxy guard,
forbidden-channel guard, blocker, influence, quotient-fiber, and
boundary-anchor provenance.
\end{definition}

\begin{controlrule}[No incomplete objects]
An object with deleted guards, erased blockers, partial influence
coverage, moved boundary anchors, quotient data selected by downstream
behavior, or branch labels interpreted as target semantics is not in
\(\ObjLC\).
\end{controlrule}

\section{Source-Family Generator}

\begin{construction}[Ledger-complete generator]
The generator \(\GenLC\) produces the smallest class \(\ObjLC\) closed
under the following source-side operations:
\begin{enumerate}
    \item seed inclusion of every previously registered finite source
    object whose full ledger is available;
    \item full source-ledger transport, where a branch-label
    permutation is allowed only if annotation, guards, blockers,
    influence, quotient, contraction, and boundary data are transported
    with it;
    \item branch refinement, where a branch may split into finitely many
    source branches only if annotation aggregation is conserved and
    proxy, forbidden-channel, blocker, influence, quotient, and boundary
    data pull back along an explicit source certificate;
    \item quotient refinement, where quotient fibers may be subdivided
    only by a source certificate that preserves boundary anchors and
    does not select representatives by downstream behavior; and
    \item certified contraction transport, where admissible contraction
    certificates compose and preserve the full source ledger.
\end{enumerate}
\end{construction}

\section{Certificate Relation}

\begin{definition}[Ledger certificate]
A certificate \(c:X\to Y\) in \(\CertLC\) is a finite sequence of
source-side generator steps from \(X\) to \(Y\) such that every step
preserves:
\[
    i,o,\quad \Ann,\quad \Proxy,\quad \Forbid,\quad
    \Blocker,\quad \Infl,\quad q,\quad C.
\]
The inverse step must also be a valid source-side generator step. A
certificate is invalid when it deletes ledger data, changes boundary
anchors, weakens guards, selects a quotient representative by downstream
behavior, or imports a target primitive.
\end{definition}

\begin{definition}[Ledger-complete certificate relation]
For \(X,Y\in\ObjLC\), define
\[
    X \EqLC Y
    \quad\Longleftrightarrow\quad
    \text{there exists } c:X\to Y \text{ in } \CertLC .
\]
\end{definition}

\begin{proposition}[Relation properties on the declared domain]
The relation \(\EqLC\) is reflexive, symmetric, and transitive on the
declared generated domain \(\ObjLC\).
\end{proposition}

\begin{proof}
Reflexivity follows from the empty certificate \(1_X:X\to X\). Symmetry
follows because \(\CertLC\) admits only reversible source-side generator
steps; reversing a valid certificate gives a valid certificate
\(c^{-1}:Y\to X\). Transitivity follows by concatenating two valid
certificates \(c:X\to Y\) and \(d:Y\to Z\), since each step preserves
the required ledger coordinates and admissible contraction data.
\end{proof}
