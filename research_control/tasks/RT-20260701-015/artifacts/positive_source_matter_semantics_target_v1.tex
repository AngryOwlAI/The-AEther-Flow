\section{Source-Side Target Domain}

\begin{definition}[Source-side matter-semantics record]
Let \(\RecSrc(E)\) be the set of finite/local source records
\[
  r=(E,\Bcur,\sigma,\pi,\kappa,\RRE,\beta,\alpha,\mathsf{cert},
      \mathsf{fail})
\]
where \(\sigma\) is source support syntax, \(\pi\) is a source-side probe or
readout-token family, \(\kappa\) is source-side certificate data, \(\RRE\) is
the retained readout-response record, \(\beta\) records scoped
\(\MSBridge(E;\Bcur)\) evidence/precondition status when available, \(\alpha\)
records scoped \(\SMSAR\) adoption-readiness evidence/precondition status
when available, \(\mathsf{cert}\) contains no-target-import certificates, and
\(\mathsf{fail}\) names controlled bottom or obstruction branches.
\end{definition}

\begin{definition}[Admissible source records]
\(\AdmSrc(E)\subseteq\RecSrc(E)\) is the subdomain of records satisfying:
\begin{enumerate}[label=A\arabic*.,leftmargin=*]
  \item every component of \(r\) is source-side syntax, source-side
  certificate data, or tracked scoped evidence/precondition status;
  \item \(\mathsf{cert}\) explicitly excludes target geometry, target metric,
  detector protocol, empirical calibration, stress-energy semantics,
  stress-energy tensor, matter action, conservation by imported theory,
  \(\MetricData(E)\), \(\geff\) expansion, Einstein equations, benchmark fit,
  generated-derivative authority, registry authority, validator authority, and
  commit authority;
  \item distinct \(\RRE\) records are not identified to manufacture a label;
  \item source relabelings and finite variations either transport the record
  by source-defined rules or produce \(\Obstruction\); and
  \item bridge evidence \(\beta\) and readiness evidence \(\alpha\) remain
  evidence/precondition inputs only.
\end{enumerate}
\end{definition}

\section{Source-Side Semantic Labels and Structures}

\begin{definition}[Target label set]
The proposal-only source-side target label set is
\[
  \LabSrc=\{\Bottom,\Pattern,\Separated,\Obstruction\}.
\]
\(\Bottom\) means insufficient source data. \(\Pattern\) means a future
candidate has supplied a source-side pattern assignment satisfying all target
obligations. \(\Separated\) means source data witness a target-relevant
distinction. \(\Obstruction\) means a required clause fails or a forbidden
import is detected.
\end{definition}

\begin{definition}[Proposal-only label map]
\(\PMSTarget\) asks a later constructor to supply a source-defined partial map
\[
  \lambdaSrc:\AdmSrc(E)\dashrightarrow \LabSrc
\]
whose values are determined only by source components of admissible records.
In this P4-T01 packet, \(\lambdaSrc\) is a target slot, not an adopted
matter-semantics function.
\end{definition}

\section{Equivalence and Separation Conditions}

\begin{definition}[Source-side target equivalence]
For \(r,r'\in\AdmSrc(E)\), the relation \(r\EqMS r'\) is admissible only if a
future candidate proves all of the following by source-side data:
\begin{enumerate}[label=E\arabic*.,leftmargin=*]
  \item \(\lambdaSrc(r)\) and \(\lambdaSrc(r')\) are both defined without
  target imports;
  \item the source supports, certificates, and \(\RRE\)-separation data are
  compatible under permitted source relabelings and finite variations;
  \item no detector protocol, empirical calibration, target metric,
  stress-energy object, matter action, conservation law import, Einstein
  equation, or benchmark fit is used; and
  \item the equivalence proof survives deletion of process-authority fields.
\end{enumerate}
If any item fails, equivalence is not established and the branch is
\(\Obstruction\) or \(\Bottom\).
\end{definition}

\begin{definition}[Source-side target separation]
For \(r,r'\in\AdmSrc(E)\), the relation \(r\SepMS r'\) is admissible when
source-side certificate data prove at least one of:
\begin{enumerate}[label=S\arabic*.,leftmargin=*]
  \item different source labels are forced by source data;
  \item \(\RRE\)-separation blocks identification;
  \item a permitted source relabeling or finite variation preserves one record
  while sending the other to \(\Bottom\) or \(\Obstruction\);
  \item one record depends on a forbidden target import or process authority.
\end{enumerate}
\(\SepMS\) is a target separation condition, not detector distinguishability
and not empirical calibration.
\end{definition}

\section{Fail-Closed Branches}

\begin{failure}[Missing source record]
If \(r\notin\RecSrc(E)\) or a required source component is absent, the target
returns \(\Bottom\).
\end{failure}

\begin{failure}[Forbidden import]
If \(\lambdaSrc\), \(\EqMS\), or \(\SepMS\) depends on detector protocols,
empirical calibration, stress-energy tensor, stress-energy semantics, matter
action, imported conservation law, target metric, Lorentzian signature, proper
time, Einstein equations, benchmark fit, generated derivatives, registry
metadata, validator PASS, approval state, role identity, or commit state, the
target returns \(\Obstruction\).
\end{failure}

\begin{failure}[\(\RRE\) collapse]
If two records with distinct \(\RRE\) data are identified to force a positive
label, the target returns \(\Obstruction\).
\end{failure}

\begin{failure}[Label nonuniqueness]
If source-side data underdetermine the label and no source-side separation or
equivalence proof resolves the ambiguity, the target returns
\(\Obstruction\) rather than importing target interpretation.
\end{failure}

\begin{failure}[Evidence-as-adoption]
If scoped \(\MSBridge\) or \(\SMSAR\) evidence is read as adopted matter
semantics, detector semantics, coupling law, matter coupling, source law,
stress-energy semantics, matter action, Einstein equations, benchmark
promotion, or completed derivation, the target returns \(\Obstruction\) for
that use.
\end{failure}

\begin{proposition}[Source-purity of admissible target candidates]
Any future candidate satisfying all clauses of \(\PMSTarget\) is source-pure
relative to the forbidden-import list in this artifact.
\end{proposition}

\begin{proof}
By unfolding the definition of \(\AdmSrc(E)\), every admissible record field is
source-side syntax, source-side certificate data, or scoped evidence/precondition
status. By the definitions of \(\lambdaSrc\), \(\EqMS\), and \(\SepMS\), a
candidate may assign labels, equivalences, or separations only from those
admissible fields. Each forbidden target import or process authority triggers
an explicit \(\Obstruction\) branch. Therefore any candidate satisfying all
clauses is source-pure relative to this target. This proof is conditional on
the future candidate actually satisfying the clauses; the present packet does
not construct or adopt that candidate.
\end{proof}
