\section{Definitions}

\begin{definition}[Source matter-semantics input scope]
A \(\SMScope(E)\) is a finite or locally finite source-side record
\[
  \mathcal I_E =
  (E,A_{\mathrm{src}},B_{\mathrm{src}},\mathcal R_E,
  \mathcal C_E,\mathcal G_E,\mathcal S_E)
\]
where \(A_{\mathrm{src}}\) and \(B_{\mathrm{src}}\) are declared
\(\MSObj\) instances, \(\mathcal R_E\) is a declared source readout or
response-token family, \(\mathcal C_E\) is an explicit source certificate
family, \(\mathcal G_E\) is a source guard family, and \(\mathcal S_E\) is a
scoped-evidence ledger. The ledger may cite \(\NarrowMSCertEq\) only as
scoped evidence/precondition. It is not a certificate by itself and not
matter-semantics adoption.
\end{definition}

\begin{definition}[Source coupling-law certificate bundle]
A \(\SCLBundle(E)\) over \(\SMScope(E)\) is a record
\[
  \mathcal B_E =
  (C^{MS}_E,C^{RR}_E,C^{NT}_E,D^{rep}_E,G^{proc}_E,F_E)
\]
with the following fields.
\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2em]
  \item \(C^{MS}_E\) is an explicit source transport, source invariance, or
  source factorization certificate for the matter-semantics comparison.
  \item \(C^{RR}_E\) records the declared \(\RREObj\) separation or exact
  certificate-indexed transport/invariance boundary.
  \item \(C^{NT}_E\) is a no-target-import certificate.
  \item \(D^{rep}_E\) is a \(\DetPlaceholder\), defined below.
  \item \(G^{proc}_E\) states that registry rows, validators, roles, handoffs,
  approvals, file order, local caches, and commits are not mathematical
  premises.
  \item \(F_E\) maps invalid branches to \(\FailBranch\).
\end{enumerate}
If any required field is missing, malformed, target-importing, detector-
semantic, or process-authority dependent, the bundle evaluates to \(\Bottom\).
\end{definition}

\begin{definition}[Detector-semantics replacement placeholder]
\(\DetPlaceholder(E)\) is a declared source-side placeholder for whatever a
future route claims can replace detector semantics inside a coupling-law
candidate. It has the form
\[
  D^{rep}_E=(\mathsf{name},\mathsf{source\_domain},
  \mathsf{readout\_interface},\mathsf{certificate},\mathsf{status}).
\]
The placeholder is valid only when all fields are source-side and its status
is either \(\mathsf{supplied\_source\_placeholder}\) or
\(\mathsf{missing\_and\_blocked}\). It is not detector semantics and not
detector-semantics adoption. If the placeholder is missing or replaced by
empirical detector protocol, the candidate validity branch fails closed.
\end{definition}

\begin{definition}[No-target coupling import guard]
\(\NTGuard(E)\) is the guard that forbids target topology, target smooth
atlas, target Lorentzian metric, proper time, detector protocol, empirical
calibration, stress-energy semantics, stress-energy tensor, matter action,
benchmark behavior, Einstein-equation premise, generated derivative,
registry metadata, validator status, role identity, handoff status, approval
status, local cache state, file order, or commit state as a mathematical
premise in this target.
\end{definition}

\begin{definition}[Source-side coupling-law target]

where \(\mathcal I_E\) is a source input scope, \(\mathcal B_E\) is a source

\section{Candidate Target Specification}

\begin{definition}[Coupling-law candidate validity predicate]
For a future candidate relation or partial map \(K_E\), define
\[
  \Validity_E(K_E,\SCLTarget(E))\in
  \{\mathsf{eligible\_for\_audit\_stress},\Bottom\}.
\]
The value is \(\mathsf{eligible\_for\_audit\_stress}\) only if all of the
following hold:
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2em]
  \item \(\SMScope(E)\) is declared and finite or locally finite.
  \item \(\SCLBundle(E)\) contains explicit valid source certificates.
  \item \(\DetPlaceholder(E)\) is supplied as source-side placeholder or
  explicitly marked missing-and-blocked.
  \item \(\NTGuard(E)\) passes.
  \item \(K_E\) names a source-side relation or partial map and all domains,
  codomains, witness payloads, and guards are source-side.
  \item \(\WitnessOb_E\) is satisfied by a finite/local witness declaration
  or by one precise obstruction explaining why no such witness exists under
  the current route.
\end{enumerate}
Any failure returns \(\Bottom\) with exactly one primary fail-closed branch.
The positive value means only eligibility for audit, stress, candidate
construction repair, selector routing, or separately authorized protected
review. It does not adopt a coupling law.
\end{definition}
