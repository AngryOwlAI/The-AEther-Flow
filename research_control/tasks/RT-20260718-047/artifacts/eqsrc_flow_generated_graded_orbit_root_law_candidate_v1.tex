\section{Ordered Action Data and Dependency Order}

\begin{definition}[Proposal ordered-action component]
A proposal record is
\[
 U=(C,\Lambda_+,\oplus,0,\Phi,\mathcal M_U,\mathcal V_U),
\]
where \((\Lambda_+,\oplus,0)\) is a source-order monoid and
\(\Phi:\Lambda_+\times C\to C\) obeys
\[
 \Phi_0=\operatorname{id}_C,
 \qquad
 \Phi_{\sigma\oplus\tau}=\Phi_\sigma\circ\Phi_\tau.
\]
The set \(C\) is one full source component under the undirected closure of
forward action steps. It is not an observer window or a target-selected
subset. The monoid, action, component, source morphisms \(\mathcal M_U\), and
variations \(\mathcal V_U\) are declared before any partition, line, or
relation is evaluated. They contain no target atlas, target metric, proper
time, detector semantics, benchmark success, generated derivative, registry,
validator, role, handoff, file-order, or task-process premise.
\end{definition}

Define action reachability by
\[
 x\preceq_\Phi y
 \quad\Longleftrightarrow\quad
 \text{there exists }\tau\in\Lambda_+\text{ with }y=\Phi_\tau(x).
\]
No transition graph, event projection, or root certificate is independent
candidate data. When \(\preceq_\Phi\) is a locally finite partial order, its
cover relation is
\[
 x\lessdot_\Phi y
 \quad\Longleftrightarrow\quad
 x\prec_\Phi y\text{ and no }z\text{ satisfies }x\prec_\Phi z\prec_\Phi y.
\]

\begin{definition}[Admission]
The proposal record \(U\) is admitted only when:
\begin{enumerate}[label=(A\arabic*)]
  \item the monoid and action identity/composition laws are source defined;
  \item action reachability is antisymmetric on the full component \(C\);
  \item every interval \([x,y]_\Phi\) is finite, so the cover relation is
  source derived rather than loaded independently;
  \item \((C,\preceq_\Phi)\) has exactly one global minimum \(r_U\);
  \item every \(x\in C\) is joined to \(r_U\) by a finite saturated cover
  chain and all such chains have one common length \(\rho_U(x)\);
  \item the even and odd rank blocks defined below are both nonempty;
  \item source morphisms and ordered-monoid reparameterizations are declared
  independently and preserve the action data; and
  \item variations are declared independently and either preserve all prior
  admission data or return a precise bottom tag.
\end{enumerate}
Failure of any item returns a tagged element of \(\Bottom\) rather than a
candidate relation.
\end{definition}

\section{Conditional Theorems}

\begin{theorem}[Reachability preorder and periodicity obstruction]
The action laws make \(\preceq_\Phi\) reflexive and transitive. If distinct
\(x,y\in C\) satisfy both \(x\preceq_\Phi y\) and
\(y\preceq_\Phi x\), admission fails. In particular, every nontrivial
periodic action orbit fails antisymmetry.
\end{theorem}
\begin{proof}
Identity gives reflexivity and action composition gives transitivity.
Bidirectional reachability of distinct points contradicts antisymmetry.
A nontrivial periodic orbit contains such a pair.
\end{proof}

\begin{theorem}[Locally finite cover generation]
If \(\preceq_\Phi\) is a locally finite partial order, then every strict
comparison \(x\prec_\Phi y\) contains a finite saturated chain of cover
relations from \(x\) to \(y\). Hence the transition graph is determined by
action reachability and is not an independent input.
\end{theorem}
\begin{proof}
The interval \([x,y]_\Phi\) is finite. A maximal chain in this finite
interval exists, and consecutive elements in a maximal chain are covers.
\end{proof}

\begin{theorem}[Global-minimum naturality and truncation guard]
An admitted action isomorphism sends the unique global minimum of \(U\) to
the unique global minimum of \(U'\). A point that is minimal only after
restricting a rootless full component to an observer window is not a global
minimum and returns \texttt{observer\_truncation\_minimum}.
\end{theorem}
\begin{proof}
An action isomorphism preserves and reflects reachability, hence preserves the
property of being below every point. Uniqueness identifies the image. A
window-minimal point with a predecessor outside the window is not minimum in
the full component.
\end{proof}

\begin{theorem}[Graded-rank well-definedness]
Under admission item (A5), \(\rho_U(r_U)=0\), every cover raises rank by one,
and \(\rho_U(x)\) is independent of the chosen saturated chain. If one point
admits saturated chains of unequal finite lengths, the candidate returns
\texttt{non\_graded\_or\_infinite\_rank}.
\end{theorem}
\begin{proof}
The first assertions restate the common-length admission property. Unequal
lengths make the proposed rank multivalued and therefore block the partition.
\end{proof}

\begin{theorem}[Action naturality and relation functoriality]
Let \((f,\alpha):U\to U'\) be an admitted source isomorphism, where
\(\alpha:\Lambda_+\to\Lambda'_+\) is an ordered-monoid isomorphism and
\[
 f(\Phi_\tau x)=\Phi'_{\alpha(\tau)}f(x).
\]
Then \(f\) preserves reachability, covers, the unique minimum, rank, parity
blocks, \(B_U\), and \(R_U\). Induced maps obey identity and composition.
\end{theorem}
\begin{proof}
Equivariance and bijectivity preserve and reflect reachability. Covers and the
unique minimum are order-theoretic, while saturated-chain lengths are carried
edge by edge. The indicator generators and kernel pair therefore transport.
\end{proof}

\begin{theorem}[Ordered-monoid reparameterization invariance]
If \(\alpha\) is an ordered-monoid automorphism and the relabeled action is
\(\Phi^\alpha_\tau=\Phi_{\alpha(\tau)}\), then reachability and every derived
candidate object are unchanged.
\end{theorem}
\begin{proof}
Surjectivity of \(\alpha\) gives the same set of action maps, so the
reachability relation is identical. All later objects are derived from it.
\end{proof}

\begin{theorem}[Conditional finite-variation robustness]
An independently declared variation preserves the candidate only when it
preserves the ordered action up to admitted isomorphism and preserves every
admission guard. Otherwise it returns the corresponding bottom tag. This is
conditional source-category robustness, not physical stability.
\end{theorem}

  \item \textbf{Bi-infinite.} On \(\mathbb Z\), let
