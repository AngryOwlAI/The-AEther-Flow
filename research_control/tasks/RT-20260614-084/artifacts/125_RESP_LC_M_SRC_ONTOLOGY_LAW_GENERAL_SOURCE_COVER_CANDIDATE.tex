\section{Source-Cover Package}

\begin{definition}[Source-cover package]
A source-cover package is a tuple
\[
  \mathcal U =
  \left(I,\{U_i\}_{i\in I},\{\Ovsrc_{ij}\}_{i,j\in I},
  \{\Pi_i\}_{i\in I},\{\Cand_{ij}\}_{i,j\in I}\right)
\]
where:
\begin{itemize}
  \item \(I\) is a finite or locally finite source index set;
  \item \(U_i\) is a source carrier, not a target chart;
  \item \(\Ovsrc_{ij}\subseteq U_i\times U_j\) is a source overlap relation;
  \item \(\Pi_i:U_i\to P_i^{\mathrm{src}}\) is a source-token profile map;
  \item \(\Cand_{ij}\) is a source-defined family of partial transition
  tokens from \(U_i\) to \(U_j\).
\end{itemize}
No target topology, target smooth atlas, metric scale, proper time, empirical
observer protocol, detector semantics, benchmark behavior, validation status,
registry row, generated derivative, role label, or handoff status is an input.
\end{definition}

\section{Selector And Transition Algebra}

\begin{definition}[Source defect vector]
For \(c\in\Cand_{ij}\), define the source defect vector
\[
  d_{ij}(c)=
  \bigl(p(c),o(c),r(c),q(c),v(c)\bigr),
\]
where the entries count or flag, using source data only:
\begin{itemize}
  \item profile mismatch across \(\Ovsrc_{ij}\);
  \item overlap-support loss;
  \item missing reverse transition token;
  \item declared triple-overlap composition mismatch;
  \item failure under the admissible source-variation class.
\end{itemize}
The preorder \(c\preceq_{ij}^{\mathrm{src}}c'\) means
\(d_{ij}(c)\) is lexicographically no larger than \(d_{ij}(c')\).
\end{definition}

\begin{definition}[Fail-closed selector]
The source selector is
\[
  \Sel_{ij}(\Cand_{ij}) =
  \begin{cases}
    c, & \text{if }c\text{ is the unique }\preceq_{ij}^{\mathrm{src}}
    \text{-least candidate with }d_{ij}(c)=(0,0,0,0,0),\\
    \Bot, & \text{otherwise.}
  \end{cases}
\]
When non-bottom, write \(\Tau_{ij}=\Sel_{ij}(\Cand_{ij})\).
\end{definition}

\begin{guard}[Inverse and cocycle guards]
The selected family passes the inverse guard only when
\[
  \mathsf{inv}_{\mathrm{src}}(\Tau_{ij})=\Tau_{ji}.
\]
It passes the cocycle guard on a declared triple overlap only when
\[
  \mathsf{comp}_{\mathrm{src}}(\Tau_{jk},\Tau_{ij})=\Tau_{ik}.
\]
If either equality is not source-defined or fails, the output is \(\Bot\).
\end{guard}

\section{General Source-Cover Candidate}

\begin{construction}[The candidate \(\GSC\)]
For every source-cover package \(\mathcal U\), define
\[
  \GSC(\mathcal U)=
  \begin{cases}
    (\Qsrc,\{\Pi_i\},\{\Tau_{ij}\},\Vargsc,\Stabgsc),
    & \text{if all declared selector inverse and cocycle guards pass},\\
    \Bot, & \text{otherwise.}
  \end{cases}
\]
Here \(\Qsrc\) is the quotient support generated by the selected source
transition tokens \(\Tau_{ij}\) on declared overlaps. The quotient support is
conditional: it exists only for the selected source-token equivalence relation.
It is not a target manifold and not adopted \(\Msrc\).
\end{construction}

\begin{definition}[Admissible variation class]
\(\Vargsc\) is the class of source-token relabelings and finite local
refinements that preserve:
\begin{itemize}
  \item source overlap truth values;
  \item source profile compatibility classes;
  \item uniqueness of the zero-defect selector minimum;
  \item inverse-token availability;
  \item triple-overlap composition equalities;
  \item the generated quotient-support relation up to source-token renaming.
\end{itemize}
\end{definition}

\begin{lemma}[Stability under admissible source variation]
If \(v\in\Vargsc\) and \(\GSC(\mathcal U)\neq\Bot\), then
\[
  \GSC(v\mathcal U)\neq\Bot
\]
and \(v\) transports the selected transitions and quotient support by
source-token relabeling.
\end{lemma}

\begin{proof}
Each defining condition for \(\Vargsc\) preserves the zero-defect selector
minimum, inverse equality, cocycle equality, and generated source-token
quotient relation. Therefore the same selected transition pattern is
available after applying \(v\), up to source-token renaming.
\end{proof}

\section{Fail-Closed Branches}

\begin{guard}[Profile collapse]
If source profiles do not separate the required overlap classes, then the
zero-defect selector minimum is absent or nonunique. The output is \(\Bot\).
\end{guard}

\begin{guard}[Selector nonuniqueness]
If two distinct transition tokens are tied as source-minimal candidates, the
selector cannot use file order, target resemblance, benchmark behavior,
validation status, role identity, registry visibility, generated derivative
visibility, or handoff status to break the tie. The output is \(\Bot\).
\end{guard}

\begin{guard}[Inverse or cocycle defect]
If a selected transition lacks a reverse source token or if triple-overlap
composition fails, the output is \(\Bot\), not a target-atlas repair.
\end{guard}

\begin{guard}[Variation outside \(\Vargsc\)]
If a perturbation changes source profiles, creates selector ties, deletes
inverse tokens, changes cocycle equalities, or otherwise leaves \(\Vargsc\),
this construction proves no stability claim. The branch fails closed.
\end{guard}
