\begin{theorem}[Explicit-equipment naturality and update intertwining]
\label{thm:intertwining}
% theorem_id: P7T05-THM-THREE-ADDRESS-COUPLING-NATURALITY-001
\textup{\texttt{P7T05-THM-THREE-ADDRESS-}\allowbreak
\texttt{COUPLING-NATURALITY-001}.}

\(\Kmap_{\widehat{\Control}}\) is invertible and
\begin{equation}
 \Kmap_{\widehat{\Control}}\Update_{\Control}
 =

\begin{proof}

\[
 \bigl(\Kmap_{\widehat{\Control}}^{-1}\nu\bigr)(x)

Equation~\eqref{eq:p7t02-update}.  Row-bijective transition transport

\[

the same finite change of variables with the displayed compatibility
square.  Associativity of ordinary map composition proves the final claim.
