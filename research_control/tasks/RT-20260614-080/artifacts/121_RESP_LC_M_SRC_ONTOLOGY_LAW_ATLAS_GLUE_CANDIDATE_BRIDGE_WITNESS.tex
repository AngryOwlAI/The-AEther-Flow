\section{Finite/Local Source-Law Input}

\begin{definition}[Minimal source package \(\Astar\)]
Let the source patch index set be \(I=\{0,1,2\}\). Let
\[
  U_i=\{a_i,b_i\}
  \qquad (i\in I).
\]
The response-labelled source germs \(\Germ_i\) are treated as explicit
source labels attached to \(U_i\). They are not target charts.

Define source profile maps by
\[
\begin{aligned}
\Prof_i(a_i)&=(A,L,i),\\
\Prof_i(b_i)&=(B,R,i).
\end{aligned}
\]
The symbols \(A,B,L,R,i\) are declared source tokens. They are not target
coordinates, detector values, proper-time readings, metric data, or
benchmark labels.
\end{definition}

\begin{definition}[Source overlap and profile compatibility]
For \(i\neq j\), define the source overlap predicate by class compatibility:
\[
  \Ovsrc_{ij}(x_i,y_j)
  \quad\Longleftrightarrow\quad
  \text{the first two source-token components of }\Prof_i(x_i)
  \text{ and }\Prof_j(y_j)\text{ agree}.
\]
Thus \(a_i\) overlaps \(a_j\), \(b_i\) overlaps \(b_j\), and cross-class
pairs fail.
\end{definition}

\begin{definition}[Candidate transitions]
Let \(\Cand_{ij}^{\star}\) be the finite set of source partial bijections
\(c:U_i\to U_j\) whose graph is contained in \(\Ovsrc_{ij}\). Define the
intended transition
\[
  \tau_{ij}^{\star}(a_i)=a_j,\qquad
  \tau_{ij}^{\star}(b_i)=b_j.
\]
Let \(\tau_{ii}^{\star}\) be the identity on \(U_i\), and let
\(\tau_{ji}^{\star}=(\tau_{ij}^{\star})^{-1}\).
\end{definition}

\begin{definition}[Selector preorder]
For \(c\in\Cand_{ij}^{\star}\), define a source score
\[
  s_{ij}(c)=
  (d_{\mathrm{class}}(c),d_{\mathrm{orient}}(c),d_{\mathrm{inv}}(c),
  d_{\mathrm{cocyc}}(c)),
\]
ordered lexicographically. The first two entries count class-token and
orientation-token mismatches; the third counts failure to match the declared
reverse source token; the fourth counts triple-overlap composition mismatch.
The selector returns the unique source-minimal candidate and returns
\(\Bot\) if no unique minimum exists.
\end{definition}

\section{Construction}

\begin{construction}[Non-bottom selected transitions]
For every ordered pair \((i,j)\), the intended transition
\(\tau_{ij}^{\star}\) has score \((0,0,0,0)\). Any candidate that crosses
the \(A/B\) class split or reverses the \(L/R\) orientation token has positive
score. Therefore \(\tau_{ij}^{\star}\) is the unique source-minimal
candidate and
\[
  \Sel_{ij}(\Cand_{ij}^{\star})=\tau_{ij}^{\star}\neq\Bot.
\]
\end{construction}

\begin{checkresult}[Inverse]
For every declared ordered pair,
\[
  \mathsf{inv}_{\mathrm{src}}(\tau_{ij}^{\star})=\tau_{ji}^{\star}.
\]
This is a source-token equality inside the finite transition algebra.
\end{checkresult}

\begin{checkresult}[Cocycle]
On the triple \((0,1,2)\),
\[
  \tau_{12}^{\star}(\tau_{01}^{\star}(a_0))=a_2
  =\tau_{02}^{\star}(a_0),
\]
and
\[
  \tau_{12}^{\star}(\tau_{01}^{\star}(b_0))=b_2
  =\tau_{02}^{\star}(b_0).
\]
The reverse and identity cases follow from the declared inverse and identity
source tokens.
\end{checkresult}

\begin{definition}[Bounded finite-variation class]
\(\Varstar\) consists of finite source-token relabelings that:
\begin{enumerate}
  \item preserve the separation of \(A\)-class and \(B\)-class tokens;
  \item preserve the \(L/R\) orientation distinction;
  \item conjugate selected transition tokens by the same source relabeling;
  \item preserve identity, inverse, and triple-overlap composition tables.
\end{enumerate}
The predicate \(\Stabstar(v,\Astar)\) holds exactly when these four clauses
hold. If a perturbation merges \(A\) and \(B\), ties the selector, deletes an
inverse token, or changes a cocycle table, it is outside \(\Varstar\) or
returns \(\Bot\).
\end{definition}

\begin{construction}[Conditional source quotient support]
Form
\[
  D_{\star}=\bigsqcup_{i\in I}(\{i\}\times U_i)
\]
and quotient by the selected transition identifications. The quotient
\(\Qstar\) has two source equivalence classes:
\[
[(0,a_0)]=[(1,a_1)]=[(2,a_2)],
\qquad
[(0,b_0)]=[(1,b_1)]=[(2,b_2)].
\]
This is conditional source quotient support relative to \(\Astar\), not a
target manifold and not full \(\Msrc\).
\end{construction}

\section{Bridge Witness Lemma}

\begin{lemma}[Minimal non-bottom bridge witness]
The finite/local package \(\Astar\) realizes a non-bottom bridge witness for
\(\MSL\). It supplies source profile semantics, unique source-minimal
selector outputs, transition algebra, inverse and cocycle checks, and a
bounded finite-variation stability class. Therefore
\[
  \Astar \longmapsto
  (\Qstar,\{\Prof_i\},\{\Sel_{ij}\},\{\tau_{ij}^{\star}\},\Varstar,\Stabstar)
\]
is a candidate bridge map under explicit source assumptions.
\end{lemma}

\begin{proof}
The profile maps separate the two source classes and preserve orientation
tokens across declared overlaps. The selector score is defined only from
source-token mismatch, inverse-token mismatch, and cocycle-token mismatch.
The intended transitions have zero score and all cross-class or orientation
violating alternatives have positive score, so the selector is non-bottom and
unique. The inverse and cocycle checks are finite source-token equalities.
The finite-variation class is closed under source relabeling by construction:
each allowed variation conjugates the selected transitions and preserves the
identity, inverse, and composition tables. Non-preserving perturbations fail
closed rather than importing target or process authority. The quotient support
therefore follows by the conditional quotient construction in \(\MSL\).
\end{proof}

\section{Failure Branches}

\begin{failure}[Profile collapse]
If a perturbation merges \(A\) and \(B\), profile separation is lost and the
selector returns \(\Bot\).
\end{failure}

\begin{failure}[Selector nonuniqueness]
If two candidate transitions share the same minimal source score, no file
order, target resemblance, benchmark success, validation status, role label,
registry row, or generated derivative may break the tie. The selector returns
\(\Bot\).
\end{failure}

\begin{failure}[Inverse defect]
If a selected transition lacks the declared reverse source token, the witness
returns \(\Bot\) on that overlap.
\end{failure}

\begin{failure}[Cocycle defect]
If a triple-overlap composition differs from the selected direct transition,
the witness returns \(\Bot\).
\end{failure}

\begin{failure}[Variation fragility]
The bounded variation class preserves the positive branch by construction.
Arbitrary source perturbations outside \(\Varstar\) remain unproved and are
not promoted to robustness.
\end{failure}
