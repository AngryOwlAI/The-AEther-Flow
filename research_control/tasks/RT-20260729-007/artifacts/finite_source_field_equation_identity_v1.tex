\section{Exact finite field equation}

\begin{definition}[Finite response operator]

\end{equation}
The operator is independent of \(h\mapsto h+k\), \(k\in\Kspace\).
Because \(L_C(u+k)=L_Cu\), it also factors through the source quotient when
only response-equivalent source configurations are distinguished.
\end{definition}

\begin{theorem}[Exact finite Euler equation and solution class]
\label{thm:euler}
For every admitted variation \([\eta]\in\Qspace\),
\begin{equation}
 D_hS_C^{\alpha,\beta}([h];u)[\eta]
 =\eta^{\mathsf T}\Eop([h];u).
 \label{eq:variation}
\end{equation}
Hence the exact finite Euler equation is
\begin{equation}
 \Eop([h];u)=0,
 \qquad\text{equivalently}\qquad
 \alpha L_Ch=\beta L_Cu.
 \label{eq:euler}
\end{equation}
It has one and only one solution class
\begin{equation}
 [h]=\frac{\beta}{\alpha}[u]\in\Qspace.
 \label{eq:solution}
\end{equation}
The source-to-equation and source-to-solution maps are
\begin{align}
 F_C^{\beta}:\Qspace&\longrightarrow\Aspace,
 &F_C^{\beta}([u])&=\beta L_Cu,\\
 R_C^{\alpha,\beta}:\Qspace&\longrightarrow\Qspace,
 &R_C^{\alpha,\beta}([u])&=\frac{\beta}{\alpha}[u].
 \label{eq:maps}
\end{align}
For \(\beta\ne0\), \(F_C^\beta\) is an isomorphism between the finite
quotient and the balanced response space.  These maps are exact in the
finite candidate and have no target-spacetime codomain.
\end{theorem}

\begin{proof}
Symmetry of \(L_C\) gives Equation~\eqref{eq:variation}.  Both terms in
Equation~\eqref{eq:response} lie in
\(\operatorname{im}L_C=\Kspace^\perp=\Aspace\).  The Euler equation implies
\[
 L_C\!\left(h-\frac{\beta}{\alpha}u\right)=0,
\]
so its solution difference lies in \(\Kspace\), proving existence and
uniqueness in \(\Qspace\).  The restriction of \(L_C\) to the quotient is
injective and has image \(\Aspace\), proving the map statement.
\end{proof}

\section{Complete finite identity structure}

Define the oriented edge flux
\begin{equation}
 J_{xy}(h,u)
 :=C_{xy}\bigl(\alpha(h_x-h_y)-\beta(u_x-u_y)\bigr).
 \label{eq:flux}
\end{equation}
Symmetry of \(C\) gives \(J_{yx}=-J_{xy}\).

\begin{theorem}[Off-shell flux and component identities]
\label{thm:identity}
For every \(x\in\Omega\),
\begin{equation}
 \mathcal E_{C,x}^{\alpha,\beta}([h];u)
 =\sum_{y\in\Omega}J_{xy}(h,u).
 \label{eq:divergence}
\end{equation}
For every subset \(B\subseteq\Omega\),
\begin{equation}
 \sum_{x\in B}\mathcal E_{C,x}^{\alpha,\beta}([h];u)
 =\sum_{\substack{x\in B\\y\notin B}}J_{xy}(h,u).
 \label{eq:cut}
\end{equation}
In particular, for every connected component \(K\in\Pi_C\),
\begin{equation}
 \sum_{x\in K}\mathcal E_{C,x}^{\alpha,\beta}([h];u)=0
 \label{eq:component}
\end{equation}
identically, before imposing the Euler equation.  Equivalently,
\begin{equation}
 k^{\mathsf T}\Eop([h];u)=0
 \quad\text{for every }k\in\Kspace.
 \label{eq:noether}
\end{equation}
Equations~\eqref{eq:component} and \eqref{eq:noether} are the complete
identities generated by the component-shift symmetry
\([h]\mapsto[h+k]\) of this finite action.
\end{theorem}

\begin{proof}
Equation~\eqref{eq:divergence} is the component form of
\(\alpha L_Ch-\beta L_Cu\).  Summing it over \(B\), the antisymmetric
fluxes on internal edges cancel pairwise, leaving only the cut term in
Equation~\eqref{eq:cut}.  A support component has no crossing edges, which
proves Equation~\eqref{eq:component}.  Component indicator vectors span
\(\Kspace\), proving Equation~\eqref{eq:noether}.  Conversely, every
identity in the left kernel of \(D_h\Eop=\alpha L_C\) lies in
\(\ker L_C=\Kspace\), so no additional independent linear identity exists
in the exact finite model.
\end{proof}

\begin{proof}
