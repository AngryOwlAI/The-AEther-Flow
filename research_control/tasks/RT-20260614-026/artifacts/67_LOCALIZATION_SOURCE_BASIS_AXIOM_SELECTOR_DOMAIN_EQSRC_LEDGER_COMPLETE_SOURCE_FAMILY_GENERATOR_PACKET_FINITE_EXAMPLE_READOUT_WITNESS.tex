    y_{\rho(u)}&=(\rho(u),\ell_u,a_u,p_u,f_u,k_u,\eta_u,q_u),\\

\begin{lemma}[Positive certificate]
The transport \(c_{\rho}:X_+\to Y_+\) is a valid \(\CertLC\)
certificate, and \(c_{\rho}^{-1}:Y_+\to X_+\) is valid.
\end{lemma}

\begin{proof}
The transport changes only the branch positions by the source
permutation \(\rho\). It carries the labels, annotations, proxy guards,
forbidden-channel guards, blocker records, influence records, quotient
data, contraction-certificate data, and boundary anchors with the
branches. Therefore every preserved coordinate required by \(\CertLC\)
is preserved. Applying \(\rho^{-1}\) returns the same ledger tuples to
their original branch positions, so the inverse certificate is valid.
The composites \(c_{\rho}^{-1}\circ c_{\rho}\) and
\(c_{\rho}\circ c_{\rho}^{-1}\) are the identity transports on the
source ledger.
\end{proof}

\section{Failed-Certificate Witness}

\begin{failedcase}[Ambiguous refinement inverse]
Let \(Z\) be a refinement attempt of \(X_+\) that replaces the
\(u\)-branch by two branches \(u_0,u_1\), preserves only the aggregate
annotation \(a_u=a_{u_0}\oplus a_{u_1}\), and records the blocker
\(k_u\) only at the aggregate level. The forward step \(r:X_+\to Z\)
looks plausible if one checks only aggregate annotations. It is not a
valid \(\CertLC\) certificate because the inverse cannot recover which
refined branch carries the blocker provenance.
\end{failedcase}

\begin{proof}
\(\CertLC\) requires reversible preservation of blocker records,
quotient data, contraction data, and boundary provenance. The attempted
refinement stores \(k_u\) only as aggregate information. Two refined
objects with different assignments of \(k_u\) to \(u_0\) or \(u_1\)
therefore map to the same aggregate ledger. The inverse is not
well-defined as a source-ledger certificate. Hence the attempted
certificate returns \(\Block\) or \(\Repair\), not \(\Pass\).
\end{proof}

\begin{failedcase}[Noncomposing quotient refinement]
Let \(s_1:X_+\to Z_1\) refine quotient fiber \(q_u\) by a source
certificate using boundary anchors \(i_0,o_0\). Let \(s_2:Z_1\to Y_+\)
use the transported branch labels but silently replace the refined
quotient representative for \(q_u\) by a different representative
\(q_u'\). Each step may look locally source-labeled, but the composite
does not preserve the same quotient data.
\end{failedcase}

\begin{proof}
Composition in \(\CertLC\) is permitted only when the intermediate and
output quotient data agree as source-ledger data. The hidden
replacement \(q_u\mapsto q_u'\) changes a preserved coordinate. The
composite therefore fails for source-side quotient reasons.
\end{proof}

\section{Concrete Source-Pure Readout}

\begin{definition}[Ledger accounting codomain]
Let \(M(S)\) denote the finite multiset-count functions on a finite
source-token set \(S\). Define the readout codomain
\[
    R_{\mathrm{lc}} =
    B_{\partial}\times M(A)\times M(P)\times M(F)\times
    M(K)\times M(I)\times M(Q)\times M(C_{\mathrm{src}}),
\]
where \(B_{\partial}\) is the set of ordered source-boundary-anchor
pairs and \(C_{\mathrm{src}}\) is the finite set of source contraction
certificate labels recorded in the object. Equality on
\(R_{\mathrm{lc}}\) is componentwise exact equality of the boundary
pair and all multiset-count functions.
\end{definition}

\begin{definition}[Source-readout map]
For \(X\in\ObjLC\), define
\[
    \ReadLC(X)=
    \bigl(
    (i_X,o_X),
    \mu_A(X),
    \mu_P(X),
    \mu_F(X),
    \mu_K(X),
    \mu_I(X),
    \mu_Q(X),
    \mu_C(X)
    \bigr),
\]
where each \(\mu\) records the source-ledger multiset of the indicated
coordinate after admissible source pullback. The computation ignores
branch names as names but counts the transported ledger coordinates.
It uses no target metric, target proper time, target locality,
observer protocol, benchmark rank, candidate behavior, validation
status, generated visibility, registry status, or authority label.
\end{definition}

\begin{lemma}[Well-formedness]
\(\ReadLC\) is well formed on \(\ObjLC\).
\end{lemma}

\begin{proof}
By definition, an object is in \(\ObjLC\) only when boundary anchors,
annotations, proxy guards, forbidden-channel guards, blocker records,
influence records, quotient-fiber data, and contraction-certificate
data are present as source-ledger data. Each component of
\(\ReadLC(X)\) is therefore defined from coordinates that every
\(\ObjLC\) object already contains. The codomain and equality relation
are source-ledger constructions.
\end{proof}

\begin{proposition}[Source-side readout invariance]
For all \(X,Y\in\ObjLC\),
\[
    X\EqLC Y \Longrightarrow \ReadLC(X)=\ReadLC(Y)
\]
in \(R_{\mathrm{lc}}\).
\end{proposition}

\begin{proof}
Assume \(X\EqLC Y\). Then there is a certificate \(c:X\to Y\) in
\(\CertLC\). Every step in \(c\) preserves the boundary anchors,
annotation accounting, proxy guards, forbidden-channel guards, blocker
records, influence records, quotient data, and contraction-certificate
data. Branch-label transport may rename branch positions, and
admissible refinement may pull source records back along an explicit
source certificate, but neither operation changes the componentwise
ledger accounting recorded in \(R_{\mathrm{lc}}\). Since componentwise
exact equality is preserved by each certificate step, it is preserved
by finite composition of certificate steps. Therefore
\(\ReadLC(X)=\ReadLC(Y)\).
\end{proof}

\begin{examplecase}[Readout on the positive witness]
For the positive pair,
\[
    \ReadLC(X_+)=\ReadLC(Y_+)
\]
because \(c_{\rho}\) transports exactly the same boundary pair and
ledger-coordinate multisets.
\end{examplecase}

\begin{examplecase}[Readout on the negative witness]
\[
    \ReadLC(X_+)\neq \ReadLC(Y_-)
\]
because the blocker multiset and proxy-guard multiset differ. The
readout therefore detects the source-side defect without importing a
target primitive.
\end{examplecase}
