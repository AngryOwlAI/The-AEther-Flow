\section{Source Domain}

\begin{definition}[P5 follow-up source domain]
Let \(D_E\subseteq\AdmPMS(E)\) be a finite/local source domain. A record
\(r\in D_E\) has source-only components
\[
  r=(S(r),C(r),\Qsrc(r)),
\]
where \(S(r)\) is retained non-\(\RRE\) source support, \(C(r)\) is source
certificate data, and \(\Qsrc(r)\) is retained \(\RRE\) source syntax. None of
these components is a detector protocol, empirical calibration, target metric,
stress-energy datum, matter action, \(\MetricData(E)\), \(\geff\), benchmark
success, validation status, registry authority, role authority, approval
authority, commit status, generated derivative, or local cache.
\end{definition}

\begin{definition}[Admissible RR\_E variation]
\(\VarRR(r,r')\) is a source-side relation saying that \(r,r'\in D_E\) differ
only by retained \(\RRE\) syntax or by a declared source-side \(\RRE\)
transport candidate. It is not empirical indistinguishability and not target
geometric equivalence.
\end{definition}

\begin{definition}[Certificate families]
For a declared source-side object \(F:D_E\to X\), the candidate-law comparison
allows only the following certificate families:
\begin{enumerate}[label=C\arabic*.,leftmargin=*]
  \item \(\TrRR(r,r')\): a source transport certificate between retained
  \(\RRE\) syntax classes;
  \item \(\InvRR(F;r,r')\): a source proof that \(F(r)=F(r')\) for the
  declared object \(F\);
  \item \(\FactRR(F)\): a source proof that \(F\) factors through a source
  quotient \(\pi_{\RRE}:D_E\to \QRR\).
\end{enumerate}
If none of these certificates is available, identification fails closed to
\(\SepRR(r,r')\) or \(\Obstruction\).
\end{definition}

\section{Selected Proposal-Only Schema}

\begin{schema}[Certificate-indexed \(\LawRR\) schema]
For each declared source object \(F:D_E\to X\) and each
\(\VarRR(r,r')\), a proposal-only source-side law schema may identify \(r\)
and \(r'\) for \(F\) only by one of the following branches:
\[
\begin{array}{ll}
\text{transport branch} & \TrRR(r,r')\ \text{and source soundness for }F, \\
\text{invariance branch} & \InvRR(F;r,r'), \\
\text{factorization branch} & \FactRR(F)\ \text{and}\ F=\bar F\circ\pi_{\RRE}, \\
\text{fail-closed branch} & \SepRR(r,r')\ \text{or}\ \Obstruction .
\end{array}
\]
The schema is certificate-indexed: the admissible identification branch must
name its certificate and the declared source object \(F\). There is no global
unrestricted \(\RRE\) irrelevance claim.
\end{schema}

\begin{lemma}[Factor-through branch]
If \(\FactRR(F)\) supplies a source quotient \(\pi_{\RRE}:D_E\to\QRR\) and a
source map \(\bar F:\QRR\to X\) such that \(F=\bar F\circ\pi_{\RRE}\), then
\(\pi_{\RRE}(r)=\pi_{\RRE}(r')\) implies \(F(r)=F(r')\).
\end{lemma}

\begin{proof}
This is the formal factor-through argument:
\[
F(r)=\bar F(\pi_{\RRE}(r))=\bar F(\pi_{\RRE}(r'))=F(r').
\]
The proof is conditional on a source quotient and factorization certificate.
It is not an unrestricted irrelevance theorem.
\end{proof}

\begin{lemma}[Certificate deletion fail-closed rule]
If the only identification route for \(r,r'\) depends on
\(\TrRR(r,r')\), \(\InvRR(F;r,r')\), or \(\FactRR(F)\), then deleting that
certificate deletes the identification route. The result is
\(\SepRR(r,r')\) or \(\Obstruction\), not detector semantics.
\end{lemma}

\begin{proof}
The selected schema admits no certificate-free identification branch. Without
a named source certificate, the two-record P5 witness family from
\texttt{RT-20260701-023} remains available: records with equal non-\(\RRE\)
support and distinct retained \(\RRE\) syntax cannot be identified for
arbitrary \(F\). Thus deletion fails closed.
\end{proof}
