\section{Target Conditions}

\begin{condition}[SCLT-TGT-001 source-evidence discipline]
All successful inputs must be source-side records in
\(\mathcal{D}^{\mathrm{SCLT}}_E\). Scoped \(\MCPA(E)\) evidence may appear
only as precondition data. It may not be converted into source-law adoption,
\(\MCPA(E)\) adoption, source-extension data adoption beyond the exact gate
result, \(\MetricData(E)\), \(\geff\), coupling-law adoption, matter
coupling, stress-energy semantics, detector semantics, matter action,
Einstein equations, or benchmark behavior.
\end{condition}

\begin{condition}[SCLT-TGT-002 no-target-import]
The candidate must include a certificate in \(\mathcal{N}_E\) ruling out
target topology, target atlas, target metric, Lorentzian signature, proper
time, stress-energy tensor, matter action, detector protocol, empirical
calibration, benchmark fit, Einstein-equation premise, generated derivative,
registry metadata, validator status, role authority, handoff status,
approval status, local cache state, file order, and commit status as
mathematical premises.
\end{condition}

\begin{condition}[SCLT-TGT-003 source-locality]
\(\mathcal{K}^{\mathrm{SCLT}}_E\) must depend only on declared finite/local
source records in \(\mathcal{P}^{\mathrm{MCPA}}_E\), \(\mathcal{W}_E\),
\(\Sigma_E\), and \(\mathcal{C}_E\). Dependence on an undeclared source
object or on any target-side structure is a fail-closed obstruction.
\end{condition}

\begin{condition}[SCLT-TGT-004 covariance and naturality]
For each declared source relabeling or source-family morphism
\[
  \rho: \mathcal{D}^{\mathrm{SCLT}}_E \to
  \mathcal{D}^{\mathrm{SCLT}}_{E'},
\]
the candidate must supply a source-side transport
\(\rho_*:\SourceCoupRec_E\to \SourceCoupRec_{E'}\) such that successful
outputs commute with \(\rho\). Failure to define this compatibility is a
source-side noncovariance obstruction, not a license to import target
covariance.
\end{condition}

\begin{condition}[SCLT-TGT-005 finite-variation behavior]
For every declared finite variation \(v\in\mathcal{V}_E\), the candidate must
state whether the target record is stable, repaired by declared source-local
data, or fail-closed. Stability may not be repaired by target-metric
normalization, empirical calibration, stress-energy conservation, detector
semantics, matter action, or benchmark behavior.
\end{condition}

\begin{condition}[SCLT-TGT-006 matter-signature and witness-slot compatibility]
Each successful target record must name the matter-signature record in
\(\Sigma_E\), the witness slot in \(\mathcal{W}_E\), and the compatibility
certificate in \(\mathcal{C}_E\). Missing compatibility fails closed rather
than inferring physical matter content.
\end{condition}

\begin{condition}[SCLT-TGT-007 fail-closed obstruction labeling]
If a target condition cannot be met, the candidate must return exactly one
primary obstruction label from \(\mathcal{O}_E\). The candidate must not
repair the failure by importing a forbidden target object or by treating
support-only project-control evidence as proof authority.
\end{condition}

\section{Eligibility Theorem}

\begin{theorem}[Source-coupling target eligibility without coupling-law adoption]
Suppose a future packet supplies \(\mathcal{D}^{\mathrm{SCLT}}_E\),
\(\SourceCoupRec_E\), a source-local candidate map or relation
\(\mathcal{K}^{\mathrm{SCLT}}_E\), and checks SCLT-TGT-001 through
SCLT-TGT-007. Then that packet becomes eligible only for a bounded Candidate
Constructor result, Smuggling Auditor review, Refuter stress test,
Theoretical Continuation Selector routing, or separately authorized protected
evidence-status review. It does not derive or adopt a coupling law or matter
coupling.
\end{theorem}

\begin{proof}
The hypotheses require only source-domain declarations, a source-side target
record, scoped evidence/precondition separation, no-target-import
certification, source-local dependence, source-family naturality,
finite-variation behavior, matter-signature and witness-slot compatibility,
and fail-closed obstruction labels. None of these hypotheses includes
canonical ontology edit, source-law adoption, \(\MCPA(E)\) adoption,
\(\MetricData(E)\) adoption, \(\geff\) scope expansion, coupling-law
adoption, stress-energy semantics, a stress-energy tensor, detector
semantics, matter action, Einstein equations, benchmark success, or completed
derivation.

Therefore satisfying the target can only make a future candidate suitable for
the next controlled research roles that construct, audit, stress, route, or
separately review evidence status. The conclusion is eligibility for bounded
continuation under blocked adoption, not coupling-law derivation, not
coupling-law adoption, not matter-coupling derivation, and not
matter-coupling adoption.
\end{proof}
