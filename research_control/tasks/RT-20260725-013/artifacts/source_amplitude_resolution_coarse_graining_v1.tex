\begin{definition}[Odd nearest-lattice quantizer]
Fix a source-amplitude resolution \(\varepsilon>0\).  With
\(\operatorname{sgn}(0)=0\), define
\[
 \Neps(a)
 =\varepsilon\,\operatorname{sgn}(a)

 \label{eq:quantizer}
\]

\[

\]

\[

\]
and the source-state reconstruction
\[

The parameter \(\varepsilon\) is a resolution in the already declared

\begin{theorem}[Typing, exact decoding, and uniform reconstruction error]
\label{thm:error}
For every \(\varepsilon>0\), \(\Ceps\) and \(\Reps\) are well typed,
\[
 \Ceps\circ\Reps=\operatorname{id}_{\Oeps},
\]
and
\[
 |\Neps(a)-a|\leq\frac{\varepsilon}{2}.
 \label{eq:amplitude-error}
\]
Using the source-amplitude norm
\(\|q_a\|_{\mathrm{amp}}=|a|\), which equals the \(C^0\) norm because
\(\|e\|_\infty=1\),
\[
 \|\Pi_\varepsilon(q_a)-q_a\|_{\mathrm{amp}}
 \leq\frac{\varepsilon}{2}.
 \label{eq:source-error}
\]
Consequently \(\Pi_\varepsilon\to\operatorname{id}_{\Qamp}\) uniformly as
\(\varepsilon\downarrow0\).
\end{theorem}

\begin{proof}
Nearest-lattice rounding with the declared tie convention gives
\eqref{eq:amplitude-error}.  Every \(c\in\varepsilon\mathbb Z\) is fixed by
\(\Neps\), proving exact decoding.  Equation \eqref{eq:source-error} follows
from \(q_{\Neps(a)}-q_a=(\Neps(a)-a)e\).  The bound is independent of \(a\).
\end{proof}

\section{Source covariance and dynamics}

\begin{theorem}[Declared source symmetry covariance]
\label{thm:covariance}
Let \(R(q_a)=q_{-a}\) be the P5-T04 amplitude reflection and let
\(\rho_\varepsilon(c)=-c\) on \(\Oeps\).  Then
\[
 \Ceps\circ R=\rho_\varepsilon\circ\Ceps,\qquad

\end{theorem}

\begin{proof}

\(\Neps(-a)=-\Neps(a)\).  The two identities follow directly.

\begin{theorem}[Controlled dynamical compatibility]
\label{thm:dynamics}
For all \(\gamma>0\), \(\lambda,\mu\geq0\), and \(a\in\mathbb R\),
\[
 \left|
 \Ceps(\Phi^\gamma_\lambda q_a)
 -\widetilde F^\varepsilon_\lambda(\Ceps q_a)
 \right|\leq\varepsilon,
 \label{eq:raw-dynamics-error}
\]
and
\[
 \left|
 \Ceps(\Phi^\gamma_\lambda q_a)
 -F^\varepsilon_\lambda(\Ceps q_a)
 \right|\leq\frac{3\varepsilon}{2}.
 \label{eq:lattice-dynamics-error}
\]
The lattice evolution has controlled semigroup defect
\[
 \left|
 F^\varepsilon_\lambda(F^\varepsilon_\mu(c))
 -F^\varepsilon_{\lambda+\mu}(c)
 \right|\leq\frac{3\varepsilon}{2}
 \quad(c\in\Oeps).
 \label{eq:semigroup-defect}
\]
\end{theorem}

\begin{proof}
Differentiation gives
\[
 \left|(f^\gamma_\lambda)'(a)\right|
 =(1+2\gamma a^2\lambda)^{-3/2}\leq1,
\]
so every \(f^\gamma_\lambda\) is \(1\)-Lipschitz.  Combine this with
\eqref{eq:amplitude-error}.  For \eqref{eq:raw-dynamics-error}, the output
rounding and propagated input rounding each contribute at most
\(\varepsilon/2\).  Requantizing the raw effective output adds at most one
further half-bin, proving \eqref{eq:lattice-dynamics-error}.  Apply the same
argument to the exact source semigroup law to obtain
\eqref{eq:semigroup-defect}.
\end{proof}

The exact flow remains the authoritative candidate dynamics.  The lattice
evolution is an approximate task-local code, not an adopted replacement law.
