\section{Declared Source Subdomain}

\begin{definition}[Positive-profile admissible subdomain]
Let \(\Adm^{\mathrm{PMS}}_{\mathrm{src}}(E)\subseteq\AdmSrc(E)\) be the set
of admissible records
\[
  r=(E,\Bcur,\sigma,\pi,\kappa,\RRE,\beta,\alpha,\mathsf{cert},
      \mathsf{fail})
\]
that satisfy the following source-side requirements:
\begin{enumerate}[label=D\arabic*.,leftmargin=*]
  \item \(\sigma\) contains a finite/local source support code
  \(S_r\) and a source pattern code \(X_r\).
  \item \(\pi\) contains a source probe-token family \(P_r\), treated only as
  source syntax.
  \item \(\kappa\) contains a source certificate profile \(K_r\) and an
  explicit no-target certificate bundle \(\Cert(r)\).
  \item \(\RRE\) contains a retained readout-response syntax class
  \(Q_r\), treated as source bookkeeping rather than detector readout or
  empirical response.
  \item \(\beta\) and \(\alpha\), when present, are scoped
  \(\MSBridge\) and \(\SMSAR\) evidence/precondition fields only.
  \item \(\mathsf{fail}\) is empty or names only controlled bottom branches
  already allowed by \(\PMSTarget\).
\end{enumerate}
If any required source-side component is absent, the record is outside this
candidate subdomain and maps to \(\Bottom\).
\end{definition}

\begin{condition}[No-target certificate bundle]
For \(r\in\Adm^{\mathrm{PMS}}_{\mathrm{src}}(E)\), \(\Cert(r)\) must
explicitly exclude detector protocols, empirical calibration, target metric,
Lorentzian signature, proper time, stress-energy semantics, stress-energy
tensor, matter action, conservation by imported GR or QFT theory,
\(\MetricData(E)\), \(\geff\) expansion, Einstein equations, benchmark fit,
generated derivative authority, registry authority, validator authority, role
authority, approval state, commit state, and file-order authority.
\end{condition}

\section{Positive Source Profile}

\begin{definition}[Source-positive profile]
For \(r\in\Adm^{\mathrm{PMS}}_{\mathrm{src}}(E)\), define the source-positive
profile
\[
  \Prof(r)=
  \bigl([S_r]_{\Transport},[X_r]_{\Transport},
  [P_r]_{\Transport},[K_r]_{\Transport},[Q_r]_{\Transport},
  \Cert(r)\bigr).
\]
The brackets mean equality or explicitly declared source-side transport under
permitted source relabelings. \(\Prof(r)\) contains only source support,
source pattern, source probe-token, source certificate, retained \(\RRE\)
syntax, and no-target certificate data.
\end{definition}

\begin{definition}[Audit-readiness predicate]
Define
\[
  \Ready(r)=1
\]
when the record has scoped \(\beta\) and \(\alpha\) evidence/precondition
fields sufficient for audit-readiness routing under prior Gate Chair results.
If either field is deleted, \(\Ready(r)=0\). Deleting \(\beta\) or \(\alpha\)
does not delete \(\Prof(r)\); it only blocks audit-readiness and any later
adoption route.
\end{definition}

\begin{definition}[Candidate label map]
Define the partial label map
\[
  \lambdaProf:\AdmSrc(E)\dashrightarrow
  \{\Bottom,\Pattern,\Separated,\Obstruction\}
\]
as follows:
\begin{enumerate}[label=L\arabic*.,leftmargin=*]
  \item If \(r\notin\Adm^{\mathrm{PMS}}_{\mathrm{src}}(E)\), set
  \(\lambdaProf(r)=\Bottom\).
  \item If \(\Cert(r)\) detects a forbidden import, set
  \(\lambdaProf(r)=\Obstruction\).
  \item If \(\Prof(r)\) is defined and \(\Ready(r)=1\), set
  \(\lambdaProf(r)=\Pattern\).
  \item If \(\Prof(r)\) is defined and \(\Ready(r)=0\), set
  \(\lambdaProf(r)=\Pattern\) with audit-readiness blocked.
\end{enumerate}
The label \(\Pattern\) means only that a positive source profile has been
constructed. It does not mean adopted matter semantics.
\end{definition}

\section{Equivalence And Separation}

\begin{definition}[Profile equivalence]
For \(r,r'\in\Adm^{\mathrm{PMS}}_{\mathrm{src}}(E)\), define
\[
  r\EqProf r'
\]
if and only if \(\Prof(r)=\Prof(r')\) after explicit source-side transport.
The relation is undefined outside the declared subdomain and returns
\(\Bottom\) rather than importing target interpretation.
\end{definition}

\begin{definition}[Profile separation]
For \(r,r'\in\Adm^{\mathrm{PMS}}_{\mathrm{src}}(E)\), define
\[
  r\SepProf r'
\]
when at least one source component of \(\Prof(r)\) is not equal to the
corresponding source component of \(\Prof(r')\) under permitted source
transport, or when \(\RRE\)-retention blocks source-side identification. If a
difference can be stated only by detector, empirical, stress-energy, matter
action, metric, Einstein-equation, or benchmark language, the branch is
\(\Obstruction\), not \(\SepProf\).
\end{definition}

\begin{theorem}[Equivalence on the profile subdomain]
\(\EqProf\) is reflexive, symmetric, and transitive on
\(\Adm^{\mathrm{PMS}}_{\mathrm{src}}(E)\).
\end{theorem}

\begin{proof}
Each profile is a finite tuple of source-side transport classes plus the
no-target certificate bundle. Reflexivity follows from equality of each
component with itself. Symmetry follows because equality and declared
source-transport equivalence are symmetric. Transitivity follows because
source-transport composition is part of the declared admissible transport
discipline. No target metric, detector protocol, stress-energy semantics,
matter action, Einstein-equation premise, benchmark result, validator result,
registry row, role identity, approval state, commit state, or file ordering is
used.
\end{proof}

\begin{lemma}[Source separation]
If \(r\SepProf r'\), the separation is witnessed by a named source component
of \(\Prof(r)\) and \(\Prof(r')\) or by retained \(\RRE\)-syntax
nonidentification.
\end{lemma}

\begin{proof}
This is immediate from the definition of \(\SepProf\). The candidate allows no
other separation witnesses. Any proposed separation depending on detector,
empirical, metric, stress-energy, matter-action, Einstein-equation, benchmark,
process-authority, or generated-derivative data is routed to
\(\Obstruction\).
\end{proof}

\section{Finite Variation And Certificate Deletion Behavior}

\begin{definition}[Permitted finite variation]
A finite variation \(\delta\) is permitted for this candidate when it sends
each source component of \(\Prof(r)\) either to a transported source component
of \(\Prof(\delta r)\) or to a declared missing-component branch.
\end{definition}

\begin{proposition}[Finite-variation fail-closed behavior]
For any permitted finite variation \(\delta\), either
\[
  \Prof(\delta r)=\Transport_\delta(\Prof(r))
\]
and \(\lambdaProf(\delta r)=\Pattern\), or the candidate returns
\(\Bottom\) or \(\Obstruction\).
\end{proposition}

\begin{proof}
The candidate profile is defined componentwise. If every component transports,
the profile transports. If a component is missing or a forbidden import is
introduced, the candidate uses the explicit bottom or obstruction branches.
\end{proof}

\begin{proposition}[Scoped-evidence deletion test]
Deleting scoped \(\beta\) or \(\alpha\) evidence/precondition fields does not
make \(\Prof(r)\) a target-derived object. It sets \(\Ready(r)=0\) and blocks
audit-readiness or adoption routing.
\end{proposition}

\begin{proof}
\(\Prof(r)\) contains source support, source pattern, source probe-token,
source certificate, retained \(\RRE\) syntax, and no-target certificate data.
It does not contain \(\beta\) or \(\alpha\) as label-defining fields.
Therefore their deletion changes only \(\Ready(r)\), not the source profile.
\end{proof}

\section{Fail-Closed Map}

\begin{failure}[Missing source component]
If \(S_r\), \(X_r\), \(P_r\), \(K_r\), \(Q_r\), or \(\Cert(r)\) is absent,
\(\lambdaProf(r)=\Bottom\).
\end{failure}

\begin{failure}[Forbidden import]
If any component of \(\Prof\), \(\lambdaProf\), \(\EqProf\), or \(\SepProf\)
requires detector protocols, empirical calibration, target metric, Lorentzian
signature, proper time, stress-energy semantics, stress-energy tensor, matter
action, imported conservation law, \(\MetricData(E)\), \(\geff\) expansion,
Einstein equations, benchmark fit, generated derivative authority, registry
authority, validator authority, role authority, approval state, commit state,
or file-order authority, the candidate returns \(\Obstruction\).
\end{failure}

\begin{failure}[\(\RRE\) collapse]
If two records with distinct retained \(\RRE\) syntax are identified without a
source-side transport certificate, the candidate returns \(\Obstruction\).
\end{failure}

\begin{failure}[Evidence-as-adoption]
If \(\MSBridge\) or \(\SMSAR\) evidence/precondition status is read as
adopted matter semantics, detector semantics, source law, source-extension data
beyond exact scoped Gate Chair results, coupling law, matter coupling,
stress-energy semantics, matter action, Einstein equations, benchmark
promotion, or completed derivation, that use returns \(\Obstruction\).
\end{failure}
