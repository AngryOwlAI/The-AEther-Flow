\section{Primitive Carrier}

\begin{definition}[Source object index]
A declared source object index is a finite or locally finite label
\[
  \iota \in I_{\mathrm{src}}
\]
attached to a declared source-side object \(X_{\iota}\in\Obj\). The index is
part of the certificate data. Equality or transport of indices is never
inferred from detector protocol, target metric structure, benchmark behavior,
registry state, validator state, role identity, handoff state, file order,
local cache state, or commit state.
\end{definition}

\begin{definition}[Certificate domain and codomain]
For every source certificate \(c\in\CertSrc\), the primitive fields
\[
  \Cdom(c) \in \Dom,\qquad \Ccod(c) \in \Cod
\]
name declared source-side domain and codomain data. The pair
\((\Cdom(c),\Ccod(c))\) must include source object indices, source labels,
source guards, and the source readout or response-token type relevant to the
certificate. These fields are not target-chart domains, detector-event
classes, stress-energy data, or matter-action data.
\end{definition}

\begin{definition}[Certificate record]
A certificate record is a tuple
\[
  c=(\kappa,\Cidx(c),\Cdom(c),\Ccod(c),W_c,G_c,S_c)
\]
where \(\kappa\) is the certificate kind, \(\Cidx(c)\) is the declared source
object-index map, \(W_c\) is the source witness payload, \(G_c\) is the
source-side validity guard, and \(S_c\) is the status field. Permitted
primitive kinds in this packet are
\[
  \kappa\in
  \{\mathrm{transport},\mathrm{invariance},\mathrm{factorization},
  \mathrm{identity},\mathrm{composition},\mathrm{restriction}\}.
\]
The status field may be valid, missing, malformed, or fail-closed.
\end{definition}

\section{Certificate Kinds}

\begin{primitive}[Source transport certificate]
A source transport certificate \(\CTr(A,B)\) is a source certificate whose
witness \(W_c\) maps declared source indices, labels, guards, and record
positions from a source object \(A\) to a source object \(B\). It may preserve
or compare source readout and response-token data only inside its declared
source domain and codomain.
\end{primitive}

\begin{primitive}[Source invariance certificate]
A source invariance certificate \(\CInv(A,B)\) is a source certificate whose
witness \(W_c\) states that the relevant source labels, guards, and response
token relations are unchanged under a declared source-side relabeling,
source-side variation, or local source comparison. It is not an empirical
invariance claim and not a target-metric invariance claim.
\end{primitive}

\begin{primitive}[Source factorization certificate]
A source factorization certificate \(\CFact(A,B;F)\) is a source certificate
whose witness \(W_c\) states that the comparison from \(A\) to \(B\) factors
through a declared source object \(F\). The object \(F\) must be present in
the certificate index. If \(F\) changes, is absent, or is supplied only by
scoped evidence-status, the certificate is malformed or missing for the
intended use.
\end{primitive}

\section{Algebra Primitives}

\begin{primitive}[Identity certificate]
For a declared source object \(A\), \(\CId_A\) is a certificate record with
\[
  \Cdom(\CId_A)=A,\qquad \Ccod(\CId_A)=A.
\]
It is only a primitive identity witness for the declared source object. This
packet does not prove that identity certificates preserve any downstream
equivalence relation; that operation law is deferred to P3-T02.
\end{primitive}

\begin{primitive}[Certificate composition]
For certificate records \(c_1,c_2\), the primitive composition expression
\[
  \CComp(c_2,c_1)
\]
is syntactically well formed only when the declared codomain of \(c_1\)
matches the declared domain of \(c_2\) and both certificates have valid
source-side status. If the match fails, if either status is missing or
malformed, or if either certificate imports forbidden target or detector data,
the composition expression maps to the fail-closed branch. Associativity and
preservation laws are not asserted in this packet.
\end{primitive}

\begin{primitive}[Certificate restriction]
For a certificate \(c\) and declared source subdomain \(U\subseteq\Cdom(c)\),
the primitive restriction expression
\[
  \CRes_U(c)
\]
keeps only the source indices, labels, guards, and source readout or
response-token data declared inside \(U\). If \(U\) is not a source-side
subdomain or is supplied by target metric, detector, benchmark, or process
authority data, the expression maps to the fail-closed branch.
\end{primitive}

\section{Malformed Missing And Fail-Closed States}

\begin{definition}[Malformed certificate]
\(\CMalformed(c)\) holds when \(c\) has one or more of the following defects:
undeclared source object index, domain-codomain mismatch, absent witness
payload, changed factorization object, target topology import, target metric
import, detector-semantics import, benchmark-behavior import, stress-energy
or matter-action import, process-authority dependence, or scoped
evidence-status treated as adoption.
\end{definition}

\begin{definition}[Missing certificate]
\(\CMissing(\tau)\) holds for a required certificate slot \(\tau\) when no
explicit certificate record is supplied for the declared source objects and
declared use. Scoped evidence-status, registry status, validator status,
handoff status, role status, approval status, file order, local cache state,
or commit state cannot fill a missing slot.
\end{definition}

\begin{definition}[Fail-closed branch]
\(\Fail(\tau)\) is the required branch for any missing or malformed
certificate slot \(\tau\). The branch returns \(\Bottom\) for the attempted
certificate-mediated equivalence, preserves the relevant declared-object
separation such as \(RR_E\) separation, and records a certificate-gap or
target-import obstruction when a concrete use requires one.
\end{definition}

\section{Source-Only Guard}

\begin{obligation}[No-target import]
Every source certificate primitive must be checkable without using target
topology, target smooth atlas, target Lorentzian metric, proper time,
empirical detector semantics, benchmark behavior, generated derivatives,
registry metadata, validators, roles, handoffs, approvals, local caches, file
order, or commit state as mathematical premises.
\end{obligation}

\begin{obligation}[Scoped evidence-status is not a primitive certificate]
The scoped Gate Chair status of \(\NarrowMSCertEq\) authorizes use as
evidence/precondition for this formalization route. It does not itself create
a transport, invariance, factorization, identity, composition, or restriction
certificate.
\end{obligation}

\begin{obligation}[P3-T02 handoff]
The next certificate packet may state and attempt operation laws for identity,
composition, restriction, malformed certificates, missing certificates, and
target-importing certificates. Those laws must be proved, made conditional, or
failed with a precise obstruction. This P3-T01 packet supplies only the
primitive vocabulary and fail-closed status discipline.
\end{obligation}
