\section{Branch Source Objects}

\begin{definitionblock}[Decorated finite branch source object]
A decorated finite branch source object is a tuple
\[
    B=(C,\preceq,b,\lambda,\Ann,\Proxy,\Forbid,\Infl)
\]
where \(C\) is a finite set of source carriers, \(\preceq\) is a finite
partial order, \(b=(i,o)\) marks source input and output boundary
carriers with \(i\preceq o\), and \(\lambda:C\setminus\{i,o\}\to
\{L,R\}\) is a source branch-label map. The decorations
\(\Ann,\Proxy,\Forbid,\Infl\) are source bookkeeping surfaces. They do
not encode target metric behavior, target proper time, target locality,
target observer protocol, benchmark rank, or candidate performance.
\end{definitionblock}

The annotation monoid is the same source-bookkeeping monoid used in the
prior refinement datum:
\[
    M=\langle
    m_{\mathrm{path}},m_{\mathrm{hash}},m_{\mathrm{scope}},
    m_{\mathrm{proxy}},m_{\mathrm{infl}},m_{\mathrm{limit}}
    \rangle_{\mathbb{N}}.
\]

\begin{datum}[Base branch object]
Define \(B_{1}\) by
\[
    C_{1}=\{i,\ell,r,o\},\qquad
    i\prec \ell\prec o,\qquad i\prec r\prec o,
    \qquad \ell\npreceq r,\quad r\npreceq \ell.
\]
Let \(\lambda(\ell)=L\) and \(\lambda(r)=R\). The two internal carriers
split the total annotation mass \(A\) into \(A_{L}+A_{R}=A\), where both
summands are nonempty source-bookkeeping sums. Proxy guards,
forbidden-channel guards, and influence edges are assigned by source
branch label, with
\[
    \Infl_{1}=\{(i,\ell),(\ell,o),(i,r),(r,o)\}.
\]
\end{datum}

\begin{datum}[Left-branch refinement]
Define \(B_{2}\) by subdividing only the left branch:
\[
    C_{2}=\{i,\ell_{1},\ell_{2},r,o\},\qquad
    i\prec \ell_{1}\prec \ell_{2}\prec o,\qquad
    i\prec r\prec o,
\]
with \(r\) incomparable to \(\ell_{1}\) and \(\ell_{2}\) except through
the common boundary carriers. The branch-label map sends
\(\ell_{1},\ell_{2}\) to \(L\) and \(r\) to \(R\). The left annotation
mass splits as \(A_{L}=A_{L,1}+A_{L,2}\), while the right annotation mass
is unchanged.
\end{datum}

\begin{datum}[Two-branch refinement]
Define \(B_{3}\) by subdividing both branches:
\[
    C_{3}=\{i,\ell_{1},\ell_{2},r_{1},r_{2},o\},
\]
\[
    i\prec \ell_{1}\prec \ell_{2}\prec o,\qquad
    i\prec r_{1}\prec r_{2}\prec o,
\]
with all left internal carriers incomparable to all right internal
carriers except through \(i\) and \(o\). The branch-label map sends
\(\ell_{1},\ell_{2}\) to \(L\) and \(r_{1},r_{2}\) to \(R\). Annotation
mass splits inside each branch and sums back to \(A_{L}\) and \(A_{R}\).
\end{datum}

\begin{calculation}[Non-chain and non-orbit checks]
Each \(B_{k}\) is non-chain because it contains incomparable left and
right internal carriers. The carrier cardinalities are
\[
    |C_{1}|=4,\qquad |C_{2}|=5,\qquad |C_{3}|=6.
\]
Therefore the three positive objects are not related by bijective
relabeling. The family is both branch-shaped and non-orbit under the
previous orbit criterion.
\end{calculation}

\section{Invariant and Comparison Ledger}

\begin{definitionblock}[Branch-preserving contraction]
For \(k\in\{1,2,3\}\), a contraction \(q_{k}:C_{k}\to C_{1}\) is
admissible only when it:
\begin{enumerate}
    \item fixes the boundary carriers \(i,o\);
    \item sends every left internal carrier to \(\ell\) and every right
    internal carrier to \(r\);
    \item is order-preserving;
    \item preserves incomparability between left and right internal
    branch carriers before quotienting;
    \item preserves total annotation mass separately on each branch;
    \item preserves mapped proxy guards and forbidden-channel guards; and
    \item maps influence chains to the corresponding base branch
    influence edges.
\end{enumerate}
\end{definitionblock}

\begin{datum}[Invariant ledger]
The branch invariant ledger \(\InvBr\) consists of:
\begin{enumerate}
    \item finite partial-order structure;
    \item fixed input and output boundary anchors;
    \item two source branch labels \(L,R\);
    \item branch-local annotation conservation;
    \item branch-preserving order contraction;
    \item preservation of left-right incomparability before quotient;
    \item mapped proxy guard preservation;
    \item forbidden-channel guard preservation;
    \item influence-edge contraction by branch; and
    \item explicit exclusion of target metric, target proper time, target
    locality, target observer protocol, benchmark rank, and candidate
    performance from the comparison.
\end{enumerate}
\end{datum}

\begin{definitionblock}[Branch-refinement comparison]
For \(B,B'\in\ObjBr=\{B_{1},B_{2},B_{3}\}\), define
\(\CmpBr(B,B';\InvBr)\) as follows:
\begin{enumerate}
    \item Return \(\EqBr(B,B')\) when both objects admit admissible
    branch-preserving contractions to \(B_{1}\) preserving every invariant
    in \(\InvBr\).
    \item Return \(\Block\) when either object deletes branch-local
    annotation mass, merges left and right branch mass before an
    invariant-preserving quotient is shown, introduces an unmapped proxy,
    admits a forbidden target channel, changes boundary anchors, deletes
    branch influence coverage, selects a branch because target behavior
    improves, or infers source absence from silence.
    \item Return \(\neg\EqBr(B,B')\) only when both objects are
    well-formed branch source objects but no admissible branch-preserving
    common contraction exists.
\end{enumerate}
\end{definitionblock}

\section{Local Receipts}

\begin{receipt}[Positive witness inventory]
The positive witness-pair inventory is
\[
    (B_{1},B_{2}),\qquad (B_{2},B_{3}),\qquad (B_{1},B_{3}).
\]
Each positive object admits admissible branch-preserving contraction to
\(B_{1}\). Therefore
\[
    \CmpBr(B_i,B_j;\InvBr)=\EqBr(B_i,B_j)
    \quad\text{for}\quad i,j\in\{1,2,3\}.
\]
\end{receipt}

\begin{receipt}[Local relation properties]
Inside \(\ObjBr\), reflexivity follows from the identity comparison on
each branch object. Symmetry follows because comparison is defined by
existence of admissible contractions to the same base branch object.
Transitivity follows because every positive object contracts to \(B_{1}\)
while preserving the same boundary anchors, branch labels, branch-local
annotation mass, proxy guards, forbidden-channel guards, influence
coverage, and left-right incomparability ledger.
\end{receipt}

\begin{receipt}[Invariant descent]
For every admissible \(q_{k}:B_{k}\to B_{1}\), the image object retains
the source boundary anchors, two-branch shape, branch-local annotation
totals \(A_{L}\) and \(A_{R}\), guard assignments, and influence-edge
coverage. Thus the invariant ledger descends from \(B_{k}\) to \(B_{1}\)
inside the declared branch-refinement family.
\end{receipt}

\begin{receipt}[Quotient descent within the declared family]
The quotient induced by \(q_{k}\) is permitted only after branch-local
annotation mass, branch labels, guards, and blocker state are preserved.
It does not identify a left branch with a right branch. It does not
identify a blocked object with a positive witness. Under those local
restrictions, quotient descent is supplied for \(\ObjBr\). No quotient
descent is supplied for arbitrary non-chain objects.
\end{receipt}

\begin{controlrule}[Branch erasure blocks the comparison]
Any map that collapses \(\ell\) and \(r\) to one internal carrier before
preserving branch-local annotation and blocker state returns \(\Block\).
Such a map erases the branch structure that this datum was built to test.
\end{controlrule}

\begin{controlrule}[Annotation weakening blocks the comparison]
Deleting annotation atoms, merging \(A_{L}\) and \(A_{R}\), weighting
atoms by downstream success, or selecting atoms because a candidate result
improves returns \(\Block\).
\end{controlrule}

\begin{controlrule}[Boundary or target import blocks the comparison]
Moving \(i\) or \(o\), identifying a boundary carrier with a target
observer protocol, using target metric behavior, using target proper time,
using target locality, using benchmark rank, or using candidate
performance as a selector returns \(\Block\).
\end{controlrule}
