\section{Target Definitions}

\begin{definition}[\(\MatterSector(E;\Bcur)\)]
\(\MatterSector(E;\Bcur)\) is a draft/control source record consisting of:
\begin{enumerate}[leftmargin=*]
\item a declared source domain \(S_E\) of source-admissible sector labels or
  sector slots;
\item a bottom label \(\bot_S\) for fail-closed undetermined sector status;
\item a source-only membership rule for when a label belongs to \(S_E\);
\item a source-only distinction rule for separating matter-sector candidates
  from nonmatter or undetermined source labels;
\item an explicit statement that labels in \(S_E\) are not target matter,
  stress-energy, detector, action, or coupling semantics.
\end{enumerate}

This is target grammar only. It does not assert that a non-bottom matter sector
exists for any \(E\).
\end{definition}

\begin{definition}[\(\ProbeToken(E;\Bcur)\)]
\(\ProbeToken(E;\Bcur)\) is a source-admissible token family
\(P_E\) whose elements are records
\[
  p=(\mathrm{id},\mathrm{source\_slot},\mathrm{guard\_scope},
  \mathrm{evidence\_path})
\]
such that all fields are source-side or tracked-control fields. A probe token
may mark a source-side place where a later construction tests a matter-sector
claim, but it is not an empirical detector, laboratory probe, apparatus
calibration, measurement event, proper-time readout, or physical observation.
\end{definition}

\begin{definition}[\(\DetectorReadout(E;\Bcur)\)]
\(\DetectorReadout(E;\Bcur)\) is a source-side readout record
\[
  R_E:P_E \longrightarrow A_E \cup \{\bot_R\}
\]
where \(P_E=\ProbeToken(E;\Bcur)\), \(A_E\) is a declared source-admissible
alphabet, and \(\bot_R\) is a fail-closed undetermined readout marker. A value
\(R_E(p)\) is source syntax for a later proof obligation. It is not detector
semantics, empirical readout, apparatus behavior, physical measurement, or
observer-localized data.
\end{definition}

\begin{definition}[\(\ResponseToken(E;\Bcur)\)]
\(\ResponseToken(E;\Bcur)\) is a source-side response record family \(Q_E\).
An element \(q\in Q_E\) is a tuple
\[
  q=(s,p,R_E(p),\rho,\gamma)
\]
where \(s\in S_E\cup\{\bot_S\}\), \(p\in P_E\), \(R_E(p)\) is the source
readout value, \(\rho\) is a source-admissible response marker, and
\(\gamma\) is a certificate bundle for \(\NoDetector(E)\), \(\NoSE(E)\), and
\(\NoAction(E)\). The tuple is not a coupling law, stress-energy response,
matter action response, detector response, or matter-coupling derivation.
\end{definition}

\begin{definition}[\(\NoDetector(E)\)]
\(\NoDetector(E)\) holds for a candidate target witness only if it does not use

\section{Conditional Target-Preservation Lemma}

\begin{proposition}[No-target-import preservation]
Suppose a later P5-T03 packet supplies a candidate tuple
\(\Target(E;\Bcur)=(S_E,P_E,R_E,Q_E,\Gamma_E)\) satisfying all definitions
above, and suppose every step in its construction uses only source-admissible
data relative to \(\Bcur\). Then the maximum conclusion licensed by the tuple
is a draft/control source-side matter-semantics candidate target. It does not
license source-law adoption, \(\MCPA(E)\) adoption, source-extension data
adoption beyond the exact scoped gate result, \(\MetricData(E)\) adoption,
\(\geff\) adoption or scope expansion, coupling-law adoption, matter-coupling
derivation or adoption, stress-energy semantics, stress-energy tensor
construction, detector semantics, matter action, Einstein-equation derivation,
benchmark promotion, completed derivation, future source-extension
impossibility, or global theory rejection.
\end{proposition}

\begin{proof}
By hypothesis, \(S_E\), \(P_E\), \(R_E\), \(Q_E\), and \(\Gamma_E\) are
source-admissible relative to \(\Bcur\). The definitions of
\(\NoDetector(E)\), \(\NoSE(E)\), and \(\NoAction(E)\) exclude empirical
detector semantics, stress-energy semantics, stress-energy tensor premises,
matter-action premises, and target dynamics as premises or conclusions. The
definition of \(\Target(E;\Bcur)\) adds only source-side membership,
assignment, relabeling, finite-variation, fail-closed, and no-notation-collapse
obligations. No definition contains adoption authority, canonical ontology
edit authority, a coupling-law assertion, matter-coupling derivation, field
equation, benchmark verdict, or Gate Chair verdict. Therefore the tuple can at
most be a draft/control source-side matter-semantics candidate target, and all
listed downstream conclusions remain blocked.
\end{proof}
