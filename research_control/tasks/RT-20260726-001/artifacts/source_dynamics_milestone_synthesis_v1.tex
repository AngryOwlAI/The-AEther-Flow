\section{New conditional response theorem}

\begin{theorem}[P5T08-THM-CONDITIONAL-TANGENT-RESPONSE]
\label{thm:response}
Fix \(\gamma>0\).  For the proposal-only cubic flow on \(\mathbb R\),
\[
 \phi_\lambda(a)=a(1+2\gamma\lambda a^2)^{-1/2},
\]
the tangent response exists for every \(a\in\mathbb R\) and
\(\lambda\geq0\), and is
\[
 D\phi_\lambda(a)
   =(1+2\gamma\lambda a^2)^{-3/2}.
\]
It obeys the response cocycle
\[
 D\phi_{\lambda+\mu}(a)
 =D\phi_\mu(\phi_\lambda(a))\,D\phi_\lambda(a)
 \qquad(\lambda,\mu\geq0),
\]
and its infinitesimal response generator is
\[
 DX_\gamma(a)=-3\gamma a^2.
\]
For \(\lambda>0\), \(0<D\phi_\lambda(a)\leq1\), with equality only at
\(a=0\).
\end{theorem}

\begin{proof}
Direct differentiation gives
\[
\begin{aligned}
\frac{\partial}{\partial a}
\left[a(1+2\gamma\lambda a^2)^{-1/2}\right]
&=(1+2\gamma\lambda a^2)^{-1/2}\\
&\quad-2\gamma\lambda a^2
 (1+2\gamma\lambda a^2)^{-3/2}\\
&=(1+2\gamma\lambda a^2)^{-3/2}.
\end{aligned}
\]
The P5-T03 semiflow identity
\(\phi_{\lambda+\mu}=\phi_\mu\circ\phi_\lambda\), followed by the
one-dimensional chain rule, gives the cocycle.  Differentiating
\(X_\gamma(a)=-\gamma a^3\) gives the infinitesimal formula.  Positivity and
the contraction bound follow from
\(1+2\gamma\lambda a^2\geq1\).
\end{proof}

\paragraph{Status.}
Theorem~\ref{thm:response} is an exact conditional result about the fixed
proposal-only candidate.  It does not make \(\lambda\) physical time, adopt
\(\gamma\), or identify the tangent response with a spacetime
characteristic.

\begin{theorem}[P5T08-THM-FACTOR-THROUGH-AMPLITUDE-RANK-BOUND]
\label{thm:rank}
Let \(U\) be a differentiable source-data space, let
\(a:U\to\mathbb R\) be differentiable, and let
\(B:\mathbb R\to E\) be differentiable for a finite-dimensional vector
space \(E\).  Every deterministic reconstruction
\[
  \mathcal R=B\circ a:U\to E
\]
that factors through the sole amplitude coordinate satisfies
\[
  \operatorname{rank}D\mathcal R_u\leq1
  \qquad\text{for every }u\in U.
\]
Consequently, if a proposed local characteristic or geometry input requires
two or more independently variable response directions at one source point,
the exact unchanged \(\Qamp\) route cannot supply that input by deterministic
post-processing alone.
\end{theorem}

\begin{proof}
The chain rule gives
\[
 D\mathcal R_u=DB_{a(u)}\circ Da_u.
\]
The intermediate space is one-dimensional, so the rank of the composition is
at most one.  Deterministic post-processing can change representation but
cannot create a second independent tangent direction.  Therefore a target
requiring rank at least two cannot be obtained from this factorization.
\end{proof}

\paragraph{Scope.}
Theorem~\ref{thm:rank} is a factorization theorem for the exact one-mode
candidate.  It is not a global no-go theorem.  Conservative source extensions
with independent local data, restriction and gluing structure, or new
source-derived dynamics remain possible.
