\section{Pointwise Labels and Kernel-Pair Relations}

\begin{definition}[Three transformation predicates]
For a map \(V:A_E\to A_E\), define pointwise discriminator-label fixing by
\[
\LabelFix(V)
\quad\Longleftrightarrow\quad
\chi_E(Vz)=\chi_E(z)\quad\text{for every }z\in A_E.
\]
Define relation preservation by
\[
\RelPres(V)
\quad\Longleftrightarrow\quad
z\,\EqCand_E\,z'\Longrightarrow
Vz\,\EqCand_E\,Vz'
\quad\text{for all }z,z'\in A_E.
\]
Define a relation automorphism by
\[
\RelAut(V)
\quad\Longleftrightarrow\quad
V\text{ is bijective and }
\left(z\,\EqCand_E\,z'
\Longleftrightarrow Vz\,\EqCand_E\,Vz'\right).
\]
\end{definition}

A label is the individual mathematical value \([z]\) in the displayed
quotient. It is not declared to be a physical observable or a canonically
named physical class. Pointwise label fixing is stronger than preservation of
the unlabeled kernel-pair partition.

\begin{theorem}[Quotient factorization separates the predicates]
For any \(V:A_E\to A_E\),
\[
\RelPres(V)
\quad\Longleftrightarrow\quad
\text{there exists a unique }\bar V:K_E\to K_E
\text{ with }\chi_EV=\bar V\chi_E.
\]
When this factorization exists, relation reflection is equivalent to
injectivity of \(\bar V\), and
\[
\LabelFix(V)\quad\Longleftrightarrow\quad \bar V=\mathrm{id}_{K_E}.
\]
\end{theorem}

\begin{proof}
If \(V\) preserves the relation, define
\(\bar V([z])=[Vz]\). Equal input classes are related, so their images are
related and the definition is well defined. Surjectivity of \(\chi_E\)
forces uniqueness. Conversely, a factorization maps equal quotient values to
equal quotient values, hence preserves the kernel pair.

Under the factorization, reflection says that
\(\bar V([z])=\bar V([z'])\) implies \([z]=[z']\), which is injectivity.
Pointwise label fixing says
\(\bar V([z])=[z]\) for every quotient class, which is precisely
\(\bar V=\mathrm{id}_{K_E}\).
\end{proof}

\begin{remark}[Codomain gauge]
A bijective \(\bar V\) is a mathematical relabeling of quotient values that
preserves the exact kernel pair. In this packet, ``codomain gauge'' means only
such a bijection. It need not be linear, pointed, family-natural, or a
physically adopted gauge symmetry.
\end{remark}

\section{Exact Translation Classification}

\begin{theorem}[Translation label--relation separation]
For any \(c\in A_E\), let \(\tau_c(z)=z+c\). Then
\[
\chi_E(\tau_cz)=\chi_E(z)+[c].
\]
Consequently,
\[
\LabelFix(\tau_c)\quad\Longleftrightarrow\quad c\in B_E.
\]
Nevertheless, every translation is a relation automorphism:
\[
z\,\EqCand_E\,z'
\quad\Longleftrightarrow\quad
\tau_cz\,\EqCand_E\,\tau_cz'.
\]
The induced quotient permutation is
\(\bar\tau_{[c]}(q)=q+[c]\).
\end{theorem}

\begin{proof}
Quotient linearity gives
\(\chi_E(z+c)=[z]+[c]\). This equals \([z]\) for every \(z\) exactly when
\([c]=0\), equivalently \(c\in B_E\). For the relation,
\[
(z+c)-(z'+c)=z-z',
\]
so membership of the pairwise difference in \(B_E\) is unchanged in both
directions. Translation is bijective, with inverse itself over \(\Ftwo\).
\end{proof}

Thus \(\AdmVar(E)\) is exactly the pointwise-label-fixing translation class.
It is a proper subclass of all relation-preserving translations whenever
\(K_E\) is nontrivial. This statement is mathematical and conditional on the
proposed chain package; it does not establish physical admissibility.

\section{Typed Ambient Relation-Changing Witness}

\begin{theorem}[Linear relation-stabilizer criterion]
For a linear map \(L:A_E\to A_E\),
\[
\RelPres(L)\quad\Longleftrightarrow\quad L(B_E)\subseteq B_E.
\]
For \(L\in\mathrm{GL}(A_E)\),
\[
\RelAut(L)\quad\Longleftrightarrow\quad L(B_E)=B_E.
\]
\end{theorem}

\begin{proof}
If \(L\) preserves the relation, then \(b\,\EqCand_E\,0\) for every
\(b\in B_E\), so \(Lb\,\EqCand_E\,0\) and \(Lb\in B_E\). Conversely,
\(z-z'\in B_E\) implies
\(Lz-Lz'=L(z-z')\in B_E\). For invertible \(L\), relation reflection gives
the reverse inclusion.
\end{proof}

On \(A_\star\), define
\[
L(e_0)=e_0,\qquad
L(e_1)=e_0+e_2,\qquad
L(e_2)=e_1.
\]
The three images form a basis, so \(L\in\mathrm{GL}(A_\star)\). But
\[
L(e_0+e_1)=e_2\notin B_\star.
\]
Hence \(0\,\EqCand_{E_\star}\,e_0+e_1\), while
\(L0\not\EqCand_{E_\star}L(e_0+e_1)\). Exhaustive enumeration gives 16
relation mismatches across the 64 ordered pairs.

This is an ambient invertible linear source-state map. It is not an admitted
boundary move. It cannot be the degree-one component of an admitted chain map
of \(E_\star\), because the chain condition would require
\[
L(\partial_2f)=\partial_2F_2f\in B_\star,
\]
contrary to \(L(e_0+e_1)=e_2\notin B_\star\). It is not established as a
physically admissible perturbation. Its sole role is to show that the
kernel-pair relation is not automatically invariant under every ambient
state-space bijection.

\begin{proof}

\end{proof}

\begin{proposition}[Conditional quotient uniqueness]

\[

\(d=\bar d\chi_E\).
\end{proposition}

\begin{proof}
Set \(\bar d([z])=d(z)\). Exact equality of kernel pairs gives
well-definedness and injectivity; surjectivity follows from that of \(d\);
