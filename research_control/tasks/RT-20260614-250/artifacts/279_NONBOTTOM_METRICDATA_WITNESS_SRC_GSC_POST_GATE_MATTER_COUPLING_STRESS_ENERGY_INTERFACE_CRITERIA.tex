\section{Interface Intent}

The phrase ``stress-energy-interface'' in this packet means a source-side
criteria surface for a future candidate that may try to relate source
matter-like response tokens, source balance tokens, and accepted
recovery-bridge evidence. It is not stress-energy semantics. It is not a
stress-energy tensor. It is not a matter action. It is not an empirical
detector model. It is not a target-space field.

The criteria below specify what a later bounded Candidate Constructor packet
must supply before any future audit, stress, or protected review can even ask
whether a source-side interface candidate is evidence. The criteria do not
construct that candidate.

\section{Source-Side Objects}

\begin{definition}[Source interface datum]
For an event/source context \(E\), a source interface datum is a tuple
\[
  \SrcME_E = (S_E,J_E,R_E,\BalSrc_E,\FluxSrc_E,\Pi_E).
\]
Here \(S_E\) is a declared finite/local source sector family, \(J_E\) is a
source-token family, \(R_E\) is a source-side readout family,
\(\BalSrc_E\) is a source balance predicate, \(\FluxSrc_E\) is a source
transition ledger, and \(\Pi_E\) is a proof-obligation ledger. None of these
objects is a stress-energy tensor, matter action, detector semantics, target
metric datum, or empirical matter field.
\end{definition}

\begin{definition}[Source-side stress-energy-interface certificate]
A source-side stress-energy-interface certificate over a declared finite/local
family \(F_E^\sharp\) is a tuple
\[
  \SEICert(E,F_E^\sharp)=
  (F_E^\sharp,\CouplingLawCrit,\BErec,\BridgeCertCand,\SrcME_E,
  \mathcal{C}_E,\mathcal{T}_E,\NoTargetImport_E,\mathcal{O}_E,\mathcal{G}_E),
\]
where \(\mathcal{C}_E\) is a criteria ledger, \(\mathcal{T}_E\) is a
finite/local test ledger, \(\NoTargetImport_E\) is a no-target-import
certificate, \(\mathcal{O}_E\) maps failures to fail-closed obstruction
labels, and \(\mathcal{G}_E\) records route and gate separation.
\end{definition}

\section{Required Criteria}

\begin{criterion}[SEI-MC-001 evidence-status separation]
The certificate must distinguish accepted source-extension evidence from
adoption. The scoped evidence statuses of \(\CouplingLawCrit\), \(\BErec\),
and \(\BridgeCertCand(E,F_E^\sharp)\) may be cited only as preconditions.
They must not be used as source-law adoption, coupling-law adoption,
matter-coupling derivation, stress-energy semantics, Einstein equations,
benchmark promotion, or completed derivation.
\end{criterion}

\begin{criterion}[SEI-MC-002 source interface datum declaration]
The certificate must declare \(F_E^\sharp\), \(S_E\), \(J_E\), \(R_E\),
\(\BalSrc_E\), \(\FluxSrc_E\), and \(\Pi_E\) using source-side objects only.
If any component is repaired by target metric, target stress-energy, matter
action, detector calibration, or empirical benchmark fit, the construction
fails closed.
\end{criterion}

\begin{criterion}[SEI-MC-003 balance-token nontriviality]
\(\BalSrc_E\) must be more than a label. On the declared finite/local family,
it must accept at least one source-token transition and reject at least one
source-token transition for source-intrinsic reasons. All acceptance and
rejection witnesses must cite source-side records.
\end{criterion}

\begin{criterion}[SEI-MC-004 source flux ledger]
\(\FluxSrc_E\) must record how declared source transitions transport or
modify interface tokens. It may be a finite/local ledger, not a continuum
tensor. It must fail closed when transition data are absent, underdeclared, or
dependent on target geometry.
\end{criterion}

\begin{criterion}[SEI-MC-005 recovery-bridge compatibility]
The certificate must state how \(\BErec\) and
\(\BridgeCertCand(E,F_E^\sharp)\) are used as recovery-bridge evidence. This
use may constrain source-token routing, proof obligations, or tests. It must
not treat \(\BErec\) as matter coupling and must not treat
\(\BridgeCertCand\) as stress-energy semantics.
\end{criterion}

\begin{criterion}[SEI-MC-006 sector-uniformity analogue]
The same source-side interface rule must apply across the declared
source-sector family \(S_E\), with only declared source parameters varying by
sector. Hand-tailored rules per sector, target-repaired universality, or
benchmark-repaired universality fail closed.
\end{criterion}

\begin{criterion}[SEI-MC-007 compatibility with scoped \(\geff\)]
Any compatibility reference to scoped \(\geff\) must remain a source-extension
compatibility check. It must not adopt \(\MetricData(E)\), expand \(\geff\)'s
scope, import Lorentzian signature, import proper time, import target metric
covariance, or invoke Einstein-equation coupling.
\end{criterion}

\begin{criterion}[SEI-MC-008 no-target-import certificate]
The certificate must include \(\NoTargetImport_E\). The scan must rule out
target topology, target atlas, target metric, stress-energy tensor, matter
action, detector calibration, empirical matter fields, benchmark fit,
Einstein equations, validator status, registry metadata, generated
derivatives, role authority, handoff status, approval status, file order,
commit status, local caches, or PDFs as mathematical premises.
\end{criterion}

\begin{criterion}[SEI-MC-009 finite/local discriminating tests]
\(\mathcal{T}_E\) must contain tests that can distinguish a genuine
source-side interface candidate from a restatement of accepted evidence. At
minimum it must test nonempty interface output, balance-token nontriviality,
flux-ledger transition coherence, sector-uniformity, recovery-bridge
compatibility, scoped-\(\geff\) boundary preservation, perturbation stability,
and no-target-import compliance.
\end{criterion}

\begin{criterion}[SEI-MC-010 route and gate separation]
Criteria satisfaction may authorize only a later bounded Candidate
Constructor packet, Smuggling Auditor review, Refuter stress test, or
separately authorized protected Gate Chair review. Criteria satisfaction is
not stress-energy semantics, not matter coupling, not a coupling-law adoption,
and not Einstein equations.
\end{criterion}

\section{Interface-Criteria Eligibility Theorem}

\begin{theorem}[Stress-energy-interface criteria eligibility theorem]
Suppose a later draft/control packet supplies
\(\SEICert(E,F_E^\sharp)\) satisfying SEI-MC-001 through SEI-MC-010 for a
declared finite/local source family \(F_E^\sharp\). Then the packet is
eligible only for bounded source-side interface candidate construction,
Smuggling Auditor review, Refuter stress testing, or separately authorized
protected review:
\[
  \SEICert(E,F_E^\sharp) \in \EligibleSEI(E,F_E^\sharp).
\]
This eligibility does not adopt a source law, adopt
\(\CouplingLawCrit(E;\Gbeta,T_{\mathrm{src}}(E))\), derive or adopt matter
coupling, import stress-energy semantics, construct a stress-energy tensor,
import a matter action, adopt \(\MetricData(E)\), change scoped \(\geff\),
derive Einstein equations, promote benchmark status, or claim completed
derivation.
\end{theorem}

\begin{proof}
SEI-MC-001 separates evidence/precondition acceptance from adoption. SEI-MC-002
requires a source-side interface datum. SEI-MC-003 and SEI-MC-004 require
nontrivial source balance and source flux ledgers without continuum tensor or
target geometry import. SEI-MC-005 prevents recovery-bridge evidence from
being overread as matter coupling or stress-energy semantics. SEI-MC-006 and
SEI-MC-009 require finite/local tests that distinguish a candidate from a
restatement of accepted evidence. SEI-MC-007 and SEI-MC-008 preserve the
scoped \(\geff\) and no-target-import boundaries. SEI-MC-010 separates
criteria satisfaction from route and gate authority. Therefore the criteria
can establish only eligibility for a later bounded construction, audit, stress,
or separately authorized protected review. None of the hypotheses supplies an
adoption verdict, target stress-energy tensor, matter action, empirical matter
semantics, variational field equation, benchmark verdict, or completed
derivation.
\end{proof}
