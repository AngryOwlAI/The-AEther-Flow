\section{Finite Source Witness}

\begin{definition}[Local three-atom witness \(X_{\mathrm{loc}}\)]
Let
\[
    a_j=(b_j,\lambda_j,\alpha_j,p_j,f_j,k_j,\iota_j,q_j),
    \qquad j\in\{0,1,2\},
\]
be three ledger atoms whose coordinates are source tokens. Define
\[
    X_{\mathrm{loc}}
    =
    (B_X,i_X,o_X,A_X,E_X,C_X)
\]
with
\[
    B_X=\{a_0,a_1,a_2\},\qquad i_X=\partial^-_X=a_0,
    \qquad o_X=\partial^+_X=a_2,
\]
\[
    E_X=\{(a_0,a_1),(a_1,a_2)\},
    \qquad C_X=\{1_X,\sigma_X\}.
\]
Here \(1_X\) is identity transport and \(\sigma_X\) is a source
relabeling certificate that sends \(b_j\mapsto b'_j\) while carrying
the full annotation, guard, blocker, influence, quotient, contraction,
and boundary-anchor data with each atom. The certificate is rejected
if any of those coordinates are deleted, weakened, or selected by
target behavior.
\end{definition}

\begin{lemma}[Membership]
\(X_{\mathrm{loc}}\in\ObjLC\), and \(1_X,\sigma_X\in\CertLC\).
\end{lemma}

\begin{proof}
Every branch token in \(B_X\) carries annotation, proxy guard,
forbidden-channel guard, blocker, influence, quotient-fiber, and
boundary-anchor provenance. The identity certificate preserves each
coordinate. The relabeling certificate preserves the same coordinates
by transported token identity, not by target interpretation. Thus the
object and certificates satisfy the ledger-complete membership and
certificate clauses. The proof is local to this declared finite
witness family.
\end{proof}

\section{Observer-Localizing Soldering Rule}

\begin{construction}[Exact local \(\ObsLocLC\)]
For \(X_{\mathrm{loc}}\), define
\[
    U_X=\{a_0,a_1,a_2\}
\]
with ordered source boundary \((a_0,a_2)\). Define the source normal
token
\[
    \nu_X=(a_0\mid a_1\mid a_2,E_X,\partial^-_X,\partial^+_X)
\]
and define \([n]_{U_X}\) to be the orbit of \(\nu_X\) under exact
\(\CertLC\) transport. Define
\[
    \Phi_{U_X}
    =
    \ReadLC(X_{\mathrm{loc}})|_{U_X}
    =
    (|B_X|,\partial^-_X,\partial^+_X,E_X,
    (p_j,f_j,k_j,\iota_j,q_j)_{j=0}^{2}).
\]
For \(c\in\{1_X,\sigma_X\}\), let \(\tau_c\) be the source-ledger
transport of atoms and coordinates. Set
\[
    T_c^n([n]_{U_X})=[\tau_c(\nu_X)],
    \qquad
    T_c^\Phi(\Phi_{U_X})=\ReadLC(cX)|_{\tau_c(U_X)}.
\]
Then
\[
    \ObsLocLC(X_{\mathrm{loc}},c)
    =
    (U_X,[n]_{U_X},\Phi_{U_X},T_c^n,T_c^\Phi).
\]
\end{construction}

\begin{theorem}[Exact-branch certificate transport]
On the exact local certificate family
\(\mathcal{C}^{=}_{\mathrm{loc}}=\{1_X,\sigma_X\}\), the construction
is source-only and functorial:
\[
    T_{1_X}^n=\mathrm{id},\qquad T_{1_X}^{\Phi}=\mathrm{id},
\]
and whenever \(d\circ c\) is defined inside
\(\mathcal{C}^{=}_{\mathrm{loc}}\),
\[
    T_{d\circ c}^n=T_d^n\circ T_c^n,
    \qquad
    T_{d\circ c}^{\Phi}=T_d^\Phi\circ T_c^\Phi .
\]
\end{theorem}

\begin{proof}
The transport \(\tau_c\) is a source-token map that preserves the full
ledger tuple of each atom and preserves ordered boundary anchors. The
identity statements follow from \(\tau_{1_X}=\mathrm{id}\). The
composition statements follow because exact certificates compose by
ordered source-ledger concatenation, and \(\ReadLC\) is restricted to
the transported source neighborhood rather than recomputed from target
observables. No metric, proper time, target locality, observer
protocol, benchmark rank, validation status, or role authority enters
the definition.
\end{proof}

\section{Finite Negative Controls}

\begin{controlrule}[Guard deletion returns \(\Block\)]
Let \(X_{\mathrm{guard}}\) be obtained from \(X_{\mathrm{loc}}\) by
deleting \(k_1\) or weakening \(f_1\) while keeping branch counts
fixed. Then \(X_{\mathrm{guard}}\notin\ObjLC\), and no
\(\ObsLocLC(X_{\mathrm{guard}},c)\) is admitted.
\end{controlrule}

\begin{controlrule}[Boundary flip is not hidden locality]
Let \(X_{\partial}\) swap \(\partial^-_X\) and \(\partial^+_X\) while
leaving the multiset \(B_X\) unchanged. Unless a source certificate
records the ordered boundary transport explicitly, \(X_{\partial}\)
does not inherit \(T^n\) or \(T^\Phi\). The boundary order is not
physical locality and is not proper time.
\end{controlrule}

\begin{controlrule}[Readout deletion is not invariance]
Let \(\Phi^{-}_{U_X}\) delete the influence coordinate \(\iota_1\).
Then \(X_{\mathrm{loc}}\) and a variant with shifted influence
assignment can agree under \(\Phi^{-}_{U_X}\) while differing under
\(\Phi_{U_X}\). Therefore coordinate deletion is not a minimality
proof and cannot be used as bridge evidence.
\end{controlrule}

\begin{controlrule}[Benchmark selectors are forbidden]
No part of \(U_X\), \([n]_{U_X}\), \(\Phi_{U_X}\), \(T^n\), or
\(T^\Phi\) may be chosen because a downstream candidate performs well,
matches a target metric, or improves benchmark rank. Such selection
returns \(\Block\).
\end{controlrule}
