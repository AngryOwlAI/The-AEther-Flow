\section{Finite Source Data}

\begin{definition}[Finite repaired source-atlas datum]
Let the source patch labels be \(I=\{0,1,2\}\). Let
\[
U_0=\{a_0,b_0\},\qquad
U_1=\{a_1,b_1\},\qquad
U_2=\{a_2,b_2\}.
\]
Define explicit source discriminator profiles
\[
\begin{aligned}
\Prof_0(a_0)&=(A,L,0), & \Prof_0(b_0)&=(B,R,0),\\
\Prof_1(a_1)&=(A,L,1), & \Prof_1(b_1)&=(B,R,1),\\
\Prof_2(a_2)&=(A,L,2), & \Prof_2(b_2)&=(B,R,2).
\end{aligned}
\]
The first coordinate is a source class tag, the second is a source orientation
tag, and the third is a patch tag. These are declared source tokens. They are
not target coordinates, metric data, detector readings, or benchmark labels.
\end{definition}

\begin{definition}[Candidate transition family]
For each \(i<j\), let the intended transition token be
\[
\tau_{ij}^{+}(a_i)=a_j,\qquad \tau_{ij}^{+}(b_i)=b_j.
\]
Let \(\tau_{ji}^{+}=(\tau_{ij}^{+})^{-1}\) and let
\(\tau_{ii}^{+}\) be the identity on \(U_i\).
\end{definition}

\begin{definition}[Source selector preorder]
For each pair \((i,j)\), the candidate set \(\Cand_{ij}\) contains source
partial bijections \(U_i\to U_j\). Assign each candidate \(c\) the source
score
\[
\sigma_{ij}(c)=
(d_{\Delta}(c),d_{\mathrm{orient}}(c),d_{\mathrm{inv}}(c),
d_{\mathrm{cocycle}}(c)),
\]
ordered lexicographically. The terms count, respectively, discriminator class
mismatches, orientation mismatches, inverse-check defects, and cocycle-check
defects. The selector \(\Sel_{ij}\) returns the unique source-minimal
candidate when it exists and returns \(\Bot\) otherwise.
\end{definition}

\section{Construction}

\begin{construction}[Finite witness \(\Wfin\)]
Let \(\Wfin\) be the datum consisting of the three source carriers, the
profile family \(\Prof\), the candidate families \(\Cand_{ij}\), the selector
preorders above, and the selected transitions \(\tau_{ij}^{+}\).

For each \(i<j\), the transition \(\tau_{ij}^{+}\) has score
\((0,0,0,0)\). A swap candidate \(a_i\mapsto b_j,\ b_i\mapsto a_j\) has
positive discriminator and orientation mismatch score. Hence
\(\tau_{ij}^{+}\) is the unique selected source-minimal candidate.
\end{construction}

\begin{checkresult}[Inverse]
For every declared pair, \(\tau_{ji}^{+}=(\tau_{ij}^{+})^{-1}\).
\end{checkresult}

\begin{checkresult}[Cocycle]
On the triple \((0,1,2)\),
\[
\tau_{12}^{+}(\tau_{01}^{+}(a_0))=a_2=\tau_{02}^{+}(a_0),
\qquad
\tau_{12}^{+}(\tau_{01}^{+}(b_0))=b_2=\tau_{02}^{+}(b_0).
\]
The inverse-oriented cocycles follow by inversion of the same selected source
tokens.
\end{checkresult}

\begin{construction}[Finite source quotient support]
Form the disjoint union
\[
D=(\{0\}\times U_0)\sqcup(\{1\}\times U_1)\sqcup(\{2\}\times U_2)
\]
and quotient by the selected transition identifications. The quotient
\(\Qdisc\) has two source equivalence classes:
\[
[(0,a_0)]=[(1,a_1)]=[(2,a_2)],
\qquad
[(0,b_0)]=[(1,b_1)]=[(2,b_2)].
\]
This is a finite source quotient support. It is not a target manifold and not
full \(\Msrc\).
\end{construction}

\section{Failure Branches}

\begin{failure}[Discriminator collapse]
If the source profiles collapse the class tags \(A\) and \(B\), then the
identity-style and swap-style candidates are no longer separated by the
source score. The selector returns \(\Bot\).
\end{failure}

\begin{failure}[Selector nonuniqueness]
If two candidates have the same minimal source score, then no role label, file
order, validation status, generated derivative, benchmark resemblance, or
target chart may break the tie. The selector returns \(\Bot\).
\end{failure}

\begin{failure}[Inverse defect]
If a selected transition lacks the declared inverse transition, the repaired
witness returns \(\Bot\).
\end{failure}

\begin{failure}[Cocycle defect]
If a triple-overlap composition differs from the selected direct transition,
the repaired witness returns \(\Bot\).
\end{failure}

\section{Finite Witness Lemma}

\begin{lemma}[Finite repaired source-atlas witness]
The datum \(\Wfin\) realizes a finite positive branch of \(\AtlasGlueDisc\):
the source discriminator profiles separate the candidate transition families,
the selector preorders choose unique source-minimal transitions, inverse and
cocycle checks pass, and the resulting source quotient support \(\Qdisc\) is
defined without target or process-authority input.
\end{lemma}

\begin{proof}
The profiles distinguish \(A\)-class from \(B\)-class source tokens and assign
the same orientation tags along intended transitions. Therefore each intended
\(\tau_{ij}^{+}\) has zero discriminator and orientation mismatch. By
construction the inverse and cocycle defect terms are also zero for these
transitions. Any swap candidate has positive discriminator mismatch, so the
intended transition is the unique source-minimal candidate for each pair.
The inverse and cocycle checks are explicit source-token equations. The
quotient support is then the ordinary quotient of a finite source set by the
selected transition equivalence relation. No target topology, target smooth
atlas, target transition map, metric scale, proper time, empirical observer
protocol, benchmark behavior, validation authority, role authority, or
generated derivative is used.
\end{proof}
