\section{Adverse Source-Side Variation Families}

\begin{definition}[Finite variation suite]
Let \(\VarLC\) be the finite source-side variation suite for the local
\(\EqLC\) witness line. A variation \(v\in\VarLC\) is admissible for
testing only when it changes source-ledger data and does not introduce
target metric, target proper time, target locality, observer protocol,
benchmark rank, candidate behavior, validation status, registry status,
generated visibility, handoff status, or authority label as a source
primitive.

The suite contains the following families:
\[
\VarLC=
\{V_{\mathrm{guard}},V_{\mathrm{infl}},V_{\mathrm{rep}},
V_{\mathrm{comp}},V_{\partial},V_{\mathrm{read}},V_{\mathrm{neg}}\}.
\]
\end{definition}

\begin{definition}[Variation family meanings]
The families are interpreted as follows.
\begin{enumerate}
    \item \(V_{\mathrm{guard}}\): proxy-guard weakening, forbidden-guard
    mutation, blocker erasure, blocker relocation, and blocker
    provenance splitting.
    \item \(V_{\mathrm{infl}}\): influence-record deletion, influence
    reassignment across refined branches, and influence-count
    preservation with provenance change.
    \item \(V_{\mathrm{rep}}\): source quotient representative
    substitution \(q\mapsto q'\) and representative-choice transport.
    \item \(V_{\mathrm{comp}}\): noncomposing quotient refinements,
    ambiguous refinement inverses, and certificate-history mismatch.
    \item \(V_{\partial}\): boundary-anchor perturbation, boundary
    representative substitution, and boundary-domain restriction.
    \item \(V_{\mathrm{read}}\): readout-coordinate deletion,
    readout-coordinate relaxation, overcomplete readout retention, and
    hidden under-inclusion of certificate history.
    \item \(V_{\mathrm{neg}}\): negative-control variants that preserve
    selected readout counts while changing source provenance,
    representative choice, inverse recoverability, or boundary
    anchoring.
\end{enumerate}
\end{definition}

\begin{controlrule}[Admissibility]
Every variation in \(\VarLC\) is a source-side stress input only. If a
variation must be selected by target success, benchmark convenience,
candidate behavior, downstream clock behavior, downstream causal
structure, matter coupling, or authority metadata, then it is not an
admissible \(\VarLC\) variation.
\end{controlrule}

\section{Representative Independence}

\begin{obligation}[Representative-independence burden]
Let \(\Theta_Q\) be a future source-side equivalence relation on
quotient representatives. The representative-independence obligation
\(\RepLC\) is satisfied only if all of the following are supplied as
source-side data:
\begin{enumerate}
    \item an explicit definition of \(\Theta_Q\) independent of target
    primitives;
    \item a lifted readout relation \(\ReadLC/\Theta_Q\) or an explicit
    proof that exact \(\ReadLC\) equality is the intended conservative
    criterion;
    \item a proof that valid \(\CertLC\) certificates preserve
    \(\Theta_Q\)-classes under finite composition; and
    \item inverse-provenance data showing that quotient representative
    changes do not erase the certificate history needed to recover the
    source ledger.
\end{enumerate}
\end{obligation}

\begin{failurecondition}[Representative failure]
The local line remains in \(\Pause\) or returns \(\Repair\) if
\(\Theta_Q\) is absent, if representative equality is relaxed without a
source-side lifted readout relation, if \(\CertLC\) composition fails
under representative substitution, or if endpoint readout equality
holds while inverse provenance is unrecoverable. The branch returns
local \(\Kill\) only if representative independence is required for the
declared source objective and every source-only repair imports a target
primitive or authority label.
\end{failurecondition}

\section{Boundary Invariance}

\begin{obligation}[Boundary-invariance burden]
Let \(\Theta_{\partial}\) be a future source-side equivalence relation
on ordered boundary anchors. The boundary-invariance obligation
\(\BndLC\) is satisfied only if all of the following are supplied:
\begin{enumerate}
    \item a source-only definition of boundary-anchor equivalence;
    \item a boundary transport law for finite \(\ObjLC\) objects;
    \item a proof that valid \(\CertLC\) certificates preserve boundary
    equivalence under finite composition and inverse recovery; and
    \item a statement of whether \(\ReadLC\) remains exact on boundary
    anchors or is replaced by a boundary-quotiented readout.
\end{enumerate}
\end{obligation}

\begin{failurecondition}[Boundary failure]
Exact boundary anchors remain conservative but not robust. A relaxed
boundary readout without \(\Theta_{\partial}\), without transport
proof, or without inverse recovery returns \(\Repair\) or \(\Block\).
If boundary equivalence can only be justified by target geometry,
observer protocol, benchmark rank, candidate behavior, or downstream
success, the attempted boundary repair is rejected inside this source
line.
\end{failurecondition}

\section{Quotient Composition and Inverse Provenance}

\begin{obligation}[Composition robustness]
The quotient-composition obligation \(\CompLC\) requires a source-side
finite category or composition ledger for certificate steps. It must
state identities, associativity, representative matching at
intermediate objects, and compatibility between endpoint readout and
intermediate certificate history.
\end{obligation}

\begin{obligation}[Inverse provenance]
The inverse-provenance obligation \(\InvLC\) requires each accepted
certificate step to carry enough source-ledger information to recover
branch-level blockers, influence records, quotient representatives,
boundary anchors, and contraction labels after refinement and
composition. Multiset equality alone is not sufficient unless a proof
shows that the omitted history is source-side irrelevant.
\end{obligation}

\begin{failurecondition}[Composition or inverse failure]
If two locally plausible certificate steps cannot compose because their
intermediate quotient representatives differ, or if a refinement keeps
\(\ReadLC\) counts fixed while moving provenance between refined
branches, the result is not an \(\EqLC\) robustness pass. It is a
source-side \(\Repair\) obligation or a local branch \(\Kill\) if no
source-only inverse-provenance repair exists.
\end{failurecondition}

\section{Readout Minimality and Discriminating Power}

\begin{definition}[Minimality]
\(\MinLC\) is a source-side readout-minimality criterion. It asks
whether any coordinate of \(\ReadLC\) can be removed while preserving
all currently required distinctions among the positive witness, the
near-miss negative witness, failed-certificate witnesses,
representative variations, boundary variations, inverse-provenance
variations, and quotient-composition variations.
\end{definition}

\begin{definition}[Discriminating power]
\(\DiscLC\) is a source-side discriminating-power criterion. It asks
whether equality of readout values, possibly after explicitly declared
source quotienting, rules out all adverse finite variations that matter
to the declared \(\EqLC\) certificate relation.
\end{definition}

\begin{proposition}[Minimality and discriminants are separate]
\(\MinLC\) and \(\DiscLC\) are independent burdens for this line. A
readout may be overcomplete and still discriminating for a local
certificate relation. A readout may be small and still fail to
discriminate inverse provenance or representative-choice defects.
\end{proposition}

\begin{proof}
The existing \(\ReadLC\) records many coordinates already preserved by
\(\CertLC\), so it may be overcomplete. The \texttt{RT-20260614-028}
stress cases also show that endpoint multiset equality can miss
branch-level provenance, certificate history, and representative-choice
defects, so the readout can be under-discriminating for converse or
robustness use. These properties concern different logical directions.
\end{proof}

\begin{obligation}[Readout obligation table]
Any future readout repair must supply a finite table with one row for
each \(\VarLC\) family and columns for:
\[
\text{coordinate tested},\quad
\text{retained or quotiented},\quad
\text{minimality reason},\quad
\text{discriminant reason},\quad
\text{failure mode if removed}.
\]
The table is source-side evidence only. It is not bridge evidence.
\end{obligation}

\begin{failurecondition}[Readout failure]
If \(\ReadLC\) equality is used as a converse for \(\EqLC\) without
\(\DiscLC\), the use returns \(\Block\). If a coordinate is removed
without preserving the negative-control and failed-certificate
distinctions, the repair returns \(\Repair\). If a coordinate is kept
only because it improves downstream target behavior or benchmark
success, the repair is rejected as target import.
\end{failurecondition}
