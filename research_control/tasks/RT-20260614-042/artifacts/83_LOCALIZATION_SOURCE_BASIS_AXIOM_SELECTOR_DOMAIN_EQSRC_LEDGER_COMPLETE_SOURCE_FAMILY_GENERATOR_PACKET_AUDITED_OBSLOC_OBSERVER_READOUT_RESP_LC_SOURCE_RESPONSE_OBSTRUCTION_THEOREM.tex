\section{Admissibility Definitions}

\begin{definition}[Determined by retained source data]
A candidate \(\RespLC\) is determined by \(S_X\) only if every
automorphism or relabeling that preserves all fields of
\(S_X\) also preserves the candidate's response output up to a stated
source-side equivalence relation.
\end{definition}

\begin{definition}[Nondegenerate response output]
A response output is nondegenerate when at least one of the following
is source-meaningful:
\begin{enumerate}
    \item a sign distinction \(\sigma_{ij}\neq \sigma'_{ij}\);
    \item a nonzero relative scale or weight \(w_{ij}\);
    \item a response token \(\rho_{ij}\) that is not merely an empty
    label.
\end{enumerate}
If all such distinctions are collapsed without a source-side
irrelevance theorem, the output cannot support a later
metric-response law.
\end{definition}

\begin{definition}[Forbidden import]
The construction may not choose sign, scale, pairing, response token,
matter hook, target metric signature, target proper time, target
observer protocol, benchmark rank, candidate behavior, validation
status, role authority, or promotion status from outside \(S_X\) and
the inspected exact-branch source-control data.
\end{definition}

\section{Symmetry Pressure}

\begin{lemma}[Sign symmetry]
The retained tuple \(S_X\) contains a finite set, a normal-orbit
token, a restricted readout, and exact transports. It contains no
ordered causal relation, orientation convention, timelike/spacelike
split, or signed pairing. Therefore the replacement
\[
    \sigma_{ij}\mapsto -\sigma_{ij}
\]
changes a signed response output but preserves every recorded field of
\(S_X\).
\end{lemma}

\begin{proof}
No field in \(S_X\) is typed as a signed interval, causal cone,
orientation, or bilinear form. The source normal-orbit token is not a
timelike vector and the readout \(\Phi_{U_X}\) is not an empirical
clock. Since no component of \(S_X\) distinguishes a sign choice, sign
reversal changes only the proposed response codomain. It leaves
\(U_X\), \([n]_{U_X}\), \(\Phi_{U_X}\), \(T^n\), and \(T^\Phi\)
unchanged.
\end{proof}

\begin{lemma}[Scale symmetry]
For every positive scalar \(\lambda>0\), the replacement
\[
    w_{ij}\mapsto \lambda w_{ij}
\]
changes a weighted response output but preserves the retained tuple
\(S_X\). The tuple contains no source-internal normalization,
dimensionful unit, or scale selector that prefers \(w\) over
\(\lambda w\).
\end{lemma}

\begin{proof}
The exact transports \(T^n,T^\Phi\) record source-certificate
transport in the exact branch. They do not specify a metric scale,
proper-time unit, detector calibration, or response normalization.
Selecting one would require an additional source primitive or a
forbidden target-facing fit. Hence positive rescaling is invisible to
the retained source tuple.
\end{proof}

\begin{lemma}[Response-token relabeling]
Let \(\tau\) be a bijection of any proposed response-token set that
does not act on \(U_X\), \([n]_{U_X}\), \(\Phi_{U_X}\), \(T^n\), or
\(T^\Phi\). Then
\[
    \rho_{ij}\mapsto \tau(\rho_{ij})
\]
changes response-token identity while preserving all retained source
data.
\end{lemma}

\begin{proof}
The tuple \(S_X\) contains no matter field, stress-energy response,
detector output, coupling channel, or empirical response class. A
permutation of response tokens therefore has no source-side anchor in
the retained tuple. It changes only the proposed \(\RespLC\) codomain.
\end{proof}

\section{Obstruction Theorem}

\begin{theorem}[RespLC source-response underdetermination]
From the retained tuple
\[
    S_X=(U_X,[n]_{U_X},\Phi_{U_X},T^n,T^\Phi)
\]
alone, no unique nondegenerate source response rule \(\RespLC\) is
determined. Any nondegenerate signed response output requires at least
one additional source-side selector for sign, scale, or response-token
meaning, unless a separate theorem proves the corresponding
underdetermined degrees of freedom irrelevant.
\end{theorem}

\begin{proof}
Assume a nondegenerate \(\RespLC\) is determined by \(S_X\). By the
definition of determination, every transformation preserving all
fields of \(S_X\) must preserve the response output up to an explicit
source-side equivalence relation.

The sign symmetry preserves \(S_X\) while changing \(\sigma_{ij}\).
The scale symmetry preserves \(S_X\) while changing \(w_{ij}\). The
response-token relabeling symmetry preserves \(S_X\) while changing
\(\rho_{ij}\). The current tuple supplies no equivalence relation that
identifies these changed outputs as source-irrelevant, and it supplies
no selector that chooses one representative as source-preferred.

Thus either the output is degenerate, or the proposed \(\RespLC\)
contains extra structure not determined by \(S_X\). This contradicts
the assumption of a unique nondegenerate response rule determined by
the retained tuple alone. Therefore the current source data do not
determine \(\RespLC\).
\end{proof}

\begin{corollary}[No positive bridge from naming]
Naming \(\RespLC\) as a missing primitive does not supply \(\Msrc\).
Without a source-side selector primitive or irrelevance theorem, the
diagnostic bridge schema
\[
    \Mtrial(S_X)
    =
    \mathrm{Geom}_{\mathrm{src}}
    \bigl(\Delta\Phi_{U_X},[n]_{U_X},T^n,T^\Phi\bigr)
\]
still lacks the sign, scale, covariance, and response/coupling data
needed for a non-promotional metric-response law.
\end{corollary}
