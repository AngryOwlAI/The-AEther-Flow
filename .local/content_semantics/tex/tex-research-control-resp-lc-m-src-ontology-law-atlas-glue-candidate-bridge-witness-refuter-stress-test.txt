\section{Stress Question}

The bounded question is:
\[
\text{Does the audited finite/local }\Astar\text{ bridge witness survive
Refuter pressure as written?}
\]

The answer is yes in a narrow sense. The positive branch survives only for the
exact finite/local source package and the explicitly bounded variation class.
The same result is not a general \(\Msrc\) construction and not arbitrary
finite-variation robustness.

\section{Profile Collapse}

\begin{stress}[Exact branch]
In \(\Astar\), the profile maps separate the two source classes by source
tokens:
\[
\Prof_i(a_i)=(A,L,i),\qquad \Prof_i(b_i)=(B,R,i).
\]
The source overlap relation matches class-compatible tokens. Therefore the
intended transition preserves the \(A/B\) split and the \(L/R\) distinction.
\end{stress}

\begin{countermodel}[Collapse perturbation]
Let \(v\) merge the \(A\)-class and \(B\)-class source-token values. Then
class separation no longer distinguishes the intended transition from
cross-class alternatives. Such a perturbation is not in \(\Varstar\), because
\(\Varstar\) explicitly preserves class separation. If admitted anyway, the
selector must return \(\Bot\).
\end{countermodel}

\begin{guard}[Profile-collapse result]
Profile collapse does not refute \(\Astar\) as written. It shows that the
positive branch depends on supplied source profile separation and cannot be
promoted to arbitrary profile robustness.
\end{guard}

\section{Selector Ties}

\begin{stress}[Exact source-minimality]
For each ordered pair \((i,j)\), the intended transition
\(\tau_{ij}^{\star}\) has source score \((0,0,0,0)\). A candidate that crosses
the class split or reverses the orientation token has positive source score.
Hence the exact finite package has a unique source-minimal candidate.
\end{stress}

\begin{countermodel}[Tie perturbation]
If a perturbation erases the orientation distinction or changes the scoring
relation so that two distinct candidates have the same minimum source score,
then there is no unique source-minimal transition. The selector cannot repair
the tie by file order, target resemblance, benchmark behavior, validation
status, role identity, registry visibility, generated derivative visibility,
handoff status, or the label ``source''.
\end{countermodel}

\begin{guard}[Tie result]
The selector-tie stress does not refute the exact witness. It confirms that
\(\Astar\) survives only while source-token data determine a unique minimum.
Tied cases must fail closed at \(\Bot\).
\end{guard}

\section{Inverse And Cocycle Defects}

\begin{stress}[Exact inverse table]
The witness supplies reverse transition tokens satisfying
\[
  \mathsf{inv}_{\mathrm{src}}(\tau_{ij}^{\star})=\tau_{ji}^{\star}.
\]
This is a finite source-token equality, not a target smooth-atlas theorem.
\end{stress}

\begin{stress}[Exact cocycle table]
On the triple \((0,1,2)\), the witness supplies
\[
  \mathsf{comp}_{\mathrm{src}}(\tau_{12}^{\star},\tau_{01}^{\star})
  =\tau_{02}^{\star},
\]
with reverse and identity cases inherited from the declared source transition
table.
\end{stress}

\begin{countermodel}[Defect perturbation]
Deleting a reverse token or perturbing the composition table so that the
triple-overlap equality fails destroys the positive branch. The output is
\(\Bot\), not a target-atlas repair.
\end{countermodel}

\begin{guard}[Transition result]
The inverse and cocycle tests are passed only because the exact finite/local
source tables are supplied. They do not prove that current ontology generates
such tables for arbitrary source covers.
\end{guard}

\section{Finite-Variation Fragility}

\begin{lemma}[Bounded stability]
For every \(v\in\Varstar\), the relabeling preserves class separation,
orientation distinction, transition conjugacy, identity, inverse, and
triple-overlap composition tables. Therefore \(v\) preserves the positive
\(\Astar\) branch and \(\Stabstar(v,\Astar)\) holds.
\end{lemma}

\begin{proof}
This follows by definition of \(\Varstar\). The allowed variations are exactly
the source-token relabelings that preserve the data used by the selector,
inverse check, cocycle check, and quotient-support construction. Conjugating
the selected transitions by such a relabeling keeps the score minimum,
inverse table, composition table, and quotient identifications unchanged up
to source-token renaming.
\end{proof}

\begin{obstruction}[Arbitrary variation remains undischarged]
The bounded stability lemma is not arbitrary finite-variation robustness.
Perturbations that merge classes, create selector ties, delete inverse tokens,
or change cocycle tables lie outside \(\Varstar\) or return \(\Bot\). Thus
\(\Astar\) supplies exact finite/local robustness only under a preservation
class that is itself supplied source data.
\end{obstruction}
