\section{Source-Law Signature}

\begin{definition}[Source atlas-glue input signature]
A source atlas-glue input package is a tuple
\[
  \mathcal A =
  (I,\{U_i\}_{i\in I},\{\Germ_i\}_{i\in I},\{\Ovsrc_{ij}\}_{i,j\in I},
  \{\mathsf{Tok}_{ij}\}_{i,j\in I})
\]
where \(I\) is a source patch index set, \(U_i\) are source carriers,
\(\Germ_i\) are response-labelled source chart germs, \(\Ovsrc_{ij}\) are
source overlap predicates, and \(\mathsf{Tok}_{ij}\) are source transition
tokens. The tuple contains no target topology, target smooth atlas, target
coordinate transition, metric scale, proper time, empirical observer
protocol, benchmark solution, generated derivative, registry authority, role
authority, or validation authority.
\end{definition}

\begin{definition}[Profile semantics]
A source profile semantics clause assigns maps
\[
  \Prof_i:U_i \longrightarrow P_i^{\mathrm{src}}
\]
and compatibility predicates
\[
  \simeq^{\Pi}_{ij}\ \subseteq P_i^{\mathrm{src}}\times P_j^{\mathrm{src}}.
\]
The values of \(\Prof_i\) must be source tokens or finite source-token
tuples. They may encode source carriers, source readout syntax, response-label
syntax, retained exact-branch certificate tags, or explicitly supplied source
comparison tokens. They may not encode target coordinates or benchmark
behavior.
\end{definition}

\begin{definition}[Candidate transition family]
For each ordered overlap pair \((i,j)\), define \(\Cand_{ij}\) as the set of
partial source-token maps \(c:U_i\rightharpoonup U_j\) such that:
\begin{enumerate}
  \item \(u\in\operatorname{dom}(c)\) implies \(\Ovsrc_{ij}(u,c(u))\);
  \item \(\Prof_i(u)\simeq^{\Pi}_{ij}\Prof_j(c(u))\);
  \item the map respects the declared source readout and response-label
  syntax;
  \item the map is defined from source transition tokens only.
\end{enumerate}
\end{definition}

\begin{definition}[Selector preorder and fail-closed selection]
For each ordered pair \((i,j)\), a selector preorder
\(\preceq^{\mathrm{src}}_{ij}\) is a source-supplied preorder on
\(\Cand_{ij}\). Define
\[
  \Sel_{ij}(\Cand_{ij}) =
  \begin{cases}
    c_{ij}, & \text{if \(c_{ij}\) is the unique
    \(\preceq^{\mathrm{src}}_{ij}\)-minimal candidate,}\\
    \Bot, & \text{otherwise.}
  \end{cases}
\]
The value \(\Bot\) is a failure branch. It is not permission to choose by
target geometry, benchmark success, validation status, role authority,
registry metadata, or arbitrary convention.
\end{definition}

\section{Candidate Law}

\begin{law}[Proposal-only source-authorized atlas-glue law]
\(\MSL\) is the following proposal-only source law. Given a source atlas-glue
input package \(\mathcal A\), the law supplies:
\begin{enumerate}
  \item profile semantics \((\Prof_i,\simeq^\Pi_{ij})\);
  \item candidate transition families \(\Cand_{ij}\);
  \item selector preorders \(\preceq^{\mathrm{src}}_{ij}\) and selected maps
  \(\Tau_{ij}=\Sel_{ij}(\Cand_{ij})\);
  \item a transition algebra with source identity maps
  \(\mathsf{id}^{\mathrm{src}}_i\), inverse operation
  \(\mathsf{inv}_{\mathrm{src}}\), and composition operation
  \(\mathsf{comp}_{\mathrm{src}}\);
  \item an admissible finite-variation class \(\Var\);
  \item a preservation predicate \(\Stab\) asserting that profile separation,
  selected transitions, inverse compatibility, and cocycle compatibility are
  stable under every variation in \(\Var\).
\end{enumerate}
Every item is source-side and every failure to satisfy a required clause
returns \(\Bot\) rather than importing target or benchmark structure.
\end{law}

\begin{assumption}[Inverse clause]
When \(\Tau_{ij}\neq\Bot\) and \(\Tau_{ji}\neq\Bot\), the source inverse
clause requires
\[
  \mathsf{inv}_{\mathrm{src}}(\Tau_{ij})=\Tau_{ji}
\]
on the common source domain where both maps are defined.
\end{assumption}

\begin{assumption}[Cocycle clause]
When \(\Tau_{ij}\), \(\Tau_{jk}\), and \(\Tau_{ik}\) are all defined on a
triple source overlap, the source cocycle clause requires
\[
  \mathsf{comp}_{\mathrm{src}}(\Tau_{jk},\Tau_{ij})=\Tau_{ik}
\]
on that overlap. This is a source-token equality and not an imported smooth
atlas theorem.
\end{assumption}

\begin{assumption}[Finite-variation preservation clause]
For every \(v\in\Var\), let \(v(\mathcal A)\) be the source-side perturbation
of the input package when it is defined. The preservation predicate
\(\Stab(v,\mathcal A)\) requires:
\begin{enumerate}
  \item profile compatibility decisions are preserved or fail closed;
  \item selector minima remain unique or fail closed;
  \item inverse clauses are preserved or fail closed;
  \item cocycle clauses are preserved or fail closed;
  \item no target or benchmark datum is introduced by the perturbation.
\end{enumerate}
\end{assumption}

\section{Conditional Payload}

\begin{lemma}[Source transition family is determined relative to supplied law data]
If \(\Cand_{ij}\) and \(\preceq^{\mathrm{src}}_{ij}\) are supplied and
\(\Sel_{ij}(\Cand_{ij})\neq\Bot\), then \(\Tau_{ij}\) is unique relative to
the supplied source law data.
\end{lemma}

\begin{proof}
The selector returns a transition only when a unique
\(\preceq^{\mathrm{src}}_{ij}\)-minimal candidate exists. Uniqueness is
therefore relative to the supplied source preorder and candidate family. If
two candidates are tied or incomparable at the minimal level, the selector
returns \(\Bot\). No non-source authority is available to break the tie.
\end{proof}

\begin{theorem}[Conditional repaired source quotient support]
Assume the proposal-only law data satisfy profile compatibility, selector
uniqueness, inverse compatibility, cocycle compatibility, and finite-variation
preservation on the required overlap family. Define
\[
  D=\bigsqcup_{i\in I}(\{i\}\times U_i)
\]
and generate a relation \(\sim_{\mathrm{src}}\) by
\[
  (i,u)\sim_{\mathrm{src}}(j,v)
  \quad\text{when}\quad
  \Tau_{ij}(u)=v.
\]
Then the reflexive, symmetric, and transitive closure of
\(\sim_{\mathrm{src}}\) defines a repaired source quotient support
\[
  \Qsrc=D/{\sim_{\mathrm{src}}}.
\]
This support is well-defined relative to the proposal-only law data. It is not
by itself \(\Msrc\) adoption, not a target manifold, not a smooth atlas, and
not a downstream metric result.
\end{theorem}

\begin{proof}
By the selector lemma, every selected transition is unique relative to its
source candidate family and source preorder. The inverse clause makes pairwise
identification direction-independent on declared overlaps. The cocycle clause
makes compatible triple-overlap paths agree at the source-token level. Thus
the generated relation has a well-defined equivalence closure on the disjoint
source support. Taking the quotient produces \(\Qsrc\).

The construction uses only the source profile maps, source transition-token
families, selector preorders, inverse and composition operations, and
finite-variation preservation clauses. It does not use target atlas structure,
target metric data, benchmark success, validation status, role authority,
registry metadata, or generated derivatives as premises. Therefore the result
is a conditional source quotient support relative to the proposal-only law
data and not an adopted \(\Msrc\).
\end{proof}
